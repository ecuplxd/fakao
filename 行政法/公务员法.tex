\section{公务员法}
\begin{enumerate}
  \item 1 道题。
  \item 公务员。依法履行公职、纳入国家行政编制、由国家财政负担工资福利的工作人员。
        俗称体制内的人。\zd{沾公字的}。排除法。盈利性质、事业单位不是。没有直接考点。
        \begin{enumerate}
          \item \zd{从事公务}。\al{务工人员、担任人大代表的演员、辅警 \cuo}。
          \item \zd{3 重编码}。职务、职级 | 级别。
          \item 职务。\zd{职务（领导）和职级（非领导）}。\yi 可以转任、兼任。\er 能上能下。\bs{长、爽、挫}。
                \begin{enumerate}
                  \item \zd{领导}。带“长”字。\al{国家级、省部级、厅局级、县处级、乡科级}。
                  \item 非领导。职级是新叫法。
                        \begin{enumerate}
                          \item 爽。巡视员 \dy 调研员。\al{一、二级巡视员；一、二、三、四级调研员}。
                          \item 挫。主任科员 \dy 科员。\al{一、二、三、四级主任科员；一、二级科员}。
                        \end{enumerate}
                \end{enumerate}
          \item （工资）级别。10 级，分正、副职。
          \item \zd{一职数级，上下交叉。三者的对应关系，由国家规定}。
        \end{enumerate}
  \item 进。上船，录用制度（\bs{委、聘}）。考的最多。背诵为主。公管 = \zd{公务员管理/主管部门组织}。
        \begin{enumerate}
          \item 委任制。\zx{录用考公、对象挫男；犯开失信，不得录用；特殊简化，省来批准；体检为人；选拔领导}。
                \begin{enumerate}
                  \item \zd{录用}委任制 = \zd{考公}务员（分\zx{国考、省考}）。两步走。录用 \yjt 1 年试用期 \yjt 委任。
                        \begin{enumerate}
                          \item 对象。主任科员以下（挫）。
                          \item 不得录用。\zx{范开失信}。犯过罪受过刑罚、开除共产党党籍、开除公职、失信惩戒名单。
                          \item 特殊岗位。\zx{流程可以简化，找省（公管）批准}。
                          \item 体检。\zx{卫生 + 人事}部门共同规定。
                          \item 试用期 1 年。不合格，\zx{取消录用。vs 不予录用}（没拿到 Offer，从始自终没公务员身份）。
                                \al{啊哈哈。不说不予录用，而是说取消其公务员录用自己}。
                        \end{enumerate}
                  \item \zd{选拔}委任制/公开选拔。只适用\zd{领导}职务（其他方式脚踏两只船）。实践中的空降干部。
                \end{enumerate}
          \item 聘任制。利益交换，花钱买服务。限定\zd{专业、辅助类}职位。
                \zx{专业、辅助类，涉密不能聘；考试或选聘，省公管批准，合同同人备；协议工资制，办法中央定；试用小 1 年、聘期 1 - 5；纠纷走劳仲}。
                \begin{enumerate}
                  \item \zd{涉密不能聘}。花钱买过来的不放心。\al{某县\xhx{保密局}聘任两名负责保密工作的程序员 \cuo}。
                  \item 方式。考试 \hz 直接选聘。\al{卫建委专家组组长}。
                  \item \zd{省（公管）批准}。\al{某县公安聘任网管的合同需经上级公安批准 \cuo}。
                  \item 合同。\zd{同级公务员主管部门（人事）备案}。性质 \ydy 劳动合同、发生纠纷劳动仲裁 \yjt 民事诉讼。
                  \item 协议工资。具体办法中央（公管）规定。
                  \item 试用期 1 - 12 月，聘期 1 - 5 年。
                \end{enumerate}
        \end{enumerate}
  \item 转。交流制度。岗位变动。\zx{内转社调}。
        \begin{enumerate}
          \item 转任。行政机关内转圈圈。
                \begin{enumerate}
                  \item 内部转任。\al{考务处到命题处}。
                  \item 外部平级转任。\al{土地局到环保局}。
                  \item \zd{遴选}。下到上。\al{遴选法官}。
                \end{enumerate}
          \item 调任。\zd{长和爽}。\al{学而优则仕}。
          \item \zx{挂职/挂职锻炼。不再归为交流制度}。临时镀金的。
        \end{enumerate}
  \item 出。下船，退出制度。
        \begin{enumerate}
          \item 辞职 = \zd{辞去公职}。
                \begin{enumerate}
                  \item \zx{不得辞职 = 需本人继续处理} + 没完成的主要/重要工作。
                  \item vs \zx{引咎辞职} = 行政问责。\zx{辞的是领导职务}，还有公务员身份。\al{辞去公职 \cuo}。
                \end{enumerate}
          \item 辞退。\zx{可以享受失业保险}。
                \begin{enumerate}
                  \item 应当辞退。能力不行，态度不好。
                        \begin{enumerate}
                          \item \zx{连续 2 年考核不称职}。\al{甲 1 年考核不称职，林业局可以此为由辞退甲 \cuo}。
                          \item \zd{连续旷工 15 日，1 年累计 30 日}。
                          \item 不能胜任现任工作。
                          \item 大错不犯，小错不断。
                        \end{enumerate}
                  \item 不得辞退。\zd{因公}致残、医疗期内、特殊情况妇女。
                  \item vs 开除。开除是处罚制度，严重违纪。辞退没那么恶劣。
                \end{enumerate}
          \item 退休。
                \begin{enumerate}
                  \item 法定条件。到年纪没能力。\al{如完全丧失工作能力}。
                  \item 提前退休、自愿退休。\zx{工 3 男 6 女 55}。工龄满 30 年 \hz 男性 60，女性 55。
                \end{enumerate}
        \end{enumerate}
  \item 权利义务。
        \begin{enumerate}
          \item \zd{接受/参加培训是权利}。\zx{前面加上应当二字，说的通 \yjt 义务，说不通 \yjt 权利。原理。法理学}。
          \item \zx{回避制度}。执行回避（近亲属）、地域回避（不能做\zx{市县乡}主要领导职务）、\zd{近亲属回避}。
                \begin{enumerate}
                  \item 防止裙带关系。
                  \item \zx{同一机关隶属于同一领导、直接上下级}关系、\zx{管人管钱管监督}不能都占着。
                        \al{甲是公安局局长，其妻在公安局所属派出所担任户籍警察 \yjt 不用回避}。
                  \item \zx{行业对口，损公肥私}。不得担任\zd{行业监管、主管部门领导}。下左右。\al{夫旅游局长，妻开旅游社}。
                \end{enumerate}
          \item 管理制度。
                \begin{enumerate}
                  \item 任职禁止。例外。学术兼职 = 工作需要 + 经过批准 + \zx{不能领取报酬}。\zx{不需要公告}。
                  \item 离职禁止。\zx{非 2 领 3 仅休职}。非领导 2 年、领导 3 年。仅适用退休 \hz 辞职的。
                \end{enumerate}
        \end{enumerate}
  \item 考核。平时考核、专项考核、\zd{定期考核}。\zx{负责人 \hz 授权的考核委员会确定}。
        \begin{enumerate}
          \item \zx{1 年为期}。
          \item 结果分为\zd{优秀、称职、基本称职和不称职} 4 档。\zx{属于人事处理}。
          \item \zx{优秀、称职 \yjt 能拿年终奖}。
          \item \zx{1 年不称职 \yjt 降低一个职务 \hz 职级}。
          \item \zx{只有不称职才能复核 \yjt 申诉}。
        \end{enumerate}
  \item 处分制度。\zd{严重违法违纪}。重点。
        \begin{enumerate}
          \item 6 种 2 类。\zx{其他无中生有}。\al{责令辞职、免职 \cuo{属于其他追责方式}}。
                \begin{enumerate}
                  \item \bs{面子罚}。\zx{职务、职级、级别无影响}。\zx{警告 6、记过 12、记大过 18}。\al{应当按照规定\xhx{降低甲级别} \cuo}。
                  \item 实质罚。\zx{降级 24、撤职 24、开除}。
                  \item \zx{以 6 为基，按 6 来加；最高 24，并上 48；重吸收轻}。并处上限 48 个月。
                  \item \al{某负责人甲因同时违反纪律和玩忽职守被分别给予\xhx{撤职和记过}处分}。
                  \item \al{应只对甲进行撤职处分 \dui}。
                  \item \al{应同时降低甲的级别 \dui}。
                  \item \al{对甲的处分期为 36 个月 \cuo}。
                \end{enumerate}
          \item 程序。
                \begin{enumerate}
                  \item 初步调查。负责人批准、\bs{暂停工作有要求，不宜继续为前提}、不能出国。
                  \item 讨论决定。负责人\zd{集体 + 书面通知} + \bs{作出生效}。\al{送达生效 \cuo}。
                  \item \zx{既说且做减轻处分，只占一条从轻}。既说 = 主动交代。且做 = 有效避免。
                \end{enumerate}
          \item 救济。复核 \hz \zd{对口的上级人事}申诉（行政内部行为的一种救济方式）。
          \item 解除。
                \begin{enumerate}
                  \item \zx{到期自动解除} = 表现好 + 无违法违纪。
                  \item \zx{除警告外，处分期间不能晋升待遇}。
                  \item \zx{职务、级别不能自动恢复}，只能慢慢升回去。\al{解除对甲的身份后，即应恢复其职务 \cuo}。
                \end{enumerate}
          \item vs \zd{政务处分}。监察机关监察委员会。\zx{只罚一次，先到先罚}。\bs{做题注意区分是哪种处分}。
        \end{enumerate}
\end{enumerate}
