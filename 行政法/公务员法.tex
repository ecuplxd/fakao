\section{公务员法}
\begin{enumerate}
  \item 1 道题。
  \item 体制内的人。沾公字的。排除法：盈利性质、事业单位不是。没有直接考点。
  \item 依法履行公职、纳入国家行政编制、由国家财政负担工资福利的工作人员。
        \begin{enumerate}
          \item \zd{从事公务}。\al{务工人员、担任人大代表的演员、辅警 \cuo}。
          \item \zd{3 重编码}。职务、职级 | 级别。
          \item 职务。\zd{领导职务和职级（非领导）可以转任、兼任}，能上能下。
                \begin{enumerate}
                  \item \zd{领导}（带长字）。\al{国家级、省部级、厅局级、县处级、乡科级}。
                  \item 非领导（职级是新叫法）。巡视员 > 调研员（前俩，爽）> 主任科员以下（挫）。
                        \al{一级巡视员、二级巡视员、一级调研员、二级调研员、三级调研员、四级调研员、一级主任科员、二级主任科员、三级主任科员、四级主任科员、一级科员、二级科员}。
                \end{enumerate}
          \item 级别。10 级，分正副职。
          \item （工资）级别。
          \item \zd{一职数级，上下交叉}。\zd{职务、职级、级别的对应关系，由国家规定}。
        \end{enumerate}
  \item 进。上船，录用制度。考的最多。背诵为主。公管 = \zd{公务员管理/主管部门组织}。
        \begin{enumerate}
          \item 委任制。
                \begin{enumerate}
                  \item \zd{录用}委任制 = \zd{考公}务员（分国考省考，没考过）。两步走。录用 \yjt 1 年试用 \yjt 委任。
                        \begin{enumerate}
                          \item 对象。主任科员以下（挫）。
                          \item 不得录用。\zd{犯过罪受过刑罚、开除共产党党籍、开除公职、失信惩戒名单}。
                          \item 特殊岗位。\zd{可以简化流程，省公管批准}。
                          \item 体检。\zd{人事 + 卫生}部门共同规定。
                          \item 试用期。1 年。不合格，取消录用。vs \zd{不予录用}（发生在录用过程中，从始自终没公务员身份）。
                        \end{enumerate}
                  \item \zd{选拔}委任制（公开选拔）。只适用\zd{领导}职务（其他方式脚踏两只船）。实践中的空降干部。
                \end{enumerate}
          \item 聘任制。限定\zd{专业、辅助}类职位。利益交换，花钱卖服务。
                \begin{enumerate}
                  \item \zd{涉密不能聘}。花钱买过来的不放心。
                  \item 方式。考试 \hz 直接选聘。\al{卫建委专家组组长}。
                  \item \bs{省公管批准}。
                  \item 合同。\zd{同级人事备案}。性质 \ydy 劳动合同、发生纠纷仲裁 \yjt 民事诉讼。
                  \item 协议工资。具体办法中央公管规定。
                  \item 试用期 1 - 12 月，聘期 1 - 5 年。
                \end{enumerate}
        \end{enumerate}
  \item 转。交流制度。岗位变动。\zd{内转社调}。
        \begin{enumerate}
          \item 转任。行政机关内转圈圈。
                \begin{enumerate}
                  \item 内部转任。\al{考务处 \yjt 命题处}。
                  \item 外部平级转任。\al{土地局 \yjt 环保局}。
                  \item \zd{遴选}。下到上。
                \end{enumerate}
          \item 调任。\zd{外到内，长和爽}。\al{学而优则仕}。
          \item \scx{挂职/挂职锻炼。不再归为交流制度。临时镀金的}。
        \end{enumerate}
  \item 出。下船，退出制度。
        \begin{enumerate}
          \item 辞职 = \zd{辞去公职}。
                \begin{enumerate}
                  \item \zd{需本人继续处理没完成的主要/重要工作的，不得辞职}。
                  \item vs \zd{引咎辞职} = 行政问责。辞的是领导职务，不是公职，还有公务员身份。
                \end{enumerate}
          \item 辞退（\zd{可以享受失业保险}）。
                \begin{enumerate}
                  \item 应当辞退。\zd{能力不行，态度不好}。
                        \begin{enumerate}
                          \item \zd{连续 2 年考核不称职}。
                          \item \zd{连续旷工 15 天，1 年累计 30 天}。
                          \item 不能胜任现任工作。
                          \item 大错不犯，小错不断。
                        \end{enumerate}
                  \item 不得辞退。\zd{因公}致残、医疗期内、特殊情况妇女。
                  \item vs \zd{开除}。开除是处罚制度，严重违纪。辞退没那么恶劣。
                \end{enumerate}
          \item 退休。
                \begin{enumerate}
                  \item 法定条件。到年纪没能力。如\al{完全丧失工作能力}。
                  \item 提前退休、自愿退休。工龄满 30 年 \hz 男性 60，女性 55。\zd{工 3 男 6 女 55}。
                \end{enumerate}
        \end{enumerate}
  \item 权利义务。
        \begin{enumerate}
          \item \zd{接受/参加培训是权利！！！！！}
          \item 回避制度。执行回避（近亲属）、地域回避（不能做\zd{市县乡主要领导职务}）、\zd{近亲属}。
                \begin{enumerate}
                  \item 防止裙带关系。
                  \item \zd{同一机关隶属于同一领导}、\zd{直接上下级关系}（上一级）、\zd{管人管钱管监督}不能都占着。
                  \item \zd{行业对口，损公肥私}。不得担任\zd{行业监管、主管部门领导成员}。\zd{配}偶、\zd{子}女、\zd{兄}弟。\al{丈夫旅游局长，妻子开旅游社}。
                \end{enumerate}
          \item 管理制度。
                \begin{enumerate}
                  \item 任职禁止。例外。学术兼职 = 工作需要 + 经过批准 + \zd{不能领取报酬}。\zd{不需要公告}。
                  \item 离职禁止。\zd{非}领导 \zd{2} 年，\zd{领}导 \zd{3} 年，\zd{仅}适用退\zd{休} \hz 辞\zd{职}的。
                \end{enumerate}
        \end{enumerate}
  \item 考核。平时考核、专项考核、\zd{定期考核}。
        \begin{enumerate}
          \item 定期考核 \zd{1} 年为期。
          \item 定期考核的结果分为\zd{优秀、称职、基本称职和不称职} 4 档。
          \item 优秀、称职 \yjt 能拿年终奖。1 年不称职 \yjt 降职。\zd{只有不称职才能申诉}。
        \end{enumerate}
  \item \zd{处分制度}。严重违法违纪。重点。
        \begin{enumerate}
          \item 6 种 2 类。警告、记过、记大过（面子罚，级别、职务、职级\zd{无影响}）| 降级 24、撤职 24、开除。
          \item \zd{以 6 为基，按 6 来加，最高 24} 个月，并处\zd{上限 48} 个月。\zd{重吸收轻}。
          \item 初步调查。负责人批准；\zd{暂停履行职务有要求，不宜继续为前提}；不能出国。
          \item 讨论决定。负责人\zd{集体} + \zd{书面通知 + 作出生效}（\al{送达生效 \cuo}）。
          \item 既说（主动交代）且做（有效避免）\zd{减轻}处分，只占一条\zd{从轻}。
          \item \zd{除警告外，处分期间不能晋升待遇}。
          \item \zd{到期自动解除 2 个条件} = 表现好 + 无违法违纪。
          \item \zd{职务、级别不能自动恢复}，只能慢慢升回去。
          \item 救济。复核 \hz \zd{对口的上级人事}申诉。
          \item vs 政务处分。监察机关监察委员会。只罚一次，先到先罚。\zd{做题注意区分是哪种处分}。
        \end{enumerate}
\end{enumerate}
