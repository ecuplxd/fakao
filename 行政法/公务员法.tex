\section{公务员法}
\begin{enumerate}
  \item 一道题。
  \item 体制内的人。沾公字的。排除法：盈利性质、事业单位不是。没有直接考点。
  \item \qd{依法履行公职、纳入国家行政编制、由国家财政负担工资福利的工作人员}。
        \begin{enumerate}
          \item 从事\zd{公务}。演员人大代表，辅警就不是。\qd{排除务工人员}。
          \item \zd{三重编码}。职务、职级、级别。
          \item 职务。\zd{领导（带长字）}和非领导。十级领导职务，分正副职。\al{国家级、省部级、厅局级、县处级、乡科级}。
          \item 职级（旧称非领导职务）。巡视员 > 调研员（前俩，爽）> 主任科员以下（挫）。官大官小。
                \al{一级巡视员、二级巡视员、一级调研员、二级调研员、三级调研员、四级调研员、一级主任科员、二级主任科员、三级主任科员、四级主任科员、一级科员、二级科员}。
          \item \zd{领导职务和职级可以转任兼任}，能上能下。
          \item （工资）级别。\zd{一职数级，上下交叉}。\al{公务员的领导职务、职级应当对应相应的级别。公务员领导职务、职级与级别的对应关系，由国家规定}。
        \end{enumerate}
  \item 进。上船，录用制度。考的最多。背诵为主。
        \begin{enumerate}
          \item 录用委任制。\zd{考公务员}。名字来源：录用 \yjt 试用 \yjt 委任。
                \begin{enumerate}
                  \item 对象挫：主任科员以下。
                  \item 公管 = \zd{公务员管理/主管部门组织}，分国考省考。没考过。
                  \item \zd{犯过罪受过刑、开除共产党党籍、开除公职、失信惩戒名单}不得录用。
                  \item 特殊岗位可以\zd{简化流程，省公管批准}。
                  \item 体检\zd{人事 + 卫生部门}共同规定。
                  \item 1 年试用期，不合格，\zd{取消录用}。\zd{不予录用}（录用过程，本来就没身份）。
                \end{enumerate}
          \item 聘任制。\zd{就只限定专业、辅助}类职位。\qd{利益交换，花钱卖服务}。
                \begin{enumerate}
                  \item \zd{省公管批准}。
                  \item \zd{涉密不能聘任}。花钱买过来的不放心。
                  \item 协议工资，具体办法中央公管规定，\ydy 劳动合同；发生纠纷仲裁 \yjt 民诉。
                  \item 合同\zd{同级人事备案}。试用期 1 - 12 月，聘期 1 - 5 年。
                  \item 方式。考试 \hz 直接选聘。领导，非领导均可聘任。
                \end{enumerate}
          \item 公开选拔（选拔委任）。空降干部。只适用\zd{领导职务，其他脚踏两只船}。
        \end{enumerate}
  \item 转。交流制度。岗位变动。
        \begin{enumerate}
          \item 转任。行政机关内转圈圈。
                \begin{enumerate}
                  \item 内部转任。考务处 \yjt 命题处。
                  \item 外部平级转任。土地局 \yjt 环保局。
                  \item \zd{遴选}。下到上。
                \end{enumerate}
          \item 调任。\zd{外到内，长和爽}。
          \item \scx{挂职/挂职锻炼不再归为交流制度}。镀金，临时的。
        \end{enumerate}
  \item 出。下船，退出制度。
        \begin{enumerate}
          \item 辞职 = \zd{辞去公职}。
                \begin{enumerate}
                  \item 不得辞职的。主要/重要工作没处理完 + \zd{需要本人继续处理}（人身专属性极强）。
                  \item \zd{引咎辞职辞的是领导职务，不是公职}，还有公务员身份。是行政问责。
                \end{enumerate}
          \item 退休。
                \begin{enumerate}
                  \item 法定条件。到年纪没能力（完全丧失工作能力）。
                  \item 提前退休、自愿退休。工龄满 30 年。男性 60，女性 55。
                \end{enumerate}
          \item 辞退。
                \begin{enumerate}
                  \item \zd{\bdy 开除}。开除是处罚制度，严重违纪，辞退没那么恶劣。
                  \item 应当辞退：\zd{能力不行，态度不好}。
                        \begin{enumerate}
                          \item \zd{连续 2 年考核不称职}。
                          \item 不能胜任现任工作。
                          \item 大错不犯，小错不断。冥顽不化。
                          \item 连着旷工 15 天，1 年累积 30 天。
                        \end{enumerate}
                  \item 不得辞退：因公致残、医疗期内、特殊情况妇女。
                  \item \zd{可以享受失业保险}。
                \end{enumerate}
        \end{enumerate}
  \item 权利义务。
        \begin{enumerate}
          \item \zd{接受/参加培训是权利！！！！！}
          \item 回避制度。执行（近亲属）、地域（不能做地方主要领导职务，市县乡）、\zd{近亲属}；
                \begin{enumerate}
                  \item 防止裙带关系。
                  \item 同一机关隶属于同一领导；\zd{直接上下级}关系（上一级）；管人管钱管监督不能都占着。
                  \item 行业对口，损公肥私：不得担任\zd{行业监管主管部门领导成员}。配偶子女兄弟。如：丈夫旅游局长，妻子开旅游社。
                \end{enumerate}
          \item 管理制度。
                \begin{enumerate}
                  \item 任职禁止。例外：工作需要 + 经过批准 + \zd{不能领取报酬}。
                  \item 离职禁止。3 年（领导）\hz\, 2 年（非领导）。\zd{适用对象：辞职 \hz 退休}。
                  \item 兼职情况\zd{不需要公告}。
                \end{enumerate}
        \end{enumerate}
  \item 考核：平时考核、专项考核、定期考核：
        \begin{enumerate}
          \item \zd{1} 年。
          \item 定期考核的结果分为优秀、称职、基本称职和不称职四个等次。
          \item 1 年不称职降职。优秀、称职能拿年终奖。
          \item \zd{只有不称职才能申诉}。
        \end{enumerate}
  \item \zd{处分制度（重点）}。严重违法违纪。
        \begin{enumerate}
          \item 5 种 2 类。警告、记过、记大过（面子罚 = 工资、级别、职务、职级\zd{无影响}）| 降级、撤职 24、开除。重吸收轻。
          \item \zd{处分期间不能晋升待遇，例外警告}。
          \item \zd{以 6 为基数，按 6 来加}，最高 24 个月，并处上限 48 个月。
          \item 初步调查。负责人批准、不宜继续可暂停履行职务、不能出国。
          \item 负责人们\zd{集体}讨论决定 + \zd{书面通知}，\zd{作出之日生效}而非送达。
          \item 政务处分。监察机关监察委员会。罚一次。先到先罚。\zd{做题注意区分是行政处分还是政务处分}。
          \item 既说（主动交代）且做（有效避免）\zd{减轻}处分，只占一条\zd{从轻}。
          \item 处分\zd{自动解除} = 表现好 + 无违法违纪 + 到期。
          \item \zd{职务、级别不能自动恢复}，只能慢慢升回去。
        \end{enumerate}
  \item 救济。复核 \hz \zd{对口的上级人事}申诉。
\end{enumerate}
