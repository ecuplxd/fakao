\section{行政许可}
\begin{enumerate}
  \item 学习脉络。概念、种类、立法（设定、具体规定）、执行（主体。自己、授权、委托、集中；程序）。
  \item \zd{一般禁止的解除 \yjt 具有主体资格}。区分行政审批、行政确认。
  \item 特征。依申请、授益、要式、外部性（许可一定是冲着外部的）。
  \item 分类。
        \begin{enumerate}
          \item 法律分类。
                \begin{enumerate}
                  \item 一般许可。形式审查 1 人，实质审查 2 人。\al{驾照}。
                  \item 特许。分配有限的自然资源 \hz 公共资源，优中选优。招投标方式。
                        \begin{enumerate}
                          \item \al{授予甲公司城市管道燃气独占专营权 \dui}。
                          \item \al{城市临时建筑许可证属于关系公共利益的特定行业准入类特许 \cuo}。
                          \item vs 行政协议的特许经营。双方，前者是单方给予。给予恩赐。
                        \end{enumerate}
                  \item 认可。对相对人的资质能力、条件进行评价，允许其从事特定活动。考试。\al{医师执业}。
                  \item 核准。对物、2 人核对。\al{电梯检测合格证}。
                  \item （初始）登记。社会团体、企业法人。实质审查。\al{颁发工商营业执照}。
                \end{enumerate}
          \item 附期限、不附期限。附期限延期提前 30 日 \yjt 第一次/后面 \yjt 没有答复 \yjt 视为默示拒绝/同意。
                \begin{kk}{延期}
                  \begin{enumerate}
                    \item 许可有效期届满 \zd{30 日}前，\zd{向作出}许可决定的行政机关提出申请。
                    \item 但法律、法规、规章\zx{另有规定}的，\zd{从其}规定。
                    \item 在该许可有效期\zd{届满前作出}是否准予延续的决定 \yjt \zd{视为}准予延续。
                  \end{enumerate}
                \end{kk}
          \item 有数量限制、无数量限制。原则根据受理顺序，特许是例外。
          \item 可转让、不可转让。一般不能转让，例外。采矿证之类的准物权。
        \end{enumerate}
  \item 立法权限。
        \begin{enumerate}
          \item 设定。从无到有。
                \begin{enumerate}
                  \item 事项范围。\zd{涉及 xx 可以设定行政许可}。
                        \begin{enumerate}
                          \item 可以立法的。特许、\zd{自然资源}、公共资源、公共利益等。
                          \item 可以不立法的。市场调节、行业自律、自主决定（\al{谈恋爱}）、事后监督。没考过。
                        \end{enumerate}
                  \item 经常性许可。稳定、日常存在的。+ 法规。
                  \item 临时性许可。
                        \begin{enumerate}
                          \item 全国 \yjt 国务院决定（性质 = 国务院的其他规范性文件）。不是国务院部门决定！！
                          \item 地方 \yjt 省规章、本级人大/人常、有效期最多 1 年。
                        \end{enumerate}
                  \item \zx{地方性法规的限制性条件}。\bs{地保、国统、企登}。地方保护主义、国家统一的资质许可、\zx{企业设立登记及其前置性许可}。
                  \item 评价/立法后评估。设定机关应当定期评价，实施机关可以适时评价。
                  \item 停止。经济事务的许可，省政府决定 \yjt 国务院批准。
                \end{enumerate}
          \item \zd{具体规定}。\zx{在法律、行政法规设定的行政许可事项范围内}。从粗到细，下位法可以细化上位法，但是不能抵触上位法。\zd{不得增设许可，不得增设条件}。
                \begin{enumerate}
                  \item \al{办理认证手续 \yjt 抵触上位法《食品安全法》}。
                  \item \al{收取的费用按一定比例返还 \yjt 上缴国库}。
                  \item \al{缴纳特别管理费 \yjt 地方保护主义}。
                  \item \al{需安装建设主管部门指定的节能设施 \yjt 不得提出购买指定商品、接受有偿服务等不正当要求}。
                \end{enumerate}
        \end{enumerate}
  \item 实施。谁发许可证，怎么发。\bs{合法性 \yjt 主体、内容、程序考虑}。
        \begin{enumerate}
          \item 主体。具有行政许可权的行政机关。
                \begin{enumerate}
                  \item 授权。+ 法规。
                  \item 委托。只能委托其他行政机关、不能转委托、要公告。
                  \item 集中。一个行政机关行使有关行政机关的 xx。省政府决定 \yjt 国务院批准。
                        \begin{enumerate}
                          \item 一个机关发一个证。一个窗口对外。
                          \item 多个机关的多个许可。高效便民。权力所有权是否转移。
                                \begin{itemize}
                                  \item 集中许可。独揽大权。以新机关为被告。
                                  \item 一站式审批/联合许可。把所有的机关放在一个大楼里面。\al{一站式服务大厅}。
                                  \item 并联式审批/统一许可。主办窗口就是一个收材料的。
                                \end{itemize}
                        \end{enumerate}
                \end{enumerate}
          \item 总结。\bs{设定、授、费 + 法规； 委托只能靠兄弟； 停止、集中省找国}。
          \item \bs{一般程序}。\al{驾照、法考}。
                \begin{enumerate}
                  \item 书面申请。\zd{材料反映真实情况、对真实性负责}。
                        \begin{enumerate}
                          \item 格式文本。非必须、不收费。\al{申请表由 2 元降为 1 元 \cuo}。例外。报名费。法律、行政法规规定。
                          \item \zd{可以委托}。高度人身属性必须本人。\al{护照}。
                        \end{enumerate}
                  \item 受理。
                        \begin{enumerate}
                          \item 材料。\yi 有错当场改。\er 不合要求。当场补正告知书 \hz 5 日内一次告知。过程性行为不可诉。
                          \item \zd{说明、解释的义务}。
                          \item 出具\zd{盖章 + 日期的书面凭据}。
                        \end{enumerate}
                  \item 审查。实质审查。\zd{关系他人重大利益 \yjt 告知 \yjt 陈述申辩}。
                  \item （书面）\zd{决定}。
                        \begin{enumerate}
                          \item 准予 \yjt 颁发，\zd{加盖标签、印章}；要\zd{公开，供公众查阅}。
                          \item 不予（包括事后撤销的） \yjt \zd{说明理由、告知救济权利、不公开}。
                        \end{enumerate}
                  \item 法律、行政法规、国务院设定的许可全国有效。例外。自我设限。\al{法考 C 证}。
                \end{enumerate}
          \item 听证程序。不收费。
                \begin{enumerate}
                  \item 依申请、依职权 = \zx{公共利益 + 公告}。
                        \begin{enumerate}
                          \item \yi 不要看到重大利益就脑子一热。\er 依申请的肯定不重大，如果重大那都用不着他们申请。
                          \item \al{涉及申请人与\xhx{他人}之间重大利益关系的，申请人、利害关系人\xhx{提出}听证\xhx{申请}的，应予以公告 \cuo}。
                          \item \al{\xhx{规划局}发放的许可证，直接涉及申请人和\xhx{附近居民}之间的\xhx{重大利益}关系 \dui}。
                        \end{enumerate}
                  \item 5207。5 日内申请，20 日把会开起来，7 日前告知。
                  \item 回避。近亲属、调查过的人不能做主持人。
                  \item 笔录。排他效力，是后续决定的唯一根据/依据。\al{应当参考 \cuo。应当根据 \dui}。
                \end{enumerate}
          \item 监督检查。
                \begin{enumerate}
                  \item 实地检查。发现违法情况，责令改正（性质是强制措施）。
                        \begin{enumerate}
                          \item 一般限期改，核准立即改。\al{电梯安全隐患}。
                          \item 特别严重吊执照。vs 撤销。本来就违法。吊销是事后违法。
                        \end{enumerate}
                \end{enumerate}
          \item 合法撤回、违法撤销、不用都注销。
                \begin{enumerate}
                  \item 许可的撤销。\yi 决定机关、上一级决定机关、被越权机关、法院有权。\er 未对具体程序作出规定。
                  \item 许可的\zd{撤回/废止}。\bs{因公共利益发生变化需要变更或者撤回已经生效的许可}。
                        \begin{enumerate}
                          \item 要先行处理的。行政机关 \yjt 法院。
                          \item \zd{给予补偿}。\yi 有标准 \yjt \zd{实际损失（成本 + 合理利润）}。\er 特许 \yjt 只补\zd{实际投入}。
                          \item vs 注销。无法继续从事许可活动。证不能用了都要注销。注销就是注销。
                                \begin{itemize}
                                  \item \al{甲取得律师执业证书后，发生交通事故成为\xhx{植物人}}。
                                  \item \al{甲违法经营的网吧\xhx{被吊销}许可证}。
                                  \item \al{甲通过行贿取得行政许可后，被行政机关发现并\xhx{撤销}其许可}。
                                \end{itemize}
                        \end{enumerate}
                \end{enumerate}
        \end{enumerate}
  \item \zd{前置审批}。
\end{enumerate}
