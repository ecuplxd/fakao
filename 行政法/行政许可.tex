\section{行政许可}
\begin{enumerate}
  \item 学习脉络。概念、种类、立法（设定、具体规定）、执行（主体。自己、授权、委托、集中；程序）。
  \item \zd{一般禁止的解除}。区分行政审批、行政确认。
  \item 特征。依申请、授益、要式、\zd{外部性}（内外看身份，一定是冲着外部的）。
  \item 分类。
        \begin{enumerate}
          \item 法律分类。
                \begin{enumerate}
                  \item 一般许可。\zd{形式审查 1 人，实质审查 2 人}。
                  \item 特许。赚大钱，优中选优。\zd{招投标}。
                  \item 认可。对相对\zd{人}的资质能力条件进行评价，允许从事特定活动。\zd{要考试}。
                  \item 核准。对\zd{物}，电梯检测合格证。\zd{2 人核对}。
                  \item 登记。社会团体、企业法人。实质审查 2 个人。
                \end{enumerate}
          \item 不附期限、附期限。附期限延期 \yjt \zd{第一次/后面，没有答复，视为默示拒绝/同意}。
          \item 有数量限制、没有数量限制。\zd{原则根据受理顺序，例外特许}。
          \item 可转让、不可转让。\zd{一般不能转让}，例外：采矿证之类的准物权。
        \end{enumerate}
  \item 立法权限。
        \begin{enumerate}
          \item 设定。从无到有。
                \begin{enumerate}
                  \item 事项范围。啥事需要立法。
                        \begin{enumerate}
                          \item 可以立法的。特许、公共资源、自然资源、公共利益等。
                          \item 可以不设定的。自主决定（\scx{谈恋爱}）、市场调节、行业自律、事后监督。没考过。
                        \end{enumerate}
                  \item 经常性许可。稳定、日常存在的。法律、行政法规、地方性法规都可以。
                  \item 临时性许可。
                        \begin{enumerate}
                          \item 全国 \yjt \zd{国务院决定}（\blh 的其他规范性文件）。不是国务院部门。
                          \item 地方 \yjt \zd{省规章，本级人/常，有效期最多 1 年}。
                        \end{enumerate}
                  \item \zd{地方的限制性条件}。地方保护主义、国家统一的资质资格许可、企业设立登记及其前置性许可。
                  \item 评价/立法后评估。设定机关应当定期评价，实施机关可以适时评价。
                  \item 停止。\zd{经济事务的许可}经过国务院的批准省政府可以停止。
                \end{enumerate}
          \item 具体规定。从粗到细。不得抵触上位法，可以细化上位法。\zd{不得增设许可，不得增设条件}。
        \end{enumerate}
  \item 实施。谁发许可证，怎么发。
        \begin{enumerate}
          \item 主体。具有行政许可权的行政机关。
                \begin{enumerate}
                  \item 授权。\zd{法律法规}有特别授权资格。
                  \item 委托。\zd{得有法律依据；只能委托别的行政机关，不能转委托}。
                  \item 集中。省政府可以决定 \yjt 国务院批准。
                        \begin{enumerate}
                          \item 一个机关发一个证。一个窗口对外。
                          \item 多个机关的多个许可。为了高效便民。权力所有权是否转移。
                                \begin{itemize}
                                  \item \zd{集中许可}。分散的权力都交给一个行政机关去行使。\zd{以新机关为被告}。
                                  \item 一站式审批联合许可。把所有的机关放在一个大楼里面。
                                  \item 并联式审批/统一许可。主办窗口就是一个\zd{收材料}的。
                                \end{itemize}
                        \end{enumerate}
                \end{enumerate}
          \item 一般程序。
                \begin{enumerate}
                  \item 申请。
                        \begin{enumerate}
                          \item 格式文本（申请表）不是必须的。一般免费，例外：\al{报名费}（\zd{法律行政法规定}）。
                          \item \zd{可以委托}。\zd{高度人身属性必须本人}（\al{护照}）。
                        \end{enumerate}
                  \item 受理。
                        \begin{enumerate}
                          \item 出具具有盖章 + 日期的书面凭据。
                          \item 材料\zd{有错就当场改}。
                          \item 材料不合要求。\zd{材料补正告知书（过程性行为不可诉）}，当场 \hz\, 5 日内一次告知。
                        \end{enumerate}
                  \item 审查。\zd{实质审 2 个人}。
                  \item （书面）决定。
                        \begin{enumerate}
                          \item \zd{准予许可要公开，不予许可不公开}。
                          \item 法律、行政法规、国务院所设定的许可\zd{没有地域限制则全国有效}。例外\zd{自我设限}（\al{法考 C 证}）。
                        \end{enumerate}
                  \item 颁发许可证。加盖标签印章。
                \end{enumerate}
          \item 特别程序。
                \begin{enumerate}
                  \item 听证。\zd{免费}。
                        \begin{enumerate}
                          \item 依申请 \hz 依职权（涉及公共利益 + \zd{公告}）。
                          \item 5207。5 天内申请，20 天把会开起来，7 天前告知。
                          \item 回避。近亲属、调查过的人不能做主持人。
                          \item 笔录。\zd{排他效力}，是后续决定的\zd{唯一依据}。
                        \end{enumerate}
                \end{enumerate}
          \item 监督检查。
                \begin{enumerate}
                  \item 实地检查。
                        \begin{enumerate}
                          \item 后果。发现存在违法情况，改正。
                          \item 一般是限期改正，\zd{核准类的立即改正}（\al{电梯安全隐患}）。
                          \item 改还不解气，还可以吊销执照。\zd{吊销是事后违法，撤销本来就违法}。
                        \end{enumerate}
                \end{enumerate}
          \item \zd{许可的撤回/废止}。因\zd{公共利益}发生变化要\zd{废止}许可。
                \begin{enumerate}
                  \item 废止之日起失去效力；要进行补偿。
                        \begin{enumerate}
                          \item 要\zd{先行处理}。行政机关 \yjt 法院。
                          \item 补偿标准。有标准 \yjt 实际损失（\zd{成本 + 合理利润}）；特许 \yjt 只补实际投入。
                        \end{enumerate}
                \end{enumerate}
          \item 注销。无法继续从事许可活动。\zd{不能用了都要注销}。
        \end{enumerate}
\end{enumerate}
