\section{行政许可}
\begin{enumerate}
  \item 学习脉络。概念、种类、立法（设定、具体规定）、执行（主体。自己、授权、委托、集中；程序）。
  \item \zd{一般禁止的解除}。区分行政审批、行政确认。
  \item 特征。依申请、授益、要式、\zd{外部性}（内外看身份，一定是冲着外部的）。
  \item 分类。
        \begin{enumerate}
          \item 法律分类。
                \begin{enumerate}
                  \item 一般许可。\zd{形式审查 1 人，实质审查 2 人}。\al{驾照}。
                  \item 特许。赚大钱，优中选优。\zd{招投标}方式。
                  \item 认可。对相\zd{对人}的资质能力条件进行评价，允许从事特定活动。\zd{要考试}。\al{医师执业}。
                  \item 核准。\zd{对物}、2 人\bs{核对}。\al{电梯检测合格证}。
                  \item （初始）登记。社会团体、企业法人。实质审查。\al{颁发工商营业执照}。
                \end{enumerate}
          \item 附期限、不附期限。附期限延期提前 30 天 \yjt \zd{第一次/后面 \yjt 没有答复 \yjt 视为默示拒绝/同意}。
          \item 有数量限制、没有数量限制。\zd{原则根据受理顺序，特许是例外}。
          \item 可转让、不可转让。\zd{一般不能转让}，例外。\zd{采矿证}之类的准物权。
        \end{enumerate}
  \item 立法权限。
        \begin{enumerate}
          \item 设定。从无到有。
                \begin{enumerate}
                  \item 事项范围。啥许可事项需要立法。
                        \begin{enumerate}
                          \item 可以立法的。特许、公共资源、自然资源、公共利益等。
                          \item 可以不立法的。市场调节、行业自律、自主决定（\al{谈恋爱}）、事后监督。没考过。
                        \end{enumerate}
                  \item 经常性许可。稳定、日常存在的。\zd{仅法律、法规}。
                  \item 临时性许可。
                        \begin{enumerate}
                          \item 全国 \yjt \zd{国务院决定}（性质 = 国务院的其他规范性文件）。\al{国务院部门决定 ×}。
                          \item 地方 \yjt \zd{省规章、本级人大/人常、有效期最多 1 年}。
                        \end{enumerate}
                  \item 地方的限制性条件。\zd{地}方\zd{保}护主义、\zd{国}家\zd{统}一的资质许可、\zd{企}业设立\zd{登}记及其前置性许可。
                  \item 评价/立法后评估。\zd{设定机关应当定期评价，实施机关可以适时评价}。
                  \item 停止。\zd{经济事务的许可}，省政府决定 \yjt 国务院批准。
                \end{enumerate}
          \item 具体规定。从粗到细，下位法\zd{可以细化}上位法，但是\zd{不能抵触}上位法。\zd{不得增设许可，不得增设条件}。
        \end{enumerate}
  \item 实施。谁发许可证，怎么发。
        \begin{enumerate}
          \item 主体。具有行政许可权的行政机关。
                \begin{enumerate}
                  \item 授权。\zd{法律、法规}。
                  \item 委托。\zd{只能委托其他行政机关、不能转委托、要公告}。
                  \item 集中。省政府决定 \yjt 国务院批准。
                        \begin{enumerate}
                          \item 一个机关发一个证。一个窗口对外。
                          \item 多个机关的多个许可。为了高效便民。权力所有权是否转移。
                                \begin{itemize}
                                  \item \zd{集中许可}。独揽大权。\zd{以新机关为被告}。
                                  \item 一站式审批/\zd{联合}许可。把所有的机关放在一个\zd{大楼}里面。\al{一站式服务大厅}。
                                  \item 并联式审批/\zd{统一}许可。主办窗口就是一个\zd{收材料}的。
                                \end{itemize}
                        \end{enumerate}
                \end{enumerate}
          \item 一般程序。\al{驾照、法考}。
                \begin{enumerate}
                  \item 申请。
                        \begin{enumerate}
                          \item \zd{格式文本（申请表）非必须}。免费。\al{格式文本费用由 2 元将为 1 元 \cuo}。例外。\zd{报名费}（\zd{法律、行政法规定}）。
                          \item \zd{可以委托}。\zd{高度人身属性必须本人}（\al{护照}）。
                        \end{enumerate}
                  \item 受理。
                        \begin{enumerate}
                          \item 出具盖章 + 日期的书面凭据。
                          \item 材料\zd{有错当场改}。
                          \item 材料不合要求。\zd{材料补正告知书}，当场 \hz 5 日内一次告知。过程性行为不可诉。
                        \end{enumerate}
                  \item 审查。\zd{实质审查}。
                  \item （书面）决定。
                        \begin{enumerate}
                          \item \zd{准予许可要公开，不予许可（包括事后撤销的）不公开}。
                          \item 法律、行政法规、国务院所设定的许可\zd{没有地域限制则全国有效}。\zd{自我设限}例外（\al{法考 C 证}）。
                        \end{enumerate}
                  \item 颁发。加盖标签、印章。
                \end{enumerate}
          \item 听证程序。\zd{免费}。
                \begin{enumerate}
                  \item 依申请、依职权（公共利益 + \zd{公告}）。
                  \item 5207。5 天内申请，20 天把会开起来，7 天前告知。
                  \item 回避。\zd{近亲属、调查过的人不能做主持人}。
                  \item 笔录。\zd{排他效力}，是后续决定的\zd{唯一依据}。\al{应当参考 \cuo 应当根据 \dui}。
                \end{enumerate}
          \item 监督检查。
                \begin{enumerate}
                  \item 实地检查。发现违法情况，\zd{责令改正（性质是强制措施）}。
                        \begin{enumerate}
                          \item \zd{一般限期改，核准立即改}。\al{电梯安全隐患}。
                          \item \zd{特别严重吊销执照}。\zd{vs 撤销本来就违法。吊销是事后违法}。
                        \end{enumerate}
                \end{enumerate}
          \item 许可的撤销。许可决定机关、上级行政机关、法院、被越权机关。
          \item \zd{许可的撤回/废止}。因\zd{公共利益}发生变化要\zd{废止}许可。
                \begin{enumerate}
                  \item 要\zd{先行处理}的。行政机关 \yjt 法院。
                  \item 补偿标准。有标准 \yjt 实际损失（\zd{成本 + 合理利润}）。特许 \yjt 只补实际投入。
                  \item vs 注销。无法继续从事许可活动。\zd{不能用了都要注销}。注销就是注销。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
