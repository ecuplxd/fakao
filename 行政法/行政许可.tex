\section{行政许可}
\begin{enumerate}
  \item 学习脉络：概念、种类、立法（设定、具体规定）、执行（主体：自己、授权、委托、集中；程序）。
  \item \zd{一般禁止的解除}。区分行政审批、行政确认。
  \item 特征：依申请、授益、要式、\zd{外部性（内外看身份}，一定是冲着外部的）。
  \item 分类。
        \begin{enumerate}
          \item 法律分类。
                \begin{enumerate}
                  \item 一般许可。\zd{形式审查 1 人，实质审查 2 人}。
                  \item 特许。赚大钱，优中选优。\zd{招投标}。
                  \item 认可。对相对\zd{人}的资质能力条件进行评价，允许从事特定活动。\zd{要考试}。
                  \item 核准。对\zd{物}，电梯检测合格证。\zd{2 人核对}。
                  \item 登记。社会团体、企业法人。实质审查 2 个人。
                \end{enumerate}
          \item 不附期限和附期限。附期限延期：\zd{第一次/后面没有答复视为默示拒绝/同意}。
          \item 有数量限制和没有数量限制。\zd{原则根据受理顺序，例外特许}。
          \item 可转让不可转让。\zd{一般不能转让}，例外采矿证之类的准物权。
        \end{enumerate}
  \item 立法权限。
        \begin{enumerate}
          \item 设定：从无到有。
                \begin{enumerate}
                  \item 事项范围。啥事需要立法。
                        \begin{enumerate}
                          \item 可以立法的：特许、公共资源、自然资源、公共利益等。
                          \item 可以不设定的。没考过。自主决定（\scx{谈恋爱}）、市场调节、行业自律、事后监督。
                        \end{enumerate}
                  \item 经常性的许可。稳定、日常存在的。法律、行政法规、地方性法规都可以。
                  \item 临时性的许可。全国\zd{国务院决定}（-的其他规范性文件）；\zd{地方：省规章，本级人/常，有效期 1 年}。
                  \item \zd{地方的限制性条件}：地方保护主义、国家统一的资质资格许可、企业设立登记及其前置性许可。
                  \item 设定（立法）机关应当定期评价，实施（执行）机关可以适时评价。
                  \item \zd{经济事务的许可}省政府可以停止，经过国务院的批准。
                \end{enumerate}
          \item 具体规定：从粗到细。不得抵触上位法，可以细化上位法。\zd{不得增设许可，不得增设条件}。
        \end{enumerate}
  \item 实施。谁发许可证，怎么发。
        \begin{enumerate}
          \item 主体。
                \begin{enumerate}
                  \item 具有行政许可权的行政机关。
                  \item 授权。\zd{法律法规}有特别授权资格。
                  \item 委托。\zd{只能是别的行政机关；不能转委托；得有法律依据}。
                  \item 集中。省政府可以决定 \yjt 国务院批准。
                        \begin{enumerate}
                          \item 一个机关发一个证，一个窗口对外。
                          \item 多个机关的多个许可。为了高效便民：
                                \begin{itemize}
                                  \item \zd{集中许可}。分散的权力都交给一个行政机关去行使。\zd{被告：新机关}。
                                  \item 一站式审批联合许可。把所有的机关放在一个大楼里面。权力所有权不转移。
                                  \item 并联式审批/统一许可。主办窗口就是一个\zd{收材料}的。
                                \end{itemize}
                        \end{enumerate}
                \end{enumerate}
          \item 一般程序。
                \begin{enumerate}
                  \item 申请。
                        \begin{enumerate}
                          \item 格式文本（申请表）不是必须的。一般免费，例外：如报名费（\zd{法律行政法规定}）。
                          \item \zd{可以委托}，\zd{高度人身属性必须本人}（如护照）。
                        \end{enumerate}
                  \item 受理。
                        \begin{enumerate}
                          \item 都得出具盖章日期书面凭据。
                          \item 材料\zd{有错就当场改}；
                          \item 材料不合要求，\zd{材料补正告知书}，当场 \hz\, 5 日内一次告知。\zd{过程性行为不可诉}。
                        \end{enumerate}
                  \item 审查。\zd{实质审两个人}。
                  \item 决定（书面）。
                        \begin{enumerate}
                          \item \zd{准予许可要公开，不予许可不公开}。
                          \item 法律行政法规、国务院所设定的许可\zd{没有地域限制则全国一律有效}。例外\zd{法律自我设限}（法考证）。
                        \end{enumerate}
                  \item 颁发许可证。加盖标签印章。
                \end{enumerate}
          \item 特别程序。
                \begin{enumerate}
                  \item 听证。
                        \begin{enumerate}
                          \item 依申请、依职权（涉及公共利益）+ \zd{要公告}；\zd{免费}。
                          \item 5 天内申请，20 天把会开起来，7 天前告知。5207。
                          \item 回避。近亲属、调查过的人。不能做主持人。
                          \item 笔录。\zd{是后续决定的唯一依据}。不能再从这个笔录之外寻求别的证据，\zd{排他效力}。
                        \end{enumerate}
                \end{enumerate}
          \item 监督检查。
                \begin{enumerate}
                  \item 实地检查。
                        \begin{enumerate}
                          \item 后果：发现存在违法情况，改正。
                          \item 一般是限期改正，\zd{核准的立即改正}（电梯安全隐患）。
                          \item 改不解气，还可以吊销执照。
                          \item \zd{吊销事后违法，撤销本来就违法}。
                        \end{enumerate}
                \end{enumerate}
          \item \zd{许可的撤回}：因\zd{公共利益}发生变化要\zd{废止}许可。
                \begin{enumerate}
                  \item 废止之日起失去效力；要进行补偿：
                        \begin{itemize}
                          \item 要\zd{先行处理}，行政机关 \yjt 法院。
                          \item 补偿标准。实际损失、实际投入。\zd{只补成本}。
                        \end{itemize}
                \end{enumerate}
          \item 注销。把许可证给撕掉。无法继续从事许可活动。\zd{不能用了都要注销}。
        \end{enumerate}
\end{enumerate}
