\section{裁判与执行}
\begin{enumerate}
  \item 一个行为一个案件一种判决。直接讲怎么判的都是错的。\al{可以适用撤销判决 \cuo}。
  \item 合法。\zx{只有“驳回原告诉讼请求”一种}。不要受题干、选项干扰！只看合不合法！\al{确认、维持合法判决 \cuo}。
  \item 违法作为。\zx{先看能不能撤 \yjt 确认违法/无效 \yjt 赔偿 + 补救。能 \yjt 撤销 \hz 变更}。
        \begin{enumerate}
          \item 不能撤销的情形 5 种。\zd{回不去}。\bs{公共利益、行为、内容、程序}。
                \begin{enumerate}
                  \item 损害公共利益。\al{盖了政府办公大楼}。
                  \item 诉讼中行政行为已被撤销 \hz 已做改变。\al{退罚款}。
                  \item \zx{不具有可撤销性的内容}。\al{限制人身自由已执行完毕、保护善意第三人、证明文件}。
                  \item 程序轻微违法，不影响实体权利。\al{晚送罚单 \dui。没有听证 \cuo}。\sjt \zd{确认违法}。
                  \item 无效 \yjt \zd{确认无效}。
                \end{enumerate}
          \item 撤销判决。\yi 法律错。\er 全部违法全部撤，部分违法部分撤。\al{征 6 亩，2 亩非法}。\san 需\zd{补充判决}（重做）。
                \begin{enumerate}
                  \item 和确认无效的转换（难点）。\al{甲起诉请求确认处罚无效，法院审查认为只是程序违法，可以撤销但不属于无效情形。甲拒绝变更诉讼请求}。
                        \begin{enumerate}
                          \item 由低到高（升舱）。请求撤销，实则无效 \yjt 直接确认无效。
                          \item 由高到低（通知）。请求无效，实则撤销 \yjt 不同意变更 \yjt \zd{驳回诉讼请求}。
                          \item \bs{无效转撤销要遵守诉讼时效限制}。\al{请求无效，实则撤销 + 过了起诉期 \yjt 裁定驳回起诉}。
                        \end{enumerate}
                  \item 和复议改变。判决\zd{撤销}的时候，应当\zd{责令复议机关重做 \hz 恢复}原行为效力。
                  \item 和复议维持。\zx{ 应当对复议决定和原行为一并作出裁判 \yjt 诉两个、审两个、判两个}。
                        \al{如法院认定取缔决定违法予以撤销，市工商局的复议决定\xhx{自然无效} \cuo}。
                        \begin{enumerate}
                          \item \al{原罚五百（违法），复作为维持 \yjt 撤销 + \xhx{原机关重做}（干活的始终是下级）}。
                          \item \al{原是不许可（违法），复不作为维持 \yjt 撤销}。
                          \item \al{原合法，复违法（\xhx{程序}）\yjt 合法驳回。不合法。\yi 轻微 \yjt 确认违法。\er 不轻微 \yjt 撤销}。
                          \item \al{原不符合复议/诉讼的受理条件，复维持 \yjt 裁定驳回起诉}。
                        \end{enumerate}
                \end{enumerate}
          \item 变更判决 \ydy 刑诉上诉不加刑。例外。\zd{被上被抗}。
          \item \zx{可撤可变。行政处罚显失公正、其他行为款额有误}。\al{法院认为处罚过重，将罚款 10W 改为 3W \dui}。
          \item \zx{不存在不可撤销的情形，那就大胆的选}。\al{如有证据证明交通事故系甲醉酒所致，法院应当判决撤销某县社保局的工伤认定 \dui}。
        \end{enumerate}
  \item 违法不作为。\zx{知道做啥，履行给付。不知做啥，重新处理。作无意义，确认违法}。
        \begin{enumerate}
          \item 知道。\yi 履行判决。行为。\al{出生日期错误}。\er 给付判决。给钱。\al{苹果被抢完}。
          \item 重新处理判决。尚需被告调查裁量。\al{判决被告 xx 天做 \cuo}。
          \item \al{甲受到乙的威胁，向公安报案，公安未采取任何措施。3 天后，甲了解到乙已被刑事拘留。甲提起诉讼}。
        \end{enumerate}
  \item 执行时效 \zd{2} 年。\zx{拒不执行}。\al{责令处理 \cuo}。
        \begin{enumerate}
          \item 给付货币的。通知银行从该行政机关的账户内划拨。
          \item 期满之日起，对该行政机关负责人罚款。
          \item 将有关情况公告。
          \item 向监委、上一级主管部门提出司法建议 \yjt 处理 + 告知。
          \item 恶劣的。拘留主观人员和其他直接责任人。
          \item 依法追究刑事责任。
        \end{enumerate}
\end{enumerate}
