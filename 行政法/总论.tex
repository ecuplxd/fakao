\section{总论}
\begin{enumerate}
  \item 特点。解决争议，控权保民。
  \item 自由选择。
        \begin{enumerate}
          \item 诉讼。爹是外人。审合法性。象征性收费。司法终局。\xydy 6 个月。\al{特别法有规定的，特别法优先 \dui}。
          \item 复议。爹是行政机关。审合法性和合理性/适当性。不收费。
                \begin{enumerate}
                  \item 原告叫申请人，被告叫被申请人，居中裁判的叫复议机关。
                  \item \dydy 60 日 \yjt 收到复议决定书之日起 15 日内 \yjt 诉讼，法律另有规定的除外。诉讼可以少，复议只多不少。
                  \item 不能二次复议。例外。省、部级原级复议 \yjt 国务院裁决（终局不可诉）。
                \end{enumerate}

                \begin{kk}{复议前置但不终局}
                  最重要的最爱考的。高精尖。背，每年一道。
                  \begin{enumerate}
                    \item 自然资源类确权。\al{土地、林木等所有权、使用权}。我已取得 \yjt 给了别人。
                    \item 涉税争议（是否纳税、方式、数额）。排除税收[处罚、强制执行、强制措施]、反倾销税。
                          \begin{enumerate}
                            \item \al{是否征收反倾销税的决定}。
                            \item \al{税务机关做出的\xhx{处罚}决定}。
                            \item \al{区税务局决定对甲公司\xhx{罚款} 30W}。
                            \item \al{市税务局责令甲公司\xhx{补缴 2W 税款}并\xhx{罚款 1W}。对于\xhx{补缴}税款的决定，甲公司可以不经复议，直接起诉 \cuo}。
                            \item \al{\xhx{扣押}相当于应缴税款的商品 \yjt 税收保全呐}。
                          \end{enumerate}
                    \item 反垄断、限制集中。\al{支付宝并购微信支付}。没考过。
                    \item 起诉条件。经过复议（实体审查）。
                          \al{市政府认为甲的复议请求\xhx{超过了复议申请期}（其实没超），于是驳回了甲的复议请求。
                            法院应当以未经过复议为由，裁定驳回甲的起诉 \dui}。
                  \end{enumerate}
                \end{kk}

                \begin{kk}{复议前置且终局（不能诉讼）}
                  \begin{enumerate}
                    \item 省政府的自然资源裁决。
                    \item 外国人、出入境、限自由。限制法律里是侵犯的意思。题目出现出入境机关字眼。
                          \al{\xhx{出入境}边防检查机关对\xhx{外国人}采取的\xhx{遣送出境}措施}。
                  \end{enumerate}
                \end{kk}
        \end{enumerate}
\end{enumerate}
