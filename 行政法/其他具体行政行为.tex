\section{其他具体行政行为}
\begin{enumerate}
  \item 只要知道它是啥就得了，没有单行立法。\zd{指导、裁决、确认、给付、不作为、征收、奖励}。
  \item 行政\zd{裁决}。\zd{\ydy 判决}。\zd{2 方争议，3 方主体}。vs 确认（单方认定）。\al{惩罚性赔偿}。
  \item 行政\zd{确认}。对民事主体的法律\zd{地位}、法律\zd{关系}、有关\zd{事实}进行甄别，予以\zd{确定、认定、证明、宣告}。
        \begin{enumerate}
          \item \zd{\ydy 公证、鉴定}。既存事实 \hz 法律关系的一种加强（上 buff）。vs 许可（\zd{给予恩赐}）。
          \item 常考的。\al{交通事故责任、火灾事故、劳动伤残等级、工伤、排污不合格认定}。
          \item 易混的。\al{结婚登记；\xhx{初始登记}是许可、变更登记是确认；\xhx{核准}是许可、\xhx{房屋合格证}是确认}。
        \end{enumerate}
  \item 行政\zd{给付}。\zd{给弱势群体好处}。\al{最低生活保障\xhx{金}，残疾补助金，社会保险金}。
  \item 行政\zd{不作为}。积极的不作为\zd{明确拒绝}（\al{许可、不予许可}），消极的不作为\zd{不理不踩}（\al{到期视为默示拒绝}）。
  \item 行政\zd{征收}/行政征用。所有权/使用权，\al{隔离酒店}。
        \begin{enumerate}
          \item \zd{带税字、费字的叫征收。不是处罚}。\al{补缴税款}。
          \item 征用一定是会给钱的，叫补偿；征收有可能是有偿的（\al{拆迁补偿}），有可能是无偿的（\al{收税}）。
        \end{enumerate}
  \item 行政\zd{奖励}。\zd{激励先进}。赋予权利\al{如给钱}。\zd{减免义务}如\al{不/少纳税}。
\end{enumerate}
