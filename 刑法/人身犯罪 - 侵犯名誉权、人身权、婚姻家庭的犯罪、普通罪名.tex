\section{人身犯罪 - 侵犯名誉权、人身权、婚姻家庭的犯罪、普通罪名}
\begin{enumerate}
  \item 侵犯名誉权、人身权。
        \begin{enumerate}
          \item 侮辱。语言暴力、行动暴力。
          \item 诽谤。
                \begin{enumerate}
                  \item 外部名誉/社会评价、虚假事实、\zd{被害人都可以承诺放弃}、语言暴力。
                  \item \zd{不能假的太离谱，要能使一般人相信}。\al{吸毒、卖淫、包养情人 \dui}。
                \end{enumerate}
          \item \sjt 都要求\zd{公然性}。\al{移动客服案（一对一）}。
          \item \zd{诬告陷害}。
                \begin{enumerate}
                  \item \zd{虚假的犯罪事实}。
                  \item \yi 向公安、司法机关告发。\er 诬告在先，司法机关反应在后。
                  \item \zd{具体确定}的“人”。\al{乙为甲做为证，向公安声称自己看到了“凶手”杀害丙}。
                  \item \al{甲诬告企业逃税的，也能追究甲诬告陷害的刑事责任 \dui}。
                  \item 诬告失败要坐牢。语言暴力。
                \end{enumerate}
          \item 刑讯逼供。
                \begin{enumerate}
                  \item 司法工作人员。\al{陪审员 \dui}。
                  \item 肉刑或变相肉刑（酷刑）。\al{审讯室单曲循环一首歌、精神折磨崩溃}。
                  \item 构成故意伤害的，想象竞合择一重。
                  \item \zd{（过失）致人死亡拟制为故意杀人}。
                \end{enumerate}
        \end{enumerate}
  \item 婚姻家庭的犯罪。
        \begin{enumerate}
          \item 重婚。同时维持两份婚姻。
                \begin{enumerate}
                  \item 法律婚 + 法律婚、法律婚 + 事实婚。
                  \item 事实婚 + 事实婚、同居不算。
                \end{enumerate}
          \item 破坏军婚。
                \begin{enumerate}
                  \item 明知军人的配偶 + [(结婚 \hz \zd{同居/包二奶})，No(不包括通奸)]
                  \item \zd{一夜情} + 一方有配偶 = \zd{通奸} + 共同生活 = \zd{同居} + 公开 + 对外夫妻名义 = \zd{事实婚} + 登记 = 法律婚。
                \end{enumerate}
          \item 通奸不是犯罪。取证困难。
          \item 暴力干涉婚姻自由。
          \item \zd{虐待}。要求是家庭成员。\sjt 结果加重犯。（过失）致人死亡（包括被害人自杀）。
          \item \zd{虐待被监护、看护人}。\zd{不是对立互斥关系，可能法条、想象竞合}。\al{幼儿园}。
        \end{enumerate}
  \item 普通罪名。
        \begin{enumerate}
          \item \zd{侵犯公民个人信息}。电子信息。\zd{要求情节严重}。
          \item 强迫劳动。
          \item 雇用童工从事危重劳动。未满 16 岁 = 童工。
          \item 非法侵入住宅。
        \end{enumerate}
\end{enumerate}
