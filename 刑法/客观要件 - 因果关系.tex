\section{客观要件 - 因果关系}
\begin{enumerate}
  \item 解决实害结果的归属问题。\bs{先因后果}！\zx{因 \yjt 介入因素 \yjt 果 \yjt 无法查明}。考的最多的。
  \item 分类。
        \begin{enumerate}
          \item 条件说。行为与结果即使有必要条件关系，也不意味着有因果关系。
          \item 相当因果说。看介入因素发生是否异常。\al{大阪南港案 \yjt 二因一果}。
          \item \bs{危险的现实化原理}。\zx{介入因素异常 \yjt 独立关系 \yjt 谁的作用更大}。
        \end{enumerate}
  \item 因。即判断是不是一个危害行为。\zd{很喜欢考追捕行为}。
        \begin{enumerate}
          \item 没有对法益制造危险。\al{小偷跑主人追 \dui。主人跑小偷追 \cuo}。
          \item 对法益制造了危险，但\zx{“危害”行为并非法律所禁止}的。\al{警察追小偷、开车上路 \dui}。
        \end{enumerate}
  \item \zx{介入因素}。\bs{自身、体质、第三}。
        \begin{enumerate}
          \item 被害人\zx{自身行为}。\zx{被害人自杀}。常考。
                \begin{enumerate}
                  \item \al{甲为讨债拘禁乙，乙深感无力还债自杀}。
                  \item \zx{没有因果关系，但有条件关系}，算\zx{造成其他严重后果加重处罚}。\al{强奸后，羞愤自杀}。
                  \item 2 个例外。\bs{暴力干涉婚姻自由致人死亡、虐待致人死亡包括被害人自杀}。
                \end{enumerate}
          \item 被害人\zx{特殊体质}。结论。先行行为对疾病的发作\bs{有引发关系}则有因果关系。\al{街上推人}。
          \item \zx{第三人}行为。
                \begin{enumerate}
                  \item \zx{阻断救助行为}。结论。如果救助行为\zx{有救活可能性}，结果就算到阻断行为上。
                        \begin{enumerate}
                          \item \al{甲推乙入水，丙扔救生圈，被\xhx{丁捞走}}。
                          \item \al{乙驾车撞伤丙，\xhx{甲谎称自己留下}打电话叫救护车，让乙赶紧将车开走}。
                          \item \al{宗教迷信，\xhx{拒绝}输血}。
                        \end{enumerate}
                \end{enumerate}
        \end{enumerate}
  \item 果。实害结果 = 事实判断 + 价值评价（坏的）。\bs{假设、管辖、规范、重叠、择一}。
        \begin{enumerate}
          \item \zx{不讨论假设}的结果。要现实发生的。\al{沙漠钻孔案}。
          \item 管辖范围内的结果。\al{德国交警尾灯案}。
          \item 是刑法\zx{规范保护范围内}的结果。
                \begin{enumerate}
                  \item \zx{罪名规范}范围。实害结果不应超出该罪名所\zd{能保护的法益范围}。\al{甲杀乙后逃走，丙路过取财}。
                  \item （过失犯罪的）\zx{注意义务规范}范围。不应超出该罪名所\zd{要注意的义务范围}。
                        \al{车进隧道超速撞到人 \yjt 注意义务是不要和前车追尾，被害人自陷风险}。
                \end{enumerate}
          \item \zx{重叠}的因果关系。A + B \yjt C。
          \item \zx{择一}的因果关系。A \hz B \yjt C。
        \end{enumerate}
  \item 无法查明。\bs{共犯部分实行，全部负责。时间无法查明，死的越早越好}（存疑时有利于被告）。
\end{enumerate}
