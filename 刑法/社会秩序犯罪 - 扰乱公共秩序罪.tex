\section{社会秩序犯罪 - 扰乱公共秩序罪}
\begin{enumerate}
  \item 罪名特别多，知识点杂乱，某些罪名的某些要点。
  \item 妨害公务。
        \begin{enumerate}
          \item 主观故意、\zx{依法 + 正在}执行的公务。\al{袭击\xhx{上班的}警察}。
          \item 认识错误。
                \begin{enumerate}
                  \item \al{甲看到朋友乙和丙在打架，上前帮忙将丙打成轻伤，丙其实是便衣警察。故意伤害}。
                  \item \al{甲误以为丙在追杀乙，重伤丙，丙其实是便衣警察。甲假想防卫，过失致人重伤}。
                \end{enumerate}
          \item 罪数。
                \begin{enumerate}
                  \item 前罪 + 妨害公务 = 数罪并罚。例外。结合犯。
                  \item \bs{走私、贩卖、运输、制造毒品/组织、运送他人偷越国（边）境 + 妨害公务 = 前罪加重处罚}。
                \end{enumerate}
        \end{enumerate}
  \item 拒不执行判决、裁定。
        \begin{enumerate}
          \item 同时构成妨害公务的，定妨害公务。
          \item 杀害、重伤执行人员的，定故意杀人、故意伤害。
        \end{enumerate}
  \item 袭警。\yi 袭 = \zx{暴力袭击 \bdy 暴力威胁}。\al{拿菜刀\xhx{威胁}民警}。\er 警 = \zx{警察本人，警犬不算}。
  \item \zd{公文、证件、印章}犯罪。
        \begin{enumerate}
          \item \zd{伪造}、变造、\zd{买卖}国家机关公文、证件、印章。
                \begin{enumerate}
                  \item 有形伪造 \yjt \zx{无制作权限的人制作}。
                  \item \zx{无形伪造 \yjt 有权限的人}。\al{收买警察办户口}。
                \end{enumerate}
          \item 伪造公司、企业、事业单位、人民团体印章。\zd{只有伪造}，弱保护。\al{伪造大学毕业证}，毕业证要盖大学印章，伪造了印章图案，
          \item \zx{伪造、变造、买卖身份证件}。特殊的国家证件，独立出来。\al{身份证、护照、驾照、社保卡}。
          \item 盗窃、抢夺、毁灭国家机关公文、证件、印章。
          \item \bs{冒名顶替}。新增。\bs{上、公、安}。上大学资格、公务员录用、就业安置待遇。\al{罗彩霞}。
        \end{enumerate}
  \item “小混混”型犯罪。
        \begin{enumerate}
          \item 招摇撞骗。
                \begin{enumerate}
                  \item \zx{冒充的必须是国家机关工作人员}。一般人相信。
                  \item 骗啥都可以。感情、名誉、钱财（\zd{和诈骗想象竞合择一重}）、色。
                  \item \zx{冒充警察的，定本罪，从重处罚}。
                \end{enumerate}
          \item \zx{聚众斗殴}。\bs{斗殴无防卫，斗殴有承诺}。
                \begin{enumerate}
                  \item \zx{致人重伤 \yjt 故意伤害罪}。\al{轻伤 \cuo}。
                  \item \zx{（过失）致人死亡 \yjt 拟制为故意杀人}。
                  \item \zx{只处罚首要分子和积极参加者}。
                        \begin{enumerate}
                          \item 首要分子 = \zx{发号施令的、不分阵营、能有多个}。
                          \item 积极参加者 = 造成实害结果的人。
                        \end{enumerate}
                \end{enumerate}
          \item 寻衅滋事。无故找碴。\zx{和若干罪不是排斥对立关系，可以想象竞合择一重}。
          \item 聚众扰乱社会秩序。
          \item \bs{高空抛物}。具体危险。新增。\al{从高空向四下无人的邻居车上扔酒瓶子 \cuo}。
                \begin{enumerate}
                  \item 高度要求。\yi \zx{只要 \dydy 2 层楼，3m 左右}。\er 存在\zx{垂直位移}。\al{朝井里扔、空对地 \dui。平地起抛 \cuo}。
                  \item \zx{故意}。vs 高空坠物（过失）。
                  \item \zx{引起其他罪，想象竞合}。\al{高空抛燃烧的蜂窝煤球，引发火灾}。
                \end{enumerate}
          \item 投放虚假危险物质。
        \end{enumerate}
  \item “大哥型”犯罪。
        \begin{enumerate}
          \item 组织、领导、参加黑社会性质组织。
                \begin{enumerate}
                  \item 有组织、有一定的经济实力、进行违法犯罪活动、\zx{形成非法控制}（这地方我说了算）。
                  \item 参加。进去后\zd{得干点违法犯罪活动}。
                  \item 责任划分。\zx{“主营业务” \yjt 组织罪行。“主营业务外” \yjt 个人罪行}。
                \end{enumerate}
          \item 赌博。\zx{长期以赌博为业}。
          \item 开设赌场。赌场扩大解释。赌博 = 有输有赢。\al{微信抢红包 \yjt 营利为目的 \dui}。
          \item 组织参与国（境）外赌博。\al{导游丁携旅游团赴境外旅游，期间组织成员进入当地\xhx{合法}赌场参与赌博 \dui}。
        \end{enumerate}
  \item 考试作弊类犯罪。
        \begin{enumerate}
          \item \zd{组织考试作弊}。\zx{考试缩小解释}，有档次的考试。\al{地方公务员考试 \cuo}。
          \item 非法出售、提供试题、答题。
          \item 代替考试。
        \end{enumerate}
  \item 计算机犯罪。
        \begin{enumerate}
          \item 计算机信息系统。\zx{自动处理数据功能的系统}。\al{计算机；网络、通信、自动化控制；定位监控系统}。
          \item 非法\zd{侵入}计算机信息系统。\zd{仅针对国家事务、国防建设}、登录访问。强保护。
          \item 非法\zd{获取}计算机信息系统数据。\zd{要求先侵入}。
          \item \zd{破坏}计算机系统。保护范围\zd{扩大到民营公司}。弱保护。
                \begin{enumerate}
                  \item 对\zd{功能}、数据、程序进行删除、修改、增加、干扰。\al{劫持域名、删评}。
                  \item 制作、传播病毒。
                \end{enumerate}
          \item 非法\zd{控制}计算机系统。\al{肉鸡}。
          \item 非法利用信息网络。\zx{可以和 N 多罪想象竞合}。\al{在网上发贩毒信息、组建微信群、设立网站}。
          \item 帮助信息网络犯罪活动。\zd{帮信罪}。
                \begin{enumerate}
                  \item 提供互联网接入、网络存储、通讯传输等技术支持。
                  \item \zx{可以和 N 多罪想象竞合}。\al{甲明知乙实施网络诈骗，仍为其提供网络接入服务}。
                  \item 虽然是独立罪名，\bs{成立还是要严格遵循共犯从属性}。\al{实际乙并未去诈骗}。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
