\section{社会秩序犯罪 - 扰乱公共秩序罪}
\begin{enumerate}
  \item 罪名特别多，知识点杂乱，某些罪名的某些要点。
  \item \kd{妨害公务}。
        \begin{enumerate}
          \item 主观故意、\zd{依法 + 正在}执行的公务。\al{袭击上班的警察}。
          \item 认识错误。
                \begin{enumerate}
                  \item \al{甲看到朋友乙和丙在打架，上前帮忙将丙打成轻伤，丙其实是便衣警察。故意伤害}。
                  \item \al{甲误以为丙在追杀乙，重伤乙，丙其实是便衣警察。甲假想防卫，过失致人重伤}。
                \end{enumerate}
          \item 罪数。
                \begin{enumerate}
                  \item 原则结合犯。前罪 + 妨害公务 = 数罪并罚。
                  \item 例外。\al{走私、贩卖、运输、制造毒品/组织、运送偷越国（边）境 + 妨害公务 = 前罪加重处罚}。
                \end{enumerate}
        \end{enumerate}
  \item 拒不执行判决、裁定。同时构成妨害公务的，定妨害公务。杀害、重伤执行人员的，定故意杀人、故意伤害。
  \item \kd{袭警}。
        \begin{enumerate}
          \item 行为方式\zd{只包括暴力袭击，不包括暴力相威胁}。\al{拿把菜刀威胁民警，吓跑了}。
          \item \zd{警察本人，警犬不算}。
        \end{enumerate}
  \item 公文、证件、印章犯罪。
        \begin{enumerate}
          \item \zd{伪造}、变造、\zd{买卖}国家机关公文、证件、印章。
                \begin{enumerate}
                  \item 伪造。有形伪造 \yjt \zd{无制作权限的人制作}。无形伪造 \yjt 有权限的人。\al{收买警察办户口}。
                \end{enumerate}
          \item 伪造公司、企业、事业单位、人民团体印章。只有伪造，弱保护。\al{伪造大学毕业证}，毕业证要盖大学印章，伪造了印章图案，
          \item \kd{伪造、变造、买卖身份证件}。特殊的国家证件，独立出来。\al{身份证、护照、驾照、社保卡}。
          \item 盗窃、抢夺、毁灭国家机关公文、证件、印章。
          \item \kd{冒名顶替}。新罪。\zd{上大学的资格、公务员录用、就业安置的待遇}。\al{罗彩霞}。
        \end{enumerate}
  \item “小混混”型犯罪。
        \begin{enumerate}
          \item \kd{招摇撞骗}。
                \begin{enumerate}
                  \item \zd{冒充的必须是国家机关工作人员}。一般人相信。
                  \item 骗啥都可以，感情、名誉、钱财（\zd{和诈骗想象竞合择一重}）、色。
                  \item 冒充警察的，定本罪，从重处罚。
                \end{enumerate}
          \item \kd{聚众斗殴}。\zd{斗殴无防卫，斗殴有承诺}。
                \begin{enumerate}
                  \item \zd{致人重伤}（\al{轻伤 ×}）的，以故意伤害罪论处。
                  \item \zd{致人重伤死亡的拟制为故意杀人}。首要分子不分阵营。
                  \item \zd{只处罚首要分子和积极参加者（造成实害结果的那个人）}。
                  \item 首要分子认定。\zd{发号施令的、能有多个}。
                \end{enumerate}
          \item 寻衅滋事。\zd{无故找碴}。
                \begin{enumerate}
                  \item \zd{和若干罪不是排斥对立关系，可以想象竞合择一重}。不是结果加重犯。
                \end{enumerate}
          \item 聚众扰乱社会秩序。
          \item \kd{高空抛物}。具体危险。新罪。\al{从高空向四下无人的邻居车上扔酒瓶子 ×}。
                \begin{enumerate}
                  \item 高度要求。\zd{只要 >= 2 层楼，3m 左右}。存在垂直位移即可。\al{朝井里扔、空对地 √。平地起抛 ×}。
                  \item 故意。vs 高空坠物（过失）。
                  \item 引起其他罪，想象竞合。\al{高空抛燃烧的蜂窝煤球，引发火灾}。
                \end{enumerate}
          \item 投放虚假危险物质。
        \end{enumerate}
  \item “大哥型”犯罪。
        \begin{enumerate}
          \item 组织、领导、参加黑社会性质组织。
                \begin{enumerate}
                  \item 有组织、有一定的经济实力、进行违法犯罪活动、形成非法控制（这地方我说了算）。
                  \item 参加。进去后\zd{得干点违法犯罪活动}。
                  \item 责任划分。“主营业务” \yjt 组织罪行。“主营业务外” \yjt 个人罪行。
                \end{enumerate}
          \item 赌博。\zd{长期以赌博为业}。
          \item \kd{开设赌场}。\zd{赌博要求有输有赢}。\al{微信抢红包 \yjt 营利为目的 √}。
          \item 组织参与国（境）外赌博。\al{丁组织境内旅游团赴境外旅游，期间组织成员进入当地合法赌场参与赌博 √}。
        \end{enumerate}
  \item 考试作弊类犯罪。
        \begin{enumerate}
          \item \kd{组织考试作弊}。缩小解释，有档次的考试。\al{地方公务员考试 ×}。
          \item 非法出售、提供试题、答题。
          \item 代替考试。
        \end{enumerate}
  \item \kd{计算机犯罪}。
        \begin{enumerate}
          \item 计算机信息系统。自动处理数据功能的系统。\al{计算机、网络设备、通信设备、自动化控制设备、定位监控系统}。
          \item 非法\zd{侵入}计算机信息系统。\zd{仅针对国家事务、国防建设}、登录访问。强保护。
          \item 非法\zd{获取}计算机信息系统数据。\zd{要求先侵入}。
          \item \zd{破坏}计算机系统。保护范围\zd{扩大到民营公司}。弱保护。
                \begin{enumerate}
                  \item 对\zd{功能}、数据、程序进行删除、修改、增加、干扰。\al{劫持域名、删评}。
                  \item 制作、传播病毒。
                \end{enumerate}
          \item 非法\zd{控制}计算机系统。\al{搞肉鸡}。
          \item 非法利用信息网络。可以和 N 多罪想象竞合，\al{在网上发贩毒信息、组建微信群、设立网站}。
          \item 帮助信息网络犯罪活动。\zd{帮信罪}。
                \begin{enumerate}
                  \item 提供互联网接入、网络存储、通讯传输等技术支持。
                  \item 可以和 N 多罪想象竞合，\al{甲明知乙实施网络诈骗，仍为其提供网络接入服务}。
                  \item 虽然是独立罪名，\zd{成立还是要严格遵循共犯从属性}。\al{实际乙并未去诈骗}。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
