\begin{enumerate}
  \item 一个名词，一个动词。
  \item 卖淫。
        \begin{enumerate}
          \item 行为主体。\zd{只能是人，男女随便排列组合}。\al{充气娃娃 \cuo}。
          \item 内容。要求发生关键的那一步。
          \item 支付方式。\zx{嫖资}可以是货币也可以是物，但\zx{不包括办事}。\al{甲乙完事后，甲请求乙辅导作业}。
        \end{enumerate}
  \item \zd{组织卖淫}、强迫卖淫、协助组织卖淫。\al{乙帮助组织卖淫的王某招募、运送卖淫女}。
        \begin{enumerate}
          \item 组织卖淫。\zx{体现管理性、控制性即可，不要求有实体，固定地点}。
          \item 强迫卖淫。\al{逼良为娼、不给上岸、竞业禁止}。
          \item 协助组织卖淫。\zd{不能扩大打击面}。招募、运送、保镖、财务人员。保安看工资待遇。\zx{帮助行为正犯化}。
        \end{enumerate}
  \item 引诱、容留、\zd{介绍卖淫}。
        \begin{enumerate}
          \item 引诱 \yjt 勾引。
          \item 容留同容留吸毒。\al{\xhx{航班上}空姐提供性服务，机长知道却放任}。
          \item 介绍 \yjt 拉皮条的/中介。vs 介绍嫖娼。介绍工作，为嫖客服务，站在嫖客立场。\al{发小卡片}。
        \end{enumerate}
  \item 引诱幼女卖淫。
  \item 传播性病。\zx{性病患者卖淫嫖娼}。\zx{扩大解释，包括艾滋病}，可能构成故意伤害罪，想象竞合择一重。
  \item \zd{淫秽物品的犯罪}。\zd{淫秽物品扩大解释}。\al{淫秽表演网络直播}。

          \item 走私淫秽物品。传播、牟利至少有一个。
        \end{enumerate}
