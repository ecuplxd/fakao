\section{主观要件 - 犯罪故意、罪过形式的区分}
\begin{enumerate}
  \item 犯罪故意。\bs{希望或者放任}。
        \begin{enumerate}
          \item \bs{主客观相一致}。主观认识和实际发生要相一致。
                \begin{enumerate}
                  \item \zx{行为、对象相一致}。
                        \begin{enumerate}
                          \item \al{不识字狗蛋卖书籍，不知里面是淫秽色情}。
                          \item \al{窝藏包庇罪}。
                        \end{enumerate}
                  \item 派生 \yjt \bs{客观决定主观原则。一个要素如果不是客观构成要件要素，则不要求主观认识到}。
                        \begin{enumerate}
                          \item \al{以为自己卖的是\xhx{鸦片}，实际卖的是\xhx{海洛因} \yjt
                                  主观认识到是\xhx{毒品}即可（类型性 \yjt \xhx{客观构成要件}要素），
                                  而\xhx{不需要具体认识}到是何种毒品。\xhx{类和实例的关系}}。
                          \item \al{甲以为树下的人是乙，误杀之 \yjt 对于故意杀人，在\xhx{人的范围内主客观相一致}}。
                        \end{enumerate}
                \end{enumerate}
          \item 理论分类。
                \begin{enumerate}
                  \item \zx{概括}故意。1 个故意。A、B、AB 均有可能。
                  \item \zx{择一}故意。2 个故意。要么 A、要么 B。\al{一杯毒药一杯饮料}。
                \end{enumerate}
          \item vs 生活中的故意、行政上的故意。\al{甲夜里为了盗窃仓库的物品，火柴照明、闯红灯}。
        \end{enumerate}
  \item 罪过形式区分。
        \begin{enumerate}
          \item \zx{直接}故意 \yjt 明知 + 希望。\zx{仅 1 种可能性}。\al{叶文洁，丈夫，政委。解开绳子，丈夫必死无疑}。
          \item \zx{间接}故意 \yjt 明知 + 放任。\zx{2 种可能性}。发生、不发生都可以。
          \item 各类过失。\yi \bs{预见的可能性大小。1、0.5、0}。\er \bs{认识条件（客观）+ 认识能力（主观）}。
                \begin{enumerate}
                  \item 已经预见 \yjt \zx{过于自信}。不想让结果发生，\zd{客观上有采取避免措施}。\al{高速开斗气车}。
                  \item 已经预见，\zx{无法避免 \yjt 不可抗力}。
                  \item 应当预见，有条件预见 \yjt \zx{疏忽大意}。\al{医生忘取纱布、马路边推人}。
                  \item \zx{无从预见 \yjt 意外事件}。\al{天黑车停路边下车掉井盖 \yjt 不具备客观认识条件}。
                \end{enumerate}
        \end{enumerate}
  \item \bs{步骤}。有无认识到会发生危害结果 \xjt。
        \begin{enumerate}
          \item \zx{有 \yjt
                对结果 \yjt
                希望/放任/反对 \yjt
                直接/间接/结果避免可能性 \yjt
                有/没有 \yjt
                过于自信/不可抗力}。
          \item \zx{没有 \yjt 结果预见可能性 \yjt 有/没有 \yjt 疏忽大意/意外事件}。
        \end{enumerate}

\end{enumerate}
