\section{共同犯罪 - 教唆犯、帮助犯}
\begin{enumerate}
  \item 教唆犯。故意引起（A）他人制造（故意 \hz 过失）违法事实（违法行为 + 违法结果）。
        \begin{enumerate}
          \item 客观。
                \begin{enumerate}
                  \item 成立。引起正犯制造违法行为。
                        \begin{enumerate}
                          \item \al{甲让乙杀丙，乙说\xhx{我早就想杀}了他}。
                          \item \al{乙想实施\xhx{诈骗}，甲说不如去\xhx{抢劫}}。
                          \item \al{乙想\xhx{抢劫}，甲说\xhx{带把枪}啊。\xhx{情节加重}。行为方式改变了}。
                          \item \al{乙想\xhx{抢 100}，甲说\xhx{抢 5W} 啊。\xhx{数额加重}。行为方式没改变。没考过}。
                          \item \al{乙想\xhx{抢劫}，甲说不如\xhx{去诈骗}。降低危险（两罪性质相同，程度不同）}。
                          \item \al{乙想\xhx{抢劫}，甲说不如\xhx{去猥亵}。降低危险但\xhx{引起新犯意}}。
                        \end{enumerate}
                  \item 既遂。引起正犯制造违法结果。
                  \item 存在因果关系。
                        \al{甲让乙杀丙，引起门卫丁怀疑，乙杀了丁后逃走。
                          乙对丙故意杀人犯罪预备，对丁故意杀人既遂。
                          甲对丙故意杀人犯罪预备（教唆犯），对丁不负责。
                          破除早期一人既遂，全部既遂}。
                \end{enumerate}
          \item 主观。故意。
                \begin{enumerate}
                  \item 说者无意听者有心。\al{甲问乙在外面靠啥发财，贩毒。乙也去贩毒}。
                  \item 故意教唆他人实施不能犯不构成教唆犯。
                \end{enumerate}
          \item 处罚。稍微了解即可。
        \end{enumerate}
  \item 帮助犯。故意促进他人制造（故意 \hz 过失）违法事实（违法行为 + 违法结果）。
        \begin{enumerate}
          \item 客观。
                \begin{enumerate}
                  \item 成立。促进正犯制造违法行为。可能有用。
                        \begin{kk}{帮助认定}
                          物理性帮助。\al{乙要盗窃丙家，让甲给把钥匙}。
                          \begin{enumerate}
                            \item \al{给了\xhx{假钥匙}，乙另行他法}。
                            \item \al{甲送钥匙途中钥匙\xhx{被偷}。\xhx{帮助行为未建立连接关系。帮助（行为）未遂}}。
                            \item \al{乙盗窃途中钥匙被偷。帮助犯预备}。
                            \item \al{门其实开着。帮助犯预备}。
                            \item \al{钥匙\xhx{打不开}门、钥匙锁孔里\xhx{被乙弄断}，乙破窗进入。\xhx{帮助犯未遂/未遂的帮助犯}}。
                            \item \al{一切顺利。既遂}。
                          \end{enumerate}
                          \tcblower
                          心理性帮助犯。\al{台湾现场助威罪}。
                        \end{kk}
                  \item 既遂。促进正犯制造违法结果。派上用场。
                  \item 存在因果关系。
                \end{enumerate}
          \item 主观。故意。\al{甲想乙借猎枪打猎，其实是拿枪杀丙}。
          \item 表述。\bs{丙明知甲乙实施贪污行为，客观上也帮助甲乙实施了贪污行为}。
          \item 中立的帮助行为。\al{明知、传授、仍}。
                \begin{enumerate}
                  \item 客观行为是否起到了实质紧迫的促进作用。\al{赌场卖茶叶蛋、出租车司机拉凶手到被害人跟前}。
                  \item 主观上是否明知对方在犯罪。\al{甲乙在五金店门前打架，老板卖菜刀给甲}。
                \end{enumerate}
        \end{enumerate}
  \item 教唆犯、帮助犯也有犯罪形态问题。容易忽视。
        \begin{enumerate}
          \item \al{甲盗窃丙，乙提供户型图，甲\xhx{未使用图纸}就找到珠宝 \yjt 乙帮助犯未遂}。
          \item 帮助（行为）未遂 \bdy 就是帮助犯 \bdy 帮助犯未遂。
                \al{甲欲\xhx{帮助}乙实施盗窃\xhx{行为，因意志以外的原因 \yjt 帮助未遂}，构成盗窃的\xhx{帮助犯未遂} \cuo}。
        \end{enumerate}
  \item \zd{共犯对正犯的要求}。原理基本同共同正犯。
        \begin{enumerate}
          \item 教唆犯 + 正犯。故意 + 过失还没考。非常重要。
                \begin{enumerate}
                  \item \al{医生\xhx{怂恿}护士}。
                  \item \al{医生\xhx{欺骗}护士 \yjt 支配力 \yjt 间接正犯}。
                  \item \al{护士将有毒针听成正常针。医生\xhx{无欺骗行为}（无支配力），护士是医生故意杀人客观阶段的“正犯”}。
                \end{enumerate}
          \item 帮助犯 + 正犯。\al{精神病患者乙在伤害丙，妻子甲\xhx{看到了不阻止}。甲不作为的帮助犯}。
          \item 对故意、过失不做要求。
          \item 对正犯责任年龄不做要求。\al{甲为儿子乙（15）虚报年龄考取警察，唆使乙刑讯逼供犯人丙}。
          \item 对正犯责任能力不做要求。
                \begin{enumerate}
                  \item \al{甲\xhx{教唆}精神病患者乙去伤害丙。甲是间接正犯（A + B）。\xhx{走了两步}，先构成教唆犯（A） \yjt 对乙有支配力（B） \yjt 升格为间接正犯}。
                  \item \al{乙是精神病患者，甲不知道 \yjt 甲不能定间接正犯（无支配力），只能定教唆犯}。
                \end{enumerate}
        \end{enumerate}
  \item 要得出一个符合正义感的结论是相对容易的，但是要给予充分论证是很难的。
\end{enumerate}
