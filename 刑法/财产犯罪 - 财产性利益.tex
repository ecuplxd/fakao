\section{财产犯罪 - 财产性利益}
\begin{enumerate}
  \item 抢劫财产性利益。债权债务。\al{应收账款}。只是放弃了当场实现债权的可能性，民法上并未失去债权。
        \begin{enumerate}
          \item \al{甲坐出租车，用 100 块\xhx{假钱付车费}，司机不收，甲拿刀\xhx{威胁找钱}，找了 90。抢劫 100（10 + 90）}。
          \item \al{甲欠乙 1000，后\xhx{不想归还}借款，\xhx{毒死}乙}。
          \item \al{甲预谋修车后\xhx{以假币骗付}，维修工乙发现有假拦住甲。甲迅速启动驶向厂门口，乙抓住雨刮器，甲仍加速，致乙摔成重伤}。
        \end{enumerate}
  \item 诈骗财产性利益。典型\zx{骗免债务}。\al{过高速伪装成军车，说你看我这军车你能收我钱么，不用收}。
        \begin{enumerate}
          \item \zx{非法的财产性利益}保\zx{不保护}。观点展示。
                \begin{enumerate}
                  \item 法律的财产说。刑法只保护\zd{民法上的合法的财产权}。
                  \item 经济的财产说。只要\zd{具有经济利益}。
                  \item \bs{法律与经济的财产说}。\zx{法秩序认可的}具有经济利益的利益 \yjt \zx{能去起诉}。
                  \item \al{嫖娼，\xhx{伪造}个人支票 1200 付嫖资，让妓女\xhx{找 400}。400 诈骗，骗免 800}。
                  \item \al{骗取赃物。乙\xhx{盗得}一辆摩托车，甲\xhx{谎称车主}要其返还}。
                  \item \al{主人骗回财物}。
                \end{enumerate}
        \end{enumerate}
  \item 盗窃财产性利益。
        \begin{enumerate}
          \item 逃单行为。
                \begin{enumerate}
                  \item 一开始就不想付费型。\al{假装顾客点菜吃完后，趁机溜走 \yjt 诈骗}。
                  \item 产生债务后不想付费。\al{使用假币付费 \yjt 使用假币、诈骗}。\sjt 都是“假”逃单行为。
                  \item 产生债务后不想付费。\al{（谎骗）+ 溜走}。
                        \begin{enumerate}
                          \item \zx{否定说。不构成}。\yi \zx{跑路行为没有转移占有效果}。\er 刑法谦抑性。该现象太普遍，监狱关不下。
                          \item 肯定说。符合要件，构成盗窃。
                        \end{enumerate}
                \end{enumerate}
          \item 电力与电费。
                \begin{enumerate}
                  \item \al{私拉电线 \yjt 真正的盗电}。
                  \item \al{先用电，月底缴纳，电表做手脚 \yjt 骗免电费，不是盗电}。
                  \item \al{电卡先充值，技术手段继续供电 \yjt 盗电}。
                \end{enumerate}
          \item 电话、网络。
                \begin{enumerate}
                  \item \al{盗打电话、盗用网络、蹭 wifi}。
                  \item 非法充值。
                  \item 诈骗电信工作人员。\al{假冒证件入网、先使用后交费}。
                \end{enumerate}
        \end{enumerate}
  \item 抢夺财产性利益。基本同盗窃。
  \item \bs{存款债权。储户是存款债权，现金是银行占有并所有的}。
        \begin{enumerate}
          \item \zx{错误汇款}。没考过。\al{乙本想给丙汇款，不慎打到了甲卡里}。
                \begin{enumerate}
                  \item \al{花掉 \yjt 不构成（种类物）}。
                  \item \al{拒不归还 \yjt 侵占}。
                  \item \al{从 ATM 取出。犯罪人\xhx{是否有取款权}。少数说/\xhx{多数说} \yjt 有权/\xhx{无权} \yjt 不构成/\xhx{构成盗窃}}。
                \end{enumerate}
          \item 取走赃款。\al{甲借卡给乙，乙自作主张实施电信诈骗，丙汇入，甲知道实情后挂失领取现金}。
          \item 帮取赃款。\al{乙诈骗丙使其汇入资金，乙告知甲，甲取柜台取走该现金}。
        \end{enumerate}
\end{enumerate}
