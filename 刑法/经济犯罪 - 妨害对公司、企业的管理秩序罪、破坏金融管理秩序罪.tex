\section{经济犯罪 - 妨害对公司、企业的管理秩序罪、破坏金融管理秩序罪}
\begin{enumerate}
  \item 妨害对公司、企业的管理秩序罪。
        \begin{enumerate}
          \item 非国家工作人员受贿罪。
          \item 对非国家工作人员行贿罪。
          \item 为亲友非法牟利罪。\zd{画交易流程图}。
                \begin{enumerate}
                  \item 国有企事业单位的人；\zd{可能和贪污罪想象竞合}。
                  \item 以明显\zd{高于/低于市场价}向自己的亲友单位采购/销售商品；低于情形没考过。
                \end{enumerate}
        \end{enumerate}
  \item 破坏金融管理秩序罪。
        \begin{enumerate}
          \item 货币犯罪。
                \begin{enumerate}
                  \item 伪造货币。
                        \begin{enumerate}
                          \item 制作假币，不能假的离谱，一般人相信。\al{天地银行}；
                          \item 正在流通，包括境外货币。\al{伪造宋朝交子}。
                        \end{enumerate}
                  \item 变造货币。\zd{在真币材料上小修小补}。\al{把硬币年份改了；完全翻新；熔铸}。
                  \item \zd{出售}、购买、运输假币。
                  \item 持有、\zd{使用}假币。
                        \begin{enumerate}
                          \item 使用。\zd{作为真币投入流通领域}。\al{把假币送给人；商场假币结账}。
                          \item 接收方知道 \yjt 使用方出售假币；不知道 \yjt 使用方使用假币。
                          \item \al{甲手头有 10W 假币，乙花 1W 买入} \yjt \al{到手后乙发现都是白纸} \yjt \al{甲发现 1W 都是假币}。
                          \item 罕见的相互骗，均是未遂。
                        \end{enumerate}
                  \item \zd{洗钱罪}。
                        \begin{enumerate}
                          \item 7 种上游犯罪，\zd{毒黑恐走贪两金}，破坏金融管理秩序罪、金融诈骗罪（不包括合同诈骗）。
                          \item 和上有犯罪的帮助犯的区分。
                                \begin{itemize}
                                  \item \al{甲对柜员乙说，我有 100W 用来贩毒，帮忙汇到境外}。事前帮（共谋了）\yjt 帮助犯。
                                  \item \al{甲对柜员乙说，我有 100W 毒资，帮忙汇到境外}。事后帮 \yjt 洗钱。
                                \end{itemize}
                          \item \zd{自洗钱}。\al{甲贩毒自己洗钱}。以前只定上游罪，现在这种行为猖獗，\zd{转为单独算，数罪并罚}。
                          \item \al{甲欲贿赂官员乙，乙说你汇到我海外账户吧}。
                        \end{enumerate}
                  \item 高利转贷罪。
                  \item \zd{骗取贷款罪}。喜欢和贷款诈骗罪一起考。区别，有没有非法占有目的，\zd{还是想还的}。
                  \item 伪造、变造金融票证。\zd{n 多金融诈骗的上游罪，牵连犯，择一重}。
                  \item 妨害信用卡管理。\zd{使用虚假身份证明骗领信用卡}。
                  \item 窃取、收买、非法提供信用卡信息。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
