\section{经济犯罪 - 妨害对公司、企业的管理秩序罪、破坏金融管理秩序罪}
\begin{enumerate}
  \item 妨害对公司、企业的管理秩序罪。
        \begin{enumerate}
          \item 非国家工作人员受贿。
          \item 对非国家工作人员行贿。
          \item \kd{为亲友非法牟利}。\zd{画交易流程图}。
                \begin{enumerate}
                  \item 主体限于国有企事业单位的人。\zd{可能和贪污想象竞合}。
                  \item 将本单位的盈利业务\zd{交由自己的亲友经营}。\al{B \yjt\, A \yjt\, D 变成了 B \yjt\, C \yjt\, D}。
                  \item 以明显\zd{高于/低于市场价}向自己的亲友单位\zd{采购/销售}。低于情形没考过。
                \end{enumerate}
          \item 非法经营同类营业。\zd{单位卖啥我卖啥}。
        \end{enumerate}
  \item 破坏金融管理秩序罪。
        \begin{enumerate}
          \item 货币犯罪。这块基本同毒品。
                \begin{enumerate}
                  \item \kd{伪造货币}。\zd{实在不能肯定选这个}。
                        \begin{enumerate}
                          \item 制作假币，不能假的离谱，一般人相信。\al{天地银行、英镑变美元}。
                          \item 范围。\zd{正在流通的境内外货币}。\al{伪造宋朝交子}。
                          \item 使用伪造货币定使用假币。
                        \end{enumerate}
                  \item \kd{变造货币}。
                        \begin{enumerate}
                          \item \zd{在真币材料上小修小补}。\al{把硬币年份改了 \dui、面额 50 变为面额 100 \dui、完全翻新、熔铸、一半真币一半假币拼接 \cuo}。
                          \item 使用变造货币不定使用假币（\sjt 两罪被独立开了）。触犯诈骗的，以诈骗论处。
                        \end{enumerate}
                  \item \sjt 后续的出售、购买、运输被吸收。
                  \item \zd{出售}、购买、运输假币。
                        \begin{enumerate}
                          \item 同毒品那块。\zd{忽略种类，定罪都只定此罪，不数罪并罚}。
                          \item \al{出售假美元、购买假人民币、运输假港币}。
                        \end{enumerate}
                  \item 持有、\zd{使用}假币。
                        \begin{enumerate}
                          \item 使用。作为真币\zd{投入流通}。\al{送人、结账、存钱 \yjt 不定诈骗，定使用假币}。
                          \item 赠送问题。\yi 接收方知道 \yjt 使用方出售假币。\er 不知道 \yjt 使用方使用假币。
                          \item \al{甲手头有 10W 假币，乙花 1W 买入 \yjt 到手后乙发现都是白纸 \yjt 甲发现 1W 都是假币。罕见的相互骗，均是诈骗未遂}。
                          \item \al{甲隐瞒真相，用假人民币向乙兑换真美元。甲使用假币、同时触犯诈骗}。
                          \item \al{甲窃取了 3 张借记卡，先后两次采取存入假币取出真币的方法，共从 ATM 内获取 6000 人民币。
                                  不构成信用卡诈骗罪，用的不是卡里主人的钱。实际盗窃了银行的钱}。
                        \end{enumerate}
                  \item 以假币换取货币。主体仅限金融工作人员。
                  \item \kd{洗钱}。
                        \begin{enumerate}
                          \item 7 种上游犯罪（不限于罪名，含行为、事实）。\zd{毒黑恐走贪两金}。两金 \yjt 破坏金融管理秩序、金融诈骗（\zd{不包括合同诈骗}）。
                                \al{由于甲实施了信用卡诈骗得行为，所以乙构成洗钱罪 \dui}。
                          \item vs 上游犯罪的帮助犯。\zd{是否知道此后用途去向}。
                                \begin{itemize}
                                  \item \al{甲对柜员乙说，我有 100W \xhx{用来}贩毒，帮忙汇到境外} \yjt 帮助犯。
                                  \item \al{甲对柜员乙说，我有 100W 毒资，帮忙汇到境外} \yjt 洗钱。
                                \end{itemize}
                          \item \zd{自洗钱}。\al{甲贩毒自己洗钱}。以前只定上游罪，现在这种行为猖獗，\zd{转为单独算，数罪并罚}。
                          \item \al{甲欲贿赂官员乙，乙说你汇到我海外账户吧}。
                          \item 成立。以上游犯罪的犯罪事实成立为前提。其他情况下都不影响审判工作（定罪量刑）。
                          \item 诉讼时效。互不干扰。
                        \end{enumerate}
                  \item 高利转贷。骗取贷款 + 转贷他人。
                  \item \kd{骗取贷款}。
                        \begin{enumerate}
                          \item vs 贷款诈骗。有没有非法占有目的（想不想还）。\zd{骗取贷款还是想还的}。
                          \item 要造成重大损失。
                          \item 主体。\zd{也可以是担保人}。
                        \end{enumerate}
                  \item 违法发放贷款。\zd{银行、金融机构工作人员}违反规定发放贷款。
                  \item \kd{伪造、变造金融票证}。\zd{N 多金融诈骗的上游罪，牵连犯，择一重}。
                  \item \kd{妨害信用卡管理}。\zd{使用虚假身份证明骗领信用卡}。
                  \item 窃取、收买、非法提供信用卡信息。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
