\section{社会秩序犯罪 - 妨害司法罪}
\begin{enumerate}
  \item 伪证。刑事诉讼领域的证人、鉴定人。
  \item 妨害作证。证人以外的言词证据。
        \begin{enumerate}
          \item 3 大诉讼领域。
          \item 暴力、胁迫或贿买阻止证人作证/指使他人做伪证。\al{甲是丙上司，为防止丙作证，将丙派往境外工作}。
        \end{enumerate}
  \item 辩护人妨害作证。基本同上。\sjt 和人有关，以证人为中心。\xjt 以物证为中心。
  \item 帮助毁灭、伪造证据。实物证据。3 个动词均要扩大解释。
        \begin{enumerate}
          \item 伪造包括变造。
          \item 毁灭包括隐藏。立法漏掉了。
          \item 帮助 = 狭义的帮助行为 + 实行行为 + 教唆行为。
                \begin{enumerate}
                  \item \al{甲自己埋尸，请乙提供铁锹 \yjt 乙帮助}。
                  \item \al{甲唆使乙帮助自己处理尸体，乙埋尸 \yjt 乙实行}。
                  \item \al{甲在乙的教唆下埋尸 \yjt 乙教唆}。
                \end{enumerate}
        \end{enumerate}
  \item \zd{虚假诉讼}。妨害司法秩序。新增。
        \begin{enumerate}
          \item 诉讼（扩大解释）= 民事诉讼 + 执行程序 \zx{- 仲裁}。
          \item 既遂。造成两个后果。\yi 开庭审理。\er 采取了财产保全或行为保全措施。
          \item 罪数。想象竞合择一重。
                \begin{enumerate}
                  \item \al{甲超市购物小票被乙捡到，乙说这是我的，不准走，甲说有种去告我}。
                  \item \al{乙和法官没有共谋。虚假诉讼、诈骗}。
                  \item \al{乙和法官合谋判甲败诉。\xhx{虚假诉讼、枉法裁判、盗窃罪} 3 个罪的共同犯罪}。
                \end{enumerate}
        \end{enumerate}
  \item 窝藏、包庇。嫌疑人/被告人。
        \begin{enumerate}
          \item 窝藏。帮助犯罪人逃匿。事前无通谋，事后帮助。
                \begin{enumerate}
                  \item 帮助 = 狭义的帮助行为 + 实行行为 + 教唆行为。同上。
                        \begin{enumerate}
                          \item \al{给路费、通风报信、让其住下}。
                          \item \al{取保候审期间逃，保证人知道跑哪了不告诉。定窝藏}。
                        \end{enumerate}
                \end{enumerate}
          \item 包庇。向司法人员、做假证帮助犯罪人逃避刑事追究 \hz 获得从宽处罚。
                \begin{enumerate}
                  \item \al{案发当天\xhx{乙和我在一起}}。
                  \item \al{乙让丁去“自首”，谎称案发当晚\xhx{是丁驾车。丁照办} \yjt 顶包，构成}。
                  \item \al{甲目睹了乙杀丙，民警询问，甲说，案发当天\xhx{我不在家} \yjt 知情不报，没有做假证}。
                \end{enumerate}
          \item 同一个犯罪人、一条龙服务不并罚，重吸收轻。\al{乙帮甲销赃、毁灭证据、隐藏证据、逃匿}。
        \end{enumerate}
  \item \zd{掩饰、隐瞒犯罪所得}、犯罪所得收益。保护司法机关的追赃秩序 \yjt 对数额无要求（赚多少都不重要）。
        \begin{enumerate}
          \item 赃物扩大解释。\yi 虚拟财产。\al{计算机数据}。\er 控制权。\al{肉鸡}。\san 犯罪工具不是。
          \item 方式/行为。\zx{窝藏、转移、代为销售、收购}。\al{装修车 \dui。装修房 \cuo}。
          \item \zd{事前参与 \yjt 上游犯罪的共犯}。\zd{事后参与 \yjt 帮助犯}。\al{甲把抢来的出租车送给乙}。
          \item 犯罪数额。新增。\yi 上游犯罪没达定罪数额，你不构成。\er 销赃不看数额。
          \item 主体。不包括本犯（含共犯），是本犯以外的第三人。\al{\xhx{甲用受贿}所得 1000W \xhx{购买了}一处别墅 \cuo}。
          \item \zd{客观上，xx 的行为属于掩饰、隐瞒行为。主观上，认识到可能是犯罪所得，因而具备明知的条件}。
        \end{enumerate}
  \item 脱逃。依法、被关押的、嫌疑人/被告人（未决犯、已决犯）！！
        \begin{enumerate}
          \item 不包括行政/治安拘留、强制戒毒、无罪的人员。\al{甲帮助乙做伪证，法院\xhx{判决乙无罪}释放，乙逃往海外}。
          \item 既遂。摆脱实际监管控制。\al{甲刚跑到监狱外的树林\xhx{即被抓回} \yjt 未遂}。
        \end{enumerate}
\end{enumerate}
