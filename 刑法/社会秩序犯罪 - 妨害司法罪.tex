\section{社会秩序犯罪 - 妨害司法罪}
\begin{enumerate}
  \item 伪证。刑事诉讼领域的\zd{证人、鉴定人}。
  \item 妨害作证。\zd{言词证据}。证人以外。
        \begin{enumerate}
          \item 3 大诉讼领域。
          \item \zd{暴力、胁迫或贿买}阻止证人作证/指使他人做伪证。\al{甲是丙上司，为防止丙作证，将丙派往境外工作}。
        \end{enumerate}
  \item 辩护人妨害作证。基本同上。\sjt \zd{和人有关，以证人为中心。\sjt 以物证为中心}。
  \item 帮助毁灭、伪造证据。\zd{实物证据}。3 个动词均要扩大解释。
        \begin{enumerate}
          \item 伪造包括变造。
          \item 毁灭包括隐藏。立法漏掉了。
          \item 帮助 = 狭义的帮助行为 + 实行行为 + 教唆行为。
                \begin{enumerate}
                  \item \al{甲自己埋尸，请乙提供铁锹 \yjt 乙帮助}。
                  \item \al{甲唆使乙帮助自己处理尸体，乙埋尸 \yjt 乙实行}。
                  \item \al{甲在乙的教唆下埋尸 \yjt 乙教唆}。
                \end{enumerate}
        \end{enumerate}
  \item \bs{虚假诉讼}。妨害\zd{司法秩序}。新增。
        \begin{enumerate}
          \item 诉讼（扩大解释）= \zd{民事诉讼} + 执行程序 - \zd{仲裁}。
          \item 既遂。造成两个后果。\yi 开庭审理。\er 采取了财产保全或行为保全措施。
          \item 罪数。想象竞合择一重。
                \begin{enumerate}
                  \item \al{甲超市购物小票被乙捡到，乙说这是我的，不准走，甲说有种去告我}。
                  \item \al{乙和法官没有共谋。虚假诉讼、诈骗}。
                  \item \al{乙和法官合谋判甲败诉。虚假诉讼、枉法裁判、盗窃罪 3 个罪的共同犯罪}。
                \end{enumerate}
        \end{enumerate}
  \item \bs{窝藏、包庇}。嫌疑人/被告人。
        \begin{enumerate}
          \item 窝藏。帮助犯罪人\zd{逃匿}。\bs{事前无通谋，事后帮助}。
                \begin{enumerate}
                  \item 帮助 = 狭义的帮助行为 + 实行行为 + 教唆行为。同上。\al{给路费、通风报信、让其住下}。
                  \item \al{取保候审期间逃，保证人知道跑哪了不告诉，定窝藏}。
                \end{enumerate}
          \item 包庇。\bs{向司法人员做假证}帮助犯罪人\zd{逃避刑事追究 \hz 获得从宽处罚}。
                \begin{enumerate}
                  \item \al{案发当天乙和我在一起}。
                  \item 顶包。\al{乙让丁去“自首”，谎称案发当晚是丁驾车。丁照办}。
                \end{enumerate}
          \item 知情不报。\al{甲目睹了乙杀丙，民警询问，甲说，案发当天我不在家}。
          \item \bs{同一个犯罪人一条龙服务不并罚，重吸收轻}。\al{销赃、毁灭证据、隐藏证据、逃匿}。
        \end{enumerate}
  \item \bs{掩饰、隐瞒犯罪所得、犯罪所得收益}。保护司法机关的\zd{追赃秩序} \yjt 不要求数额了（赚多少都不重要）。
        \begin{enumerate}
          \item 赃物扩大解释。\yi 虚拟财产。\al{计算机数据}。\er 控制权。\al{肉鸡}。犯罪工具不是。
          \item 方式。窝藏、转移、代为销售、收购。\al{装修车 \dui，装修房 \cuo}。
          \item \bs{事前参与 \yjt 上游犯罪的共犯。事后参与 \yjt 帮助犯}。\al{甲把抢来的出租车送给乙}。
          \item \bs{犯罪数额}。新增。\yi 上游犯罪没达定罪数额，你不构成。\er \zd{销赃不看数额}。
          \item \bs{主体。不包括本犯（含共犯），本犯以外的第三人}。\al{甲用受贿所得 1000W 购买了一处别墅 \cuo}。
        \end{enumerate}
  \item 脱逃。\bs{依法被关押的嫌疑人/被告人（未决犯、已决犯）。不包括行政/治安拘留、强制戒毒人员、无罪的人}。
        \begin{enumerate}
          \item \al{甲帮助乙做伪证，法院判决乙无罪释放，乙逃往海外}。
          \item 既遂。\zd{摆脱实际监管控制}。\al{甲刚跑到监狱外的树林即被抓回 \yjt 未遂}。
        \end{enumerate}
\end{enumerate}
