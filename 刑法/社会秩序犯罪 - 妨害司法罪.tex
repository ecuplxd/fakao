\section{社会秩序犯罪 - 妨害司法罪}
\begin{enumerate}
  \item 伪证。刑事诉讼领域。证人、鉴定认定。
  \item 妨害作证。\zd{言词证据}。
        \begin{enumerate}
          \item 3 大诉讼领域。
          \item \zd{暴力、胁迫或贿买}、阻止证人作证、指使他人做伪证。
        \end{enumerate}
  \item 辩护人妨害作证。基本同上。
  \item 帮助毁灭、伪造证据。\zd{实物证据}。3 个动词均要扩大解释。
        \begin{enumerate}
          \item 伪造包括变造。
          \item 毁灭包括隐藏。立法漏掉了。
          \item 帮助 = 狭义的帮助行为 + 实行行为 + 教唆行为。
                \begin{enumerate}
                  \item \al{甲自己埋尸，请乙提供铁锹}。乙帮助。
                  \item \al{甲唆使乙帮助自己处理尸体，乙埋尸}。乙实行。
                  \item \al{甲在乙的教唆下埋尸}。乙教唆。
                \end{enumerate}
        \end{enumerate}
  \item 虚假诉讼。较新。
        \begin{enumerate}
          \item 提起虚假的\zd{民事诉讼}，包括执行程序（扩大解释），但不包括仲裁。
          \item 既遂。妨害了司法秩序。造成两个后果。开庭审理 \hz 财产保全或行为保全。
          \item 罪数。想象竞合，择一重。
                \begin{enumerate}
                  \item \al{甲超市购物小票被乙捡到，乙说这是我的，不准走，甲说有种去告我}。
                  \item \al{乙和法官合谋判甲败诉。3 个罪}。
                \end{enumerate}
        \end{enumerate}
  \item 窝藏、包庇。嫌疑人、被告人。
        \begin{enumerate}
          \item 窝藏。\zd{事前无通谋，事后帮助}。
                \begin{enumerate}
                  \item 帮助犯罪人\zd{逃匿}。
                  \item 帮助 = 狭义的帮助行为 + 实行行为 + 教唆行为。同上。\al{给路费、通风报信、让其住下}。
                \end{enumerate}
          \item 包庇。
                \begin{enumerate}
                  \item \zd{向司法人员做假证明}帮助犯罪人\zd{逃避刑事追究 \hz 获得从宽处罚}。\al{案发当天乙和我在一起}。
                  \item 顶包。\al{乙让丁去“自首”，谎称案发当晚是丁驾车。丁照办}。
                \end{enumerate}
          \item 知情不报。\al{甲目睹了乙杀丙，民警询问，甲说，案发当天我不在家}。
        \end{enumerate}
  \item 掩饰、隐瞒犯罪所得、犯罪所得收益。保护司法机关的追赃活动，司法秩序。
        \begin{enumerate}
          \item \zd{虚拟财产（计算机数据）也能成为赃物}。犯罪工具不是。
          \item 方式。窝藏、转移、代为销售、收购。
          \item 事后参与 \yjt 帮助犯。事前参与 \yjt 上游犯罪的共犯。\al{甲把抢来的出租车送给乙}。
          \item \zd{犯罪数额}。新考点。\yi 上游犯罪没达定罪数额，你不构成。\er \zd{销赃不看数额}。
        \end{enumerate}
  \item 脱逃罪。
        \begin{enumerate}
          \item \zd{依法被关押的嫌疑人、被告人（未决犯、已决犯）}。不包括行政/治安拘留、强制戒毒人员。
          \item 既遂。\zd{摆脱实际监管控制}。\al{甲刚跑到监狱外的树林即被抓回}。
        \end{enumerate}
\end{enumerate}
