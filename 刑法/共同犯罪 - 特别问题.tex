\section{共同犯罪 - 特别问题}
\begin{enumerate}
  \item 片面的共犯。以为自己在单干。\al{甲欲杀丙，看到乙在追杀丙，便\xhx{暗中设置绳索}将丙绊倒，乙顺利杀死丙}。
        \begin{enumerate}
          \item 观点展示。
                \begin{enumerate}
                  \item 肯定说。单方面的意思联络、参与意识即可。\al{甲故意杀人的片面帮助犯（既遂），乙故意杀人既遂}。
                  \item 否定说。必须是相互的 \yjt 单独定罪。
                \end{enumerate}
          \item 类型。成立片面犯，一定要起到实质性帮助作用！！（物理、心理）。
                \begin{enumerate}
                  \item 片面帮助 \yjt 没有争议，用肯定说。\al{甲看到乙入户盗窃，为其望风}。
                        \begin{enumerate}
                          \item \al{\xhx{乙不知情}。没有异常发生。既遂后得知甲为自己望风，分给甲 1000}。
                          \item \al{期间主人丙回来了，甲拉住丙\xhx{拖延时间}，乙顺利盗窃既遂}。
                        \end{enumerate}
                  \item 片面教唆 \yjt 没有争议，用肯定说。
                  \item 片面实行 \yjt 用肯定说。\input{draw/片面实行.tex}
                  \item 片面共犯 + 无法查明 \yjt 不构成片面犯 \yjt 不构成共同犯罪 \yjt 退化为故意的同时犯。
                \end{enumerate}
        \end{enumerate}
  \item 共犯与身份。
        \begin{enumerate}
          \item 分别分析 + 想象竞合。\al{投保人和保险经理一起合谋骗钱}。
          \item 传统。两个人构成共同犯罪，最终一定要定同一个罪名。\input{draw/特别问题.tex}
        \end{enumerate}
  \item 和不作为犯。
        \begin{enumerate}
          \item 步骤。
                \begin{enumerate}
                  \item 先把作为义务人找出来。有些作为义务人不好找。考先行行为（有没有陷入危险状态）。
                  \item 再找没有作为义务的人。
                  \item 再看他们的行为。实行行为/教唆/帮助 \yjt 构成不作为的实行/教唆/帮助犯。
                \end{enumerate}
          \item 例子。
                \begin{enumerate}
                  \item \al{甲高位截瘫多年，其\xhx{\yi 妻乙}不经困扰，\xhx{\san 让}\xhx{\er 丙}小推车\xhx{\san 把甲丢到路边} \yjt 乙不作为的教唆犯}。
                  \item \al{\xhx{\yi 甲}看到自己年迈的母亲长期被\xhx{\er 保姆}\xhx{\san 虐待}，\xhx{\san 放任不管} \yjt
                          甲不作为帮助犯、不作为的片面正犯。如果保姆知道甲的存在 \yjt 不作为的正犯}。
                  \item \al{甲乙一起打伤丙女，此时\xhx{\er 乙}欲\xhx{\san 行苟且之事}，\xhx{\yi 甲}\xhx{\san 不阻止} \yjt 甲不作为帮助犯}。
                \end{enumerate}
        \end{enumerate}
  \item 和实行过限。A、B 两罪是否超出共同故意的范围。变味了。
        \begin{enumerate}
          \item \al{甲乙\xhx{盗窃}小芳，甲在客厅，乙在卧室偷，\xhx{强奸}了小芳}。
          \item \al{甲教唆乙\xhx{盗窃}小芳，乙不仅实施了盗窃还进行了\xhx{强奸}}。
          \item \al{甲雇乙去教训丙，\xhx{嘱咐不要弄死}，丙反抗，乙杀之。甲 \yjt 故意伤害 + 过失致人死亡 \yjt 拟制}。
          \item \al{乙要\xhx{盗窃}丙女，嘱咐\xhx{甲帮忙望风}，乙盗窃过程中还顺便\xhx{强奸}了丙女}。
          \item vs 实行不足。A、B 同性质且 B \xyh A。
                \begin{enumerate}
                  \item \al{甲教唆乙\xhx{入户抢劫}，但乙实施的是拦路\xhx{抢劫}。甲是抢劫罪的共犯 \dui。入户抢劫包容评价拦路抢劫}。
                  \item \al{甲以为乙要\xhx{杀人}为其提供杀人凶器，结果乙仅欲\xhx{伤害}他人}。
                \end{enumerate}
        \end{enumerate}
  \item 和认识错误的结合。难度很大。但没有新的知识点。
        \begin{enumerate}
          \item \al{甲乙欲杀博物馆的丙，各开一枪，甲打中了花瓶，乙啥都没打中}。
          \item \al{甲教唆乙杀害丙，丙照片给乙，嘱咐不要杀错人了，结果还真杀错了}。\input{draw/认识错误.tex}
          \item \al{言明不要造成其他后果，乙几次都未果，决定\xhx{爆炸丙家}，导致丙妻子丁死亡}。\input{draw/认识错误2.tex}
          \item \al{甲要杀死丙，误以为前方丁是丙，\xhx{指示乙开枪}，乙照办}。
        \end{enumerate}
  \item 教唆犯和间接正犯之间产生认识错误。一律定教唆犯，不定间接正犯 \yjt 没有支配力。
        \al{甲\xhx{让乙}去杀丙，\xhx{不知乙是精神病患者}，乙果杀了丙}。
  \item 共犯人成为被害人。两两解构。
        \begin{enumerate}
          \item \al{甲想盗窃丙的渔网，让乙提供工具。甲到海边，没看到丙的渔网，把旁边乙的渔网拿走了}。
          \item \al{乙女教唆甲男强奸室友丙女，结果当天丙女不在，甲误把乙女当丙女进行了强奸。甲对丙强奸犯罪预备，对乙强奸既遂。乙对丙强奸的教唆犯（犯罪预备）}。
        \end{enumerate}
\end{enumerate}
