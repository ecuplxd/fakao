\section{客观违法阻却事由 - 被害人承诺}
\begin{enumerate}
  \item 一般的被害人承诺。\bs{权限范围、时间、能力、意思 \yjt 自己；事实、动机受骗}。
        \begin{enumerate}
          \item 权限和范围。\al{轻伤、名誉、人身自由 \dui。重伤、生命 \cuo}。
          \item 时间。\zx{事前承诺}。时间连续型的分阶段讨论。
          \item 客观性。被害人承诺是客观存在的，\zx{不要求行为人主观意识到}。\al{SM}。
          \item 承诺能力。对承诺的事项\zx{正确的理解能力}。\al{骗小孩、精神病人的钱 \yjt 盗窃}。
          \item 承诺的\bs{意思表示}。\zx{真实}。考试喜欢考欺骗。\bs{也适用于认定诈骗中}。\bs{自己；事实、动机受骗}。
                \begin{kk}{自己产生认识错误}
                  \begin{enumerate}
                    \item 不知道 \yjt 有效。
                    \item 知道，但\zx{仅单纯利用 \yjt 有效}。其实就是\zd{看你笑话}的意思。
                    \item 知道且\bs{继续实施欺骗行为（维持、强化） \yjt 无效}。\al{司机来补轮胎}。
                          \begin{enumerate}
                            \item \zx{有告知义务 \yjt 不作为欺骗}。\al{乙以为自己的马绝症了请求兽医安乐死}。
                            \item \zx{无告知义务 \yjt 作为欺骗}。\zd{犹疑不定 \yjt 内心确信}。\al{乙以为自家着火请求甲灭火，甲说火势很大}。
                          \end{enumerate}
                  \end{enumerate}
                \end{kk}

                \begin{kk}{受欺骗产生认识错误}
                  \begin{enumerate}
                    \item \zx{事实性欺骗}。即如果我\bs{知道则不会}。\al{尹志平小龙女 \yjt 事实性错误，承诺无效 \yjt 强奸}。
                    \item \zx{动机性欺骗}。\bs{动机错误不影响被害人承诺有效性}。\zx{包括卑劣的动机}，不要带入道德审判。
                          \begin{enumerate}
                            \item \al{导演骗女演员 \yjt 动机错误，承诺有效 \yjt 不构成强奸}。
                            \item \al{甲汽修店老板为了让司机们前来补胎，\xhx{高速路口撒钉子} \yjt 该行为本身不构成欺骗}。
                          \end{enumerate}
                  \end{enumerate}
                \end{kk}
        \end{enumerate}
  \item 推定的被害人承诺。
        \begin{enumerate}
          \item \zx{一般人标准}。不一定以被害人为标准。\al{落水救人 \yjt 即便被害人本身是想自杀}。
          \item 可能\zx{和紧急避险竞合}。\al{坚决要求顺产结果双双死亡、疫情期间送医}。
        \end{enumerate}
\end{enumerate}
