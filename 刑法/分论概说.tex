\section{分论概说}
\begin{enumerate}
  \item 侵害个人、财产、国家法益。\bs{人、财、公、经、社、贪、渎}。
        \begin{enumerate}
          \item \zd{财产}。6 大罪名。
          \item \zd{人身}。
          \item \zd{贪污、贿赂}。\sjt 一半分值，抓大放小。
          \item 危害公共安全、经济、社会秩序、渎职。
        \end{enumerate}
  \item 选择性罪名。\al{非法制造、买卖、运输、邮寄、储存枪支、弹药、爆炸物}。\bs{主要考罪数和认识错误问题}。
        \begin{enumerate}
          \item 打顿号的罪名。\zx{可以拆分使用，也可以整体使用}。
          \item \bs{对象认识错误。看客观，客观是啥定啥}。选择性罪名的认识错误是\zd{同一犯罪构成内}的错误。
                \begin{enumerate}
                  \item \al{买的枪，寄的子弹}。
                  \item \al{甲盗得一个手提包，结果里面是手枪 \yjt 盗窃罪 \dui。盗窃枪支罪 \cuo}。
                \end{enumerate}
                % \item 非选择性罪名的认识错误是不同犯罪构成间的错误。\al{生产、销售、提供假药/生产、销售、提供劣药。主观卖假药，客观卖劣药}。
          \item 串联（有先后）不并罚。\al{购买、出售}。并联（同步进行）并罚。\al{运输、使用}。
        \end{enumerate}
  \item 兜底罪名。\al{生产、销售伪劣产品；走私禁止进出口的货物、物品}。
  \item 罪状。\bs{简单、叙明、引证、空白}。
        \begin{enumerate}
          \item 简单罪状。
          \item 叙明罪状。描述的很详细。
          \item \zx{引证}罪状。引用\zx{内部}。
          \item \zx{空白}罪状。援引刑法\zx{外部}。
        \end{enumerate}
  \item 特殊规定。\zx{把假定条件取消，还能否按条文定}。
        \begin{enumerate}
          \item 注意规定。正常办事，重复强调。\al{携带挪用的公款\xhx{潜逃国外}定贪污。法官注意，这种情况要定贪污}。
          \item 法律拟制。特事特办。常考。
                \begin{enumerate}
                  \item \bs{拟制的抢劫}。\al{事后转化抢劫、携带凶器抢夺、聚众打砸抢过程中毁坏/抢夺财物}。
                  \item \zx{过失拟制的故意杀人}。\bs{故意、非法、聚众；暴力、虐待、刑讯}。\al{故意伤害、非法拘禁、聚众斗殴、暴力取证、虐待被监管人、刑讯逼供}。
                \end{enumerate}
        \end{enumerate}
  \item 考。罪与罪的区分、罪与罪的联系（罪数知识点）。\bs{不要记罪名的定义，记行为模式/公式}。
  \item 一些动、名词的解释。
        \begin{enumerate}
          \item 买卖 = 买 + 卖。\zx{2 个行为}。
                \begin{enumerate}
                  \item 有偿。物物、钱物交易均可。
                  \item 买。为自己使用而买进、为卖出而买进。
                  \item 卖。单纯的卖出、先买进后卖出（转卖）。
                  \item 除了毒品、枪支弹药，原则只处罚销售者，不处罚购买者 \yjt \zx{购买者不是销售者的帮助犯}。
                  \item vs \zx{贩卖。只有卖构成}。
                  \item 出售、销售、倒卖、贩卖、出卖、出租、出借、赠与。根据民法衡量。
                \end{enumerate}
          \item 生产。销售目的（\zd{非自用}）。
          \item 销售。\zd{\bdy 捐赠}。
          \item 财物。有体、无体/虚拟、财产性利益、债权、股票、应收款、罚款等。\zd{有经济价值}。
          \item 助。帮助行为正犯化。
          \item 传播。向社会公众。
          \item 他人。\bs{一定要排除“合法”所有、占有本尊}。
        \end{enumerate}
  \item 正犯化的含义。（定罪量刑时）不再适用总则关于从犯/预备犯的规定。
        \begin{enumerate}
          \item 准备行为正犯化。\al{准备实施恐怖活动}。
          \item 帮助行为正犯化。\al{帮助恐怖活动、协助组织卖淫}。
        \end{enumerate}
  \item 特殊的罪。
        \begin{enumerate}
          \item 逃税。处罚阻却事由，独一份。
          \item 骗取出口退税。想象竞合数罪并罚，独一份。
          \item 食品、药品监管渎职。故意、过失共享同一个法定刑。
        \end{enumerate}
  \item 做题小技巧。\bs{找法益、找主体、找行为、找对象}。
        \begin{enumerate}
          \item 看到\zd{环境}丢给社会秩序犯罪 - 环境犯罪。
          \item 看到\zd{食品}丢给经济犯罪 - 食品犯罪。
          \item \al{缉毒警察\scx{掩护、包庇走私毒品的犯罪分子，构成}放纵走私罪}。
          \item \al{\scx{聚众锯断高速公路}隔离栏，\scx{成立聚众扰乱交通}秩序\scx{罪}。有人参与才有秩序可言}。
          \item \al{\scx{甲将纯净水掺入到}工业酒精\scx{中，冒充白酒销售。
                    甲的行为不属于“在生产、销售的食品中掺入有毒、有害的非食品原料”，
                    不成立生产、销售}有毒、有害\scx{食品罪}}。
          \item \al{\scx{甲从山林间拾得许多蘑菇，明知是}毒蘑菇\scx{还卖给别人，其行为构成销售}有毒、有害\scx{食品罪}}。
        \end{enumerate}
\end{enumerate}
