\section{执行}
\begin{enumerate}
  \item 财产性的执行是重点。
  \item \bs{生效裁判}。
  \item 法院/监狱/公安/社区矫正。\zd{收钱杀人不干活/徒刑监禁 3 月起/不足 3 拘夺政权/社区最闲陪你玩}。
  \item 死刑。
        \begin{enumerate}
          \item 最高法院长签发死刑令。
          \item 原则上\zd{一审法院}执行，\zd{死缓在监狱}执行，\zd{死立执服刑地中院}。
          \item 收到执行令后 7 天内、刑场 \hz 羁押场所、枪决 \hz 注射。其他方法最高法批准。
          \item 会见权。
                \begin{enumerate}
                  \item 应当告知临死前有权会见\zd{近亲属}。申请会见并提供联系方式的应当通知。
                  \item 近亲属以外 = 申请 + 审查 + 正当理由 + 确保安全。
                  \item 会见未成年子女经监护人同意。
                  \item 检察院派员\zd{到场监督}，临死前要\zd{验明正身}，问他有没有什么遗言。
                \end{enumerate}
        \end{enumerate}
  \item 死缓 2 年、无期徒刑、有期徒刑和拘役。
        \begin{enumerate}
          \item 生效后 10 天内\zd{只交文书不交罪犯}。判决书、裁定书、起诉书副本。
          \item 同案犯如果和死刑犯有牵连，最高法\zd{复核问讯死刑犯后}才可交付执行。
          \item 执行期满发释放证明书。
        \end{enumerate}
  \item 剥夺政治权利。不重要。
        \begin{enumerate}
          \item 死刑、无期的，剥夺终身。减刑后，改为 [3, 10]。
        \end{enumerate}
  \item \kd{财产刑}。罚金、没收财产（立即，现有财产）。
        \begin{enumerate}
          \item 原则一审法院 \yjt 财产所在地的同级法院来\zd{代}（比如不动产）。
          \item \zd{所有 = 合法拥有}。追缴赃款赃物一追到底（含孳息）。善意第三人不追缴，纠纷通过其他诉讼程序解决。
          \item 罚金。一次缴纳 \hz 分期缴纳，交够为止，有钱就\zd{应当}及时追缴、应当查扣冻。
          \item 申请免除罚金，一定是不可抗拒的灾祸。
          \item 不是所有债务，而是\zd{正常债务}。
          \item 已被侦查机关\zd{查扣冻的，无需解除}即可直接处置。
          \item \zd{按顺序}而不按比例清偿，\zd{医赔债罚没}，优先受偿权插队 \yjt \zd{医债赔罚没}。
          \item \zd{执行回转（只有财产刑有）}。返还原物/价金、赔偿相关的损失。
        \end{enumerate}
  \item 缓刑。暂缓执行。审判阶段作出，用宣告。\zd{附条件的不再执行}。家里蹲。
        \begin{enumerate}
          \item 对象。\xydy 3 年有期或拘役。\zd{累}犯、\zd{首}要犯罪分子不行。
          \item 考验期。判 3/5 年 \yjt 2/3 年。
          \item 生效后 10 日内到\zd{经居地社区矫正机构}报道。
          \item 有重大立功就可以减刑。\al{不认真遵守监规、不接受教育改造 \cuo}。
          \item 缓刑考验。
                \begin{enumerate}
                  \item 漏罪 \hz 新罪。\zd{谁判谁撤}。
                  \item 违法违规。谁决定谁来撤，\zd{做出生效}，不能上诉不能抗。
                  \item 成功。\zd{原判刑罚不再执行 \bdy 刑罚执行完毕}。
                \end{enumerate}
          \item 管制。先行羁押 1 日折抵刑期 2 日。
        \end{enumerate}
  \item \zd{执行的变更}。刑种发生变化。
        \begin{enumerate}
          \item 死刑。
                \begin{enumerate}
                  \item 发现\zd{其他}犯罪/证据、\zd{共犯}到案、\zd{立功}、怀\zd{孕} \yjt 暂停执行。
                  \item 下级法院发现错。暂停 \yjt 层报高院 \yjt 裁定停止 \yjt 核实后 \yjt 层报最高法。N 字。
                  \item 高院发现错。裁定停止 \yjt 核实后 \yjt 层报最高法。V 字。
                  \item \sjt 2 个 \yjt 最高法处理见死刑复核。
                \end{enumerate}
          \item 死缓 2 年。
                \begin{enumerate}
                  \item 从裁判宣告 \hz 送达之日起算\zd{考验期}。
                  \item 转死刑。2 年内故意犯罪，情节恶劣，层报核准。
                  \item 减刑。2 年期满 \yjt 无期。重大立功 \yjt 25 年有期。2 年期满起算。即刑期最低为 25 + 2 = 27。
                  \item 限制减刑（只针对死刑）。累犯、杀伤奸抢投毒火爆。
                \end{enumerate}
          \item 暂予监外执行。\zd{叫终止}，不叫解除。
                \begin{enumerate}
                  \item 有期、拘役、特殊的无期。\zd{疾病、孕乳、不自理}。\zd{省政府指定}的医院。\zd{死缓不行}。
                  \item 决定。
                        \begin{enumerate}
                          \item 执行前。法院。
                          \item 执行后。具体分析。监狱 \hz 看守所提书面意见 \yjt \zd{省监狱管理机关} \hz 市以上公安批准。
                        \end{enumerate}
                  \item 收监。\zd{不符条件、违规定、情形消失期未满}。\zd{谁决定出去，谁决定回来}。
                  \item 社区矫正机关执行。
                  \item 贿赂非法、\zd{脱逃}出去的期间\zd{不计}入刑期。
                \end{enumerate}
          \item \kd{减刑}。\zd{针对自由刑}。
                \begin{enumerate}
                  \item 无次数限制，但有幅度和限度要求。
                  \item 管制、拘役、有期至少执行原判刑期 1/2、无期至少 13 年。
                  \item \zd{无期死刑在高院}，剩下在中院。审期 1 个月、合议庭。
                  \item 考验。
                        \begin{enumerate}
                          \item 漏罪新罪 \yjt 谁审谁撤。
                          \item 违法违规 \yjt 原决定法院来撤销。
                          \item 遵守规定 \yjt 视为执行完毕。
                        \end{enumerate}
                  \item 一定要开庭。
                        \begin{enumerate}
                          \item \zd{重大立功不一般；影响重大有意见；有权有势又有钱}。
                          \item 破格减刑。\zd{职务、黑}社会性质、\zd{金}融类犯罪。
                          \item 应当通知\zd{检、执行机关、罪犯}出庭。
                        \end{enumerate}
                \end{enumerate}
          \item \kd{假释}。针对有期、无期。有条件提前释放。执行阶段作出，用裁定。
                \begin{enumerate}
                  \item 不得假释。
                        \begin{enumerate}
                          \item 累犯。
                          \item 杀伤奸抢投毒火爆组。
                                \yi 单个判 \dydy10 年。
                                \er 多个同类罪数罪并罚 \dydy 10。\al{投毒 8，盗窃 5 \yjt 13 \cuo。投毒 8，强奸 5 \yjt 13 \dui}。
                        \end{enumerate}
                  \item 实质条件。\zd{无人身危害性、再犯可能性}。确有悔改表现（\al{退赃}）。
                  \item 成功。\zd{原判刑罚视为执行完毕}。
                  \item 特别假释（\al{原判刑期没执行够}）\yjt 最高法。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
