\section{执行}
\begin{enumerate}
  \item 财产性的执行是重点。
  \item \bs{生效裁判}。
  \item 法院/监狱/公安/社区矫正。\bs{收钱杀人不干活/徒刑监禁 3 月起/不足 3 拘夺政权/社区最闲陪你玩}。
  \item 死刑。\bs{地点、方式、会见、监督}。
        \begin{enumerate}
          \item 最高法院长签发死刑令。
          \item 原则上\zd{一审法院}执行，\zd{死缓在监狱}执行，\zd{死立执服刑地中院}。
          \item 收到执行令后 \zd{7} 日内、刑场 \hz 羁押场所、枪决 \hz 注射。其他方法最高法批准。
          \item 会见权。
                \begin{enumerate}
                  \item 应当告知临死前有权会见\zd{近亲属}。申请会见并提供联系方式的应当通知。
                  \item 会见未成年子女经监护人同意。
                  \item 会见近亲属以外，一系列操作 = 申请 + 审查 + 正当理由 + 确保安全。
                  \item 检察院派员\zd{到场监督}，临死前要\zd{验明正身}，问他有没有什么遗言。
                \end{enumerate}
          \item 死刑变更。\bs{刀下留人}。
                \begin{enumerate}
                  \item \bs{其他、共犯、立功、孕，暂停执行}。发现其他犯罪/证据、共犯到案、立功、怀孕。
                  \item 下级法院发现错。暂停 \yjt 层报高院 \yjt 裁定停止 \yjt 核实后 \yjt 层报最高法。N 字。
                  \item 高院发现错。裁定停止 \yjt 核实后 \yjt 层报最高法。V 字。
                  \item \sjt 2 个 \yjt 最高法处理见死刑复核。
                \end{enumerate}
          \item 死缓变更。\bs{求生求死}。
                \begin{enumerate}
                  \item 从裁判宣告 \hz 送达之日起算\zd{考验期}。
                  \item 减刑。2 年期满 \yjt 无期。重大立功 \yjt 25 年有期。2 年期满起算。即刑期最低为 25 + 2 = 27。
                  \item 限制减刑（只针对死刑）。\zd{累犯、杀伤奸抢投毒火爆}。
                  \item 转死刑。2 年内故意犯罪、情节恶劣、层报核准。
                \end{enumerate}
        \end{enumerate}
  \item 死缓、无期、有期和拘役。\bs{文书、共犯死刑}。
        \begin{enumerate}
          \item 生效后 10 日内\bs{只交文书不交罪犯}。判决书、裁定书、起诉书副本。
          \item 同案犯如果和死刑犯有牵连，\bs{最高法复核问讯死刑犯后}才可交付执行。
          \item 执行期满发释放证明书。
        \end{enumerate}
  \item 剥夺政治权利。死刑、无期，剥夺终身。\zd{减刑后，改为 3 - 10 年}。不重要。
  \item 财产刑。罚金、没收财产（立即，现有财产）。\bs{追、缴、收、赔、回转}。
        \begin{enumerate}
          \item 原则\bs{一审法院} \yjt 财产所在地的同级法院来\zd{代}（比如不动产）。
          \item \zd{所有 = 合法拥有}。赃款赃物及孳息一追到底。善意第三人不追缴，纠纷通过其他诉讼程序解决。
          \item 罚金。一次缴纳 \hz 分期缴纳。交够为止，有钱就\zd{应当}及时追缴、应当查扣冻。
          \item 一定是不可抗拒的灾祸，才能申请免除罚金。
          \item 不是所有债务，而是\zd{正常债务}。
          \item 已被侦查机关\bs{查扣冻的，无需解除}即可直接处置。
          \item \zd{按顺序清偿}而不按比例。\bs{医赔债罚没}，优先受偿权插队 \yjt \bs{医债赔罚没}。
          \item \bs{执行回转（只有财产刑有）}。返还原物/价金、赔偿相关的损失。
        \end{enumerate}
  \item 管制、有期缓刑、拘役缓刑。\zd{审判阶段作出、宣告}。\bs{条件、考验}。
        \begin{enumerate}
          \item \bs{3；累、首}。\xydy 3 年有期或拘役。累犯、首要犯罪分子不行。
          \item 生效后 10 日内到\bs{经居地社区矫正机构}报道。
                % \item 考验期。判 3/5 年 \yjt 2/3 年。
          \item 缓刑考验。
                \begin{enumerate}
                  \item 漏罪新罪。\zd{谁判谁撤}。
                  \item 违法违规。谁决定谁来撤、\zd{做出生效}、不能上诉不能抗。
                  \item 成功。原判刑罚\bs{不再执行}。\bdy 刑罚执行完毕。
                \end{enumerate}
          \item 管制。先行羁押 \bs{1:2 折抵}刑期。
        \end{enumerate}
  \item 执行的变更。刑种发生变化。
        \begin{enumerate}
          \item 暂予监外执行。\bs{一放一收}。\zd{叫终止，不叫解除}。
                \begin{enumerate}
                  \item 拘役、有期、特殊的无期。\al{死缓 \cuo}。\bs{疾病、孕乳、不自理}。\bs{省政府指定}的医院。
                  \item 决定。看所处的阶段，谁居主导地位。
                        \begin{enumerate}
                          \item 执行前。法院。
                          \item 执行后。监狱/看守所提书面意见 \yjt \bs{省监狱管理机关/市以上公安}批准。
                        \end{enumerate}
                  \item 收监。\bs{不符条件、违规定、情形消失期未满}。\zd{谁决定出去，谁决定回来}。
                  \item \zd{社区矫正机构}执行。
                  \item 贿赂、非法脱逃出去的期间不计入刑期。
                \end{enumerate}
          \item \bs{减刑}。针对自由刑。\bs{刑期、谁裁、开庭、出庭}。
                \begin{enumerate}
                  \item 无次数限制，但有幅度和限度要求。
                  \item 管制、拘役、有期至少执行原判刑期 1/2、无期至少 13 年。
                  \item \bs{无期死刑在高院，剩下在中院}。合议庭、审期 1 个月。
                  \item 重大\zd{立功}可以减刑。\al{不认真遵守监规、不接受教育改造不可以减刑 \cuo 只看立功}。
                  \item 一定要开庭。
                        \begin{enumerate}
                          \item \bs{重大立功不一般；重大影响有意见；有权有势又有钱}。
                          \item 破格减刑。\bs{职务、黑、金}。职务、黑社会性质、金融类犯罪。
                          \item 应当通知\zd{检察院、罪犯、执行机关}出庭。
                        \end{enumerate}
                \end{enumerate}
          \item \bs{假释}。针对有期、无期。有条件提前释放。\zd{执行阶段作出，用裁定}。\bs{不得、特别}。
                \begin{enumerate}
                  \item 不得假释。
                        \begin{enumerate}
                          \item 累犯。
                          \item 杀、伤、奸、抢；投毒、火、爆组。
                                \yi 单个判 \dydy 10 年。
                                \er \zd{多个同类罪数罪并罚} \dydy 10 年。\al{投毒 8，盗窃 5 \yjt 13 \cuo。投毒 8，强奸 5 \yjt 13 \dui}。
                        \end{enumerate}
                  \item 实质条件。\zd{无人身危害性、再犯可能性}。确有悔改表现。\al{如退赃}。
                  \item 特别假释 \yjt 最高法。\al{原判刑期没执行够就被假释}。
                \end{enumerate}
          \item 减刑、假释考验。
                \begin{enumerate}
                  \item 漏罪新罪 \yjt 谁审谁撤。
                  \item 违法违规 \yjt 原决定法院撤。
                  \item 遵守规定 \yjt 原判刑罚\bs{视为执行完毕}。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
