\section{其他诉讼参与人}
\begin{enumerate}
  \item \zx{法诉辩证翻鉴}。\al{侦查人员、有专门知识的人 \cuo}。
  \item 代理人。
        \begin{enumerate}
          \item 法定代理人。由\zd{法律规定}的对被代理人负有专门的保护义务并代其进行诉讼的人。
                \al{流浪汉甲，A 县民政部门可担任甲的法定代理人并在审理时到场 \dui}。
                \begin{enumerate}
                  \item 无、限制行为能力人。\bs{小孩爹妈，傻子媳妇}。
                  \item \bs{权限 \ydy 本人，地位很独立}。可以越权，个别权利没法替代，如\zd{陈述权}。
                \end{enumerate}
          \item 诉讼代理人。基于被代理人的\zd{委托}而代表被代理人\zd{参加诉讼}的人。
                \begin{enumerate}
                  \item 范围以及人数直接类推辩护人。
                  \item \bs{权限看授权，地位仅代言}。\bs{没有会见、通信以及申请变更强制措施的权利}。
                  \item 除了嫌疑人/被告人（他叫辩护人）均可委托。
                        \begin{enumerate}
                          \item 自诉。\zd{随时}。自诉人及其法定代理人。
                          \item 公诉案件。\zd{移送审查之日起}。+ 近亲属。近亲属 = 上下左右。父母、配偶、子女、\zd{同胞}兄弟姐妹。干爹 \ydy 大龄男朋友不算。
                          \item 附民。随公诉、自诉。
                          \item 没收程序。嫌疑人/被告人的近亲属和其他利害关系人。
                          \item 强制医疗。被申请人 \hz 被告人。
                        \end{enumerate}
                \end{enumerate}
        \end{enumerate}
  \item 辩护人。嫌疑人/被告人及其法定代理人委托 \hz 法律援助进行指派。辩护与代理专题。
  \item 证人。\zd{诉讼外}了解案情的\zd{当事人以外}的\zd{自然人}。特例。\zd{卧底}。
        \begin{enumerate}
          \item \bdy 见证人（事后参与） \bdy 专家证人 \bdy \zd{作证人员}（范围更广。包括证人、鉴定人、被害人等）。
          \item \zd{辨别是非、正确表达即可}。具体分析。
                \al{猥亵案件，甲乙丙（3 年级）因年幼，其陈述 \hz 证言必须有其他证据印证才能采信 \cuo}。
          \item 优先性和不可替代性。\bs{身份冲突，优先适用证人身份。不需要回避}。
          \item 权利。
                \begin{enumerate}
                  \item 使用本民族语言文字进行控诉、查阅证言笔录。
                  \item \zd{经济补助只给证人}。交通、住宿、就餐等\zd{费用谁找的}/哪个阶段，\zd{谁}/哪个机关\zd{补助}。\zd{财政出钱}。
                  \item （公检法）保障人身安全。\bs{作证人员及近亲，危恐黑毒要保护，信息音容接住他}。\al{侦查人员 \cuo}。
                  \item 禁止特定人员接触，人身住宅，其他方面。
                  \item 主动寻求保护，不限罪名。
                \end{enumerate}
          \item 义务。
                \begin{enumerate}
                  \item 应当出庭。\bs{有影响 + 有异议 + 有必要}。例外。\bs{小孩有病，太远在国外}。不出庭 \bdy 不做证。\al{视频}。
                  \item 拒绝后果。\bs{排除拒绝 + 不真实}、\zd{强制出庭、训诫或拘留}。\al{传唤出庭 \cuo}。
                        \begin{enumerate}
                          \item 法警执行，公安协助。\al{强制出庭令交由公安执行 \cuo}。
                          \item \zx{不是只要不出庭就不能用}！！！！+ 真实性无法确认。\al{其证言仍可能作为证据 \dui}。
                          \item \zx{被告人的上下左不能强制，被告人右 + 被害人可以}！\al{ \cuo}。
                          \item 只有情节严重，才可以拘留，\xyh 10 天。{向上一级复议期间拘留处罚暂缓执行 \cuo}。
                        \end{enumerate}
                \end{enumerate}
        \end{enumerate}
  \item 翻译人员。\zd{指派 \hz 聘请}。外国人、无国籍人、少数民族、盲聋哑。进行语言、文字、手势翻译。
  \item 鉴定人。接受公安司法机关的\zd{指派 \hz 聘请}，运用自己\zd{专门知识 \hz 技能}对刑事案件中的专业问题来进行分析、判断，并提出书面意见的\zd{自然人}。
        \zx{个人负责制}。\al{鉴定意见可由单位出具 \cuo}。
        \begin{enumerate}
          \item 应当出庭。法院通知就是\zd{有异议 + 有必要}。\yi 拒绝就排除。\er 正当理由不出庭。\zd{延期审理 \hz 重新鉴定}。
          \item vs 证人。鉴定人是事后参加了解的案情、必须没有利害关系。
          \item vs 有专门知识的人。
                \begin{enumerate}
                  \item 没有资格，\zd{每类最多请 2 人}。
                  \item 权利义务参照鉴定人、不是鉴定意见、但可作为证据使用。\bs{不是法定证据，和民诉不一样}。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
