\section{其他诉讼参与人}
\begin{enumerate}
  \item \bs{法诉辩证翻鉴}。
  \item \kd{代理人}。
        \begin{enumerate}
          \item 法定代理人。由\zd{法律规定}的对被代理人负有专门的保护义务并代其进行诉讼的人。
                \begin{enumerate}
                  \item 无、限制行为能力人。\bs{小孩爹妈，傻子媳妇}。
                  \item \bs{权限 \ydy 本人，地位很独立}。可以越权，个别权利没法替代，如\zd{陈述权}。
                \end{enumerate}
          \item 诉讼代理人。基于被代理人的\zd{委托}而代表被代理人参加诉讼的人。
                \begin{enumerate}
                  \item 除了嫌疑人/被告人（他叫辩护人）均可委托。
                  \item 当事人、法代。\zd{公诉案件 + 一个近亲属}。
                  \item 近亲属。上下左右。父母、配偶、子女、\zd{同胞}兄弟姐妹。干爹 \ydy 大龄男朋友不算。
                  \item \bs{权限看授权，地位仅代言}。
                \end{enumerate}
        \end{enumerate}
  \item 辩护人。嫌疑人/被告人及其法代委托 \hz 法律援助进行指派。辩护与代理专题。
  \item \kd{证人}。\zd{诉讼外}了解案情的\zd{当事人以外}的\zd{自然人}。特例。\zd{卧底}。
        \begin{enumerate}
          \item \bdy 见证人（事后参与）\bdy 专家证人 \bdy \zd{作证人员}（范围更广。包括证人、鉴定人、被害人等）。
          \item \zd{辨别是非、正确表达即可}。具体分析。
          \item 优先性和不可替代性。\bs{身份冲突，优先适用证人身份。不需要回避}。
          \item 权利。
                \begin{enumerate}
                  \item 使用本民族语言文字进行控诉、查阅证言笔录。
                  \item \zd{经济补助只给证人}。交通、住宿、就餐等\zd{费用谁找的}/哪个阶段，\zd{谁}/哪个机关\zd{负责}。\zd{财政出钱}。
                  \item （公检法）保障人身安全。\bs{作证人员及近亲，危恐黑毒要保护，信息音容接住他}。不包括侦查人员。
                  \item 禁止特定人员接触，人身住宅，其他方面。
                  \item 主动寻求保护，不限罪名。
                \end{enumerate}
          \item 义务。
                \begin{enumerate}
                  \item 应当出庭。\bs{有影响 + 有异议 + 有必要}。例外。\bs{小孩有病，太远在国外}。不出庭 \bdy 不做证。\al{视频}。
                  \item 拒绝后果。\zd{强制出庭、训诫或拘留。排除拒绝不真实}。法警执行。\bs{被告人的上下左不能强制}。
                \end{enumerate}
        \end{enumerate}
  \item 翻译人员。\zd{指派 \hz 聘请}。外国人、无国籍人、少数民族、盲聋哑。进行语言、文字、手势翻译。
  \item \kd{鉴定人}。接受公安司法机关的\zd{指派 \hz 聘请}，运用自己\zd{专门知识 \hz 技能}对刑事案件中的专业问题来进行分析、判断，并提出书面意见的\zd{自然人}。
        \begin{enumerate}
          \item 应当出庭。\zd{有异议 + 有必要}。\yi 拒绝就排除。\er 正当理由不出庭。\zd{延期审理 \hz 重新鉴定}。
          \item vs 证人。鉴定人是事后参加了解的案情、必须没有利害关系。
          \item vs \zd{专门知识的人}。没有资格，\zd{每类最多请 2 人}、权利义务参照鉴定人、\zd{报告可以作为证据使用}。
        \end{enumerate}
\end{enumerate}
