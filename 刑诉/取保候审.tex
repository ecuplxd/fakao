\section{取保候审}
\begin{enumerate}
  \item 公检法责令嫌疑人、被告人提供保证人 \hz 交纳保证金，保证不逃避侦查、起诉、审判活动，并随传随到。
  \item 各自的决定只能及于各自的阶段，\zd{公安/国安执行}。
  \item \zd{条件}。不能逮捕可以保。\zd{证据、徒刑、危险三缺一}。\zd{徒刑以下不危险，疾病孕乳超期限}。
        \begin{enumerate}
          \item 积极条件。关键看有没有\zd{危险性}。
          \item 消极条件。原则。\zd{累}犯、\zd{主}犯、自伤自\zd{残}、严重\zd{暴}力、其他\zd{严}重犯罪情形。
        \end{enumerate}
  \item 方式。\zd{2 选 1}，不能同时。
        \begin{enumerate}
          \item 保证人。1 -2 人。\zd{老（75 岁以上）的幼的没钱的}。\zd{无牵连有能力，有权有房又有钱}。
                \begin{enumerate}
                  \item 本案无关、有政治权利、有固定住处、有钱交罚款。
                  \item 义务。\zd{监督、报告、罚款、刑责}（不包括民事赔偿责任）。
                  \item \zd{一头一尾决定机关，其他都是执行机关}。一头/一尾 \yjt 决定/解除。
                \end{enumerate}
          \item 保证金。
                \begin{enumerate}
                  \item 起点 1000 人民币，上不封顶，交纳到指定的银行账户。
                  \item 违反了义务 \yjt 没收。
                  \item 实施新的犯罪，先暂扣。\yi 故意 \yjt 没收。\er 其他 \yjt 退还。
                  \item 金额确定。删除了\scx{当地的经济发展水平}的规定。
                \end{enumerate}
        \end{enumerate}
  \item 启动。依申请 \hz 依职权。
        \begin{enumerate}
          \item 报各自的老大决定。\zd{被取保候审人居住地的派出所}执行。
          \item \zd{每个阶段 12 个月，累加制}。
        \end{enumerate}
  \item \kd{解除}。依申请 \hz 依职权。新修。
        \begin{enumerate}
          \item 必然依职权解除。撤案 \hz 终止侦查。
          \item \zd{自动解除}。措施已变更、不起诉、裁判生效 \hz 已执行。\zd{变更、不诉、无罪责，附加生效主执行}。
        \end{enumerate}
  \item \zd{义务}。
        \begin{enumerate}
          \item 5 个法定义务。
                \begin{enumerate}
                  \item 活动范围\zd{市县}。执行机关批准，决定机关同意。
                  \item 信息变动 24 小时\zd{报告}执行机关。
                  \item \zd{随传}随\zd{到}。
                  \item \zd{不}干\zd{扰证人}作证；\zd{不毁}灭伪造篡改\zd{证}据。\bdy 不能接触。
                \end{enumerate}
          \item 4 个酌定义务。\zd{特定的地方、人、活动、证}。
        \end{enumerate}
  \item 违反后果。更严厉的措施，往上走。
        \begin{enumerate}
          \item 交钱了 \yjt 没收部分或全部。
          \item 区别情形。
                \begin{enumerate}
                  \item 具结悔过 \yjt 重新交纳保证金、提出保证人，监视居住，予以逮捕。
                  \item 能逮捕 \yjt 先行拘留。
                  \item \zd{新罪}；企图\zd{逃}跑、自\zd{杀}；\zd{串}供；\zd{毁}灭、伪造\zd{证}据，\zd{还要打击报复他} \yjt 应当逮捕。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
