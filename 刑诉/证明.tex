\section{证明}
\begin{enumerate}
  \item 证明对象/证明客体。待证案件事实。
        \begin{enumerate}
          \item = 实体法事实（\al{量刑}）+ 程序法事实（自愿性、真实性、合法性）。
          \item 免证事实。\zd{常识法规和定律；推定生效无异议}。\al{案件基本事实免证 \cuo 刑诉没有自认}。
          \item vs 证据事实。证据本身的来源、构成等要素。\al{用于鉴定的体液检材是否被污染的事实 \yjt 鉴材质证}。
        \end{enumerate}
  \item 证明责任。\zd{提供证据 + 说服责任}。兼有 \hz 居其一。\al{财产来源不明}。
        \begin{enumerate}
          \item 公诉案件。\bs{原则公诉人承担证明责任，持有型犯罪被告人、辩护人需要承担一定的证明责任}。
                \al{非法持有枪支案中，被告人负有提出证据的责任 \dui}。
          \item 自诉案件。主要是谁主张、谁举证，但不是法则（必须）。
          \item 被告的无罪辩护是消极主张，不承担证明责任。
          \item \zd{区分辩解}。
        \end{enumerate}
  \item 证明标准。
        \begin{enumerate}
          \item 立案阶段 \yjt 有犯罪事实存在。\zd{发生了}。
          \item 逮捕阶段 \yjt \zd{他干了；已有查证属实了}。
          \item 侦查终结、提起公诉、有罪判决。\bs{事实清楚，证据确实、充分}。55 条第 2 款。
                \begin{enumerate}
                  \item 定罪量刑的\zd{事实都有证据证明} \yjt 清楚。
                  \item 据以\zd{定案的证据均}经过法定程序\zd{查证属实} \yjt 确实。
                  \item 综合\zd{全案证据}，对所认定事实\zd{均已排除合理怀疑} \yjt 充分。
                  \item \zd{定罪}证据存疑 \yjt 疑罪\zd{从无}。\al{是否强奸}。
                  \item \zd{量刑}证据存疑 \yjt 疑案\zd{从轻}。\al{强奸 1 次 还是 2 次}。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
