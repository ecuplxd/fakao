\section{证明}
\begin{enumerate}
  \item 证明对象/证明客体。一切法律要件事实。
        \begin{enumerate}
          \item = 实体法事实（\al{定罪、量刑}）+ 程序法事实（\al{认罪认罚自愿性、真实性、合法性}）。
          \item \zx{免证事实。同民诉，但刑诉没有自认}。\al{大家都认可的程序性事项 \dui}。
          \item vs 证据事实。证据本身的来源、构成等要素，\zd{不属于证明对象}。\al{用于鉴定的检材是否被污染的事实 \cuo}。
        \end{enumerate}
  \item 证明责任。证明主体就其诉讼主张承担的提供证据予以证明的责任。\zd{提供证据 + 说服责任}。兼有 \hz 居其一。
        \begin{enumerate}
          \item 证明主体 = 公诉人、被告人、自诉人、附民原告。\al{证明主体都是刑事诉讼主体 \dui}。
          \item 公诉案件。\zd{原则公诉人承担证明责任，持有型犯罪被告人、辩护人需要承担一定的证明责任}。
                \begin{enumerate}
                  \item \al{甲应对自己符合减刑条件承担证明责任 \cuo}。
                  \item \al{非法持有枪支、巨额财产来源不明。被告人承担说服责任 \cuo。被告人负有提出证据的责任 \dui}。
                  \item \al{检应就甲的失职行为与水源污染之间的因果关系加以证明 \dui{不要和民法混}}。
                  \item \al{没收程序中，甲妻称有部分财产系合法财产不应没收，甲妻无需承担提供证据的责任 \dui}。
                \end{enumerate}
          \item 自诉案件。\al{依据谁主张，谁举证的\xhx{法则}确定 \cuo}。
          \item 被告的无罪辩护是消极主张，不承担证明责任。
          \item \zd{区分辩解}。
        \end{enumerate}
  \item 证明标准。
        \begin{enumerate}
          \item 立案阶段 \yjt 有犯罪事实存在。\zd{发生了}。
          \item 逮捕阶段 \yjt \zd{他干了；已有查证属实了}。
          \item 侦查终结、提起公诉、有罪判决。\zd{事实清楚，证据确实、充分}。55 条第 2 款。
                \begin{enumerate}
                  \item 定罪量刑的\zd{事实都有证据证明} \yjt 清楚。
                  \item 据以\zd{定案的证据均}经过法定程序\zd{查证属实} \yjt 确实。
                  \item 综合\zd{全案证据}，对所认定事实\zd{均已排除合理怀疑} \yjt 充分。
                  \item \zd{定罪}证据存疑 \yjt 疑罪\zd{从无}。\al{是否强奸}。
                  \item \zd{量刑}证据存疑 \yjt 疑案\zd{从轻}。\al{强奸 1 次 还是 2 次}。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
