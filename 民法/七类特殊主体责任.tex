\section{7 类特殊主体责任}
\begin{enumerate}
  \item 内外有别。\zd{抓内部外部关系}。\zd{人多的地方都是过错}。
  \item 监护人。
        \begin{enumerate}
          \item 无过错。尽到监护职责，\zd{可以减轻}。\al{无需承担侵权责任 ×}。
          \item \zd{被监护人有独立财产 \yjt 监护人补充}。
        \end{enumerate}
  \item 用工。\zd{危险控制理论}。\zd{对内要证明有故意 \hz 重大过失才能追偿}。
        \begin{enumerate}
          \item 员工。对外没责任。\zd{结论。只要看到员工、公司 + 员工出现在选项一律错}。
          \item 有用工无合同，有合同无用工。谁用工谁无过错，无用工方过错按份/相应的责任。\al{劳务派遣}。
                \begin{enumerate}
                  \item \al{甲用工，乙派遣丙。乙承担赔偿责任，甲承担相应的责任 ×；甲乙连带 ×；丙承担责任 ×}。
                  \item \al{甲承担赔偿责任，乙承担相应的责任 √}。\zd{别混了}。
                \end{enumerate}
          \item 执行职务行为。\zd{外观主义}认定。
                \begin{enumerate}
                  \item \al{快递小车}。
                  \item \al{公司\scx{明文禁止代驾，乙为获得高额报酬而代驾}，乙应承担赔偿责任 ×}。
                  % \item 命题陷阱。\al{甲执行职务过程中，与乙发生口角，将乙打伤}。
                \end{enumerate}
          \item \sjt 被侵权，双重赔偿。
          \item 接受劳务方。无过错。\al{甲雇佣乙料理家务}。
        \end{enumerate}
  \item 帮工（无偿的）\ydy 用工。\al{甲家盖房，邻居乙丙前来帮忙。丙因失误从高处摔下受伤，乙\xhx{不小心}撞伤小孩丁}。
        \begin{enumerate}
          \item 侵权他人。免责\zd{明确拒绝帮工}。
          \item 被他人侵权。第三人承担赔偿责任 \hz 被帮工人\zd{适当补偿}。
          \item 帮工人过错。\al{对丙的损害，甲不承担赔偿责任 √ \yjt 与有过失}。
        \end{enumerate}
  \item 定作人。过错（对承揽人支配力非常弱）。免责看选任、指示有无过错。
  \item 网络服务提供者。\zd{避风港规则}。\zd{立即}删除、屏蔽、断开链接（只能请求一次），否则对\zd{扩大部分连带}。
  \item 安保义务人。过错补充。\zd{不以实际交易为前提}。\al{公共场所、商场、小区物业}。
  \item 教育机构。
        \begin{enumerate}
          \item 学校不是监护人。教育、管理、保护。\al{小甲父母的监护职责已移转到学校 ×}。
          \item 过错补充。
          \item 学校组织的校外活动视为在校期间。
        \end{enumerate}
\end{enumerate}
