\section{继承关系}
\begin{enumerate}
  \item 1 道多选。核心考点，\zd{继承方式}：法定、遗嘱。
  \item \zd{如何推定死亡先后顺序}。最大限度防止个人财产归国家。题目没说，推定事故以外有继承人。
        \begin{enumerate}
          \item 无继承人，不管顺序，归国家所有。
          \item 有继承人。按常理，\zd{长辈先死，晚辈后死；同辈同死，互不继承}。
          \item 有的有，有的没。没的先死。没考过。
        \end{enumerate}
  \item 遗产。财产综合体。
        \begin{enumerate}
          \item 个人\zd{合法}财产。
          \item \zd{死亡时}（常考）。\zd{死亡赔偿金参照遗产处理}。
          \item 7 项人格利益（8 项近亲属）获得的赔偿金不是。
          \item 配偶的共同财产先剥离（\zd{析产}）。
        \end{enumerate}
  \item 继承权的放弃。
        \begin{enumerate}
          \item 特定时间段内放弃，死亡前放弃行为无效，\zd{死亡后遗产分割前}。
          \item 不得以放弃权利不履行义务（不要遗产不抚养；物业费）。
          \item 向利害关系人，其他继承人说；书面。
          \item 丧失。为了争遗产不择手段；故意杀害被继承人、其他继承人 \yjt \zd{绝对丧失}，考试常考。
        \end{enumerate}
  \item 继承方式。
        \begin{enumerate}
          \item 遗嘱。
                \begin{enumerate}
                  \item \zd{公自录口代打}。2 个以上合格见证人。越靠后的方式越容易无效。
                        \begin{enumerate}
                          \item 公证。
                          \item 自书遗嘱。写完成立，写签注（年月日）。
                          \item 录。记录\zd{姓名、肖像、年月日}。没考过。
                          \item 口。危急情况下（遗言），消除后无效。
                          \item 代。代写字；签名；年月日。
                          \item 打。打印 + \zd{每一页签名}。
                        \end{enumerate}
                  \item 只要没生效就可以撤回、变更；实行与遗嘱相反行为视为撤回。
                  \item \zd{数份遗嘱}不冲突 \yjt 都有效，各自处理；冲突 \yjt 从后往前处理。
                  \item 遗嘱继承人先死的，\zd{等同于没写遗嘱，不生效力，按法定继承处理}，绕回来考代位继承，转继承。
                  \item 部分无效 \yjt 死亡时没给未成年留；处分配偶共同财产；附义务的（\zd{心愿}）继承人必须履行。
                \end{enumerate}
          \item 法定继承。按人头点。
                \begin{enumerate}
                  \item 第一顺位。第二顺位考不到。
                        \begin{enumerate}
                          \item 法定继承人 \yjt 父母、配偶、子女（婚生、非婚生、养子女、继子女、胎儿）。
                          \item 拟制继承人 \yjt 丧偶儿媳，女婿 + 赡养义务，不论再婚；孙子女、外孙子女的代位继承。
                        \end{enumerate}
                \end{enumerate}
          \item 代位继承。\al{甲死后，其儿子乙也死了，留下一子丙}。
                \begin{enumerate}
                  \item \zd{有抚养关系的继子女无\blh}（无血缘关系，无感情关系）。\al{家有儿女，刘星姥姥，小雨小雪}。
                  \item \zd{新型代位继承类型}，侄子侄女、外甥外甥女。\al{甲乙兄弟，父母双亡，甲无妻无子，乙有一子}。
                \end{enumerate}
          \item 转继承。继承开始后遗产分割前，继承人死了。\input{draw/继承关系.tex}
          \item 遗嘱执行人。指定 \yjt 推选 \yjt 指定。
          \item 遗赠。同遗嘱，对象不同而已；60 日；沉默视为放弃；要式行为。
          \item 遗赠抚养协议。
                \begin{enumerate}
                  \item 解决老人养老（另一个制度：成年人意定监护）。
                  \item 内容：生养死葬；扶养人不一定是自然人，组织也行（养老院）；\zd{不能存在法定抚养义务}。
                  \item 遗赠抚养协议 > 遗嘱 \hz 遗赠 > 法定继承。
                  \item 不用还钱。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
