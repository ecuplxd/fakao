\section{继承关系}
\begin{enumerate}
  \item 核心考点。\zd{继承方式}。法定、遗嘱。1 道多选。
  \item \zd{如何推定死亡先后顺序}。最大限度防止个人财产归国家。题目没说，推定事故以外有继承人。
        \begin{enumerate}
          \item 无继承人。不管顺序，归国家所有。
          \item 有继承人。按常理，\zd{长辈先死，晚辈后死；同辈同死，互不继承}。
          \item 有的有继承，有的没继承人。\zd{无继承者先死}。
        \end{enumerate}
  \item 遗产。财产综合体。常考。
        \begin{enumerate}
          \item \yi \zd{死亡时}遗留的\er\zd{个人}\san\zd{合法}财产。\zd{死亡赔偿金不属于遗产 √ \yjt 死后得到，参照遗产处理}。
          \item 7 项人格利益，8 类近亲属获得的赔偿金不是（死后得到）。
          \item 配偶的共同财产先剥离。\zd{析产}。
          \item 遗产未分割之前，遗产由继承人\zd{共同共有}。
        \end{enumerate}
  \item 分配。
        \begin{enumerate}
          \item 生活特殊困难 + 缺乏劳动能力的继承人，\zd{应当予以照顾}。
          \item 尽了主要扶养义务 \hz 与被继承人共同生活的继承人\zd{应当多分}。
          \item 不尽扶养义务的人，\zd{应当不分或少分}遗产。
          \item 协商同意的可以不均等分割。
        \end{enumerate}
  \item 继承权的放弃。
        \begin{enumerate}
          \item 特定时间段内放弃。\zd{死亡后遗产分割前}。死亡前放弃行为无效。
          \item \zd{不得以放弃权利不履行义务}。不要遗产所以不履行抚养义务。物业费也是。
          \item 向利害关系人、其他继承人\zd{说 \hz 书面}。
          \item 丧失。\zd{绝对丧失}。为了争遗产\zd{不择手段}。考试常考。\al{出言不敬，丧失继承权 ×}。
                \begin{enumerate}
                  \item 故意\zd{杀害}被继承人。
                  \item 为争夺遗产而杀害其他继承人。
                  \item \zd{遗弃、虐待}被继承人情节严重。\sjt 对被继承人。
                  \item 伪造、篡改、隐匿、销毁遗嘱，情节严重。\xjt 对遗嘱。
                  \item 欺诈、胁迫手段迫使/妨碍被继承人设立、变更、撤回遗嘱，\zd{情节严重}。
                \end{enumerate}
        \end{enumerate}
  \item 继承方式。
        \begin{enumerate}
          \item 遗嘱。单方民事法律行为。\al{是无相对人的意思表示 √}。
                \begin{enumerate}
                  \item 方式。\zd{公自录口代打}。\zd{2 个以上合格见证人}。越靠后的方式越容易无效。
                        \begin{enumerate}
                          \item 公证。
                          \item 自书遗嘱。写完成立。写签注（年月日）。
                          \item 录。记录\zd{姓名、肖像、年月日}。没考过。
                          \item 口。危急情况下（遗言），消除后无效。
                          \item 代。代写字、签名、年月日。
                          \item 打。打印 + \zd{每一页签名}。
                        \end{enumerate}
                  \item 只要没生效就可以撤回、变更。实行\zd{与遗嘱相反行为}视为撤回。\al{当着众亲友的面烧毁遗嘱}。
                  \item \zd{数份遗嘱}。\yi 不冲突 \yjt 都有效，各自处理。\er 冲突 \yjt 从后往前处理。
                  \item \zd{遗嘱继承人先死的，等同于没写遗嘱，不发生效力，按法定继承处理}。绕回来考代位继承、转继承。
                  \item 部分无效 \yjt 死亡时没给未成年留、处分配偶共同财产、附义务的（\zd{遗愿}）继承人必须履行。
                \end{enumerate}
          \item 法定继承。按人头分。
                \begin{enumerate}
                  \item 第一顺位。
                        \begin{enumerate}
                          \item 法定继承人 \yjt 父母、配偶、子女（婚生、非婚生、养子女（\zd{登记}）、\zd{有抚养关系的继子女}、胎儿）。
                          \item 拟制继承人 \yjt 丧偶儿媳/女婿 + 赡养义务（不论再婚与否）、孙子女、外孙子女的代位继承。
                        \end{enumerate}
                  \item 第二顺位。一般不到考。
                \end{enumerate}
          \item 代位继承/间接继承。继承人先于被继承人死亡。
                \begin{enumerate}
                  \item \zd{有抚养关系的继子女无代位继承}（无血缘、感情关系）。\al{家有儿女，刘星姥姥，小雨小雪}。
                  \item \zd{新型代位继承}。侄子侄女、外甥外甥女。\al{甲乙兄弟，父母双亡，甲无妻无子，乙有一子}。
                  \item 丧失继承权的代位继承人缺乏劳动能力 + 没有生活来源 \hz 尽赡养义务较多的，\zd{可以适当分给遗产}。
                \end{enumerate}
          \item 转继承。继承开始后遗产分割前，继承人死了。\input{draw/继承关系.tex}
          \item 遗嘱执行人。
                \begin{enumerate}
                  \item 指定 \yjt 推选 \yjt 法院指定。
                  \item 仅可按照遗嘱的规定将遗产在继承人之间进行分配。
                \end{enumerate}
          \item 遗赠。
                \begin{enumerate}
                  \item 同遗嘱，对象不同而已。\zd{60 日、沉默视为放弃、要式}。\al{老王日记里写道死后藏书均送给老李}。
                  \item \zd{\bdy 赠与}。\al{甲临终前将书稿赠与乙并立即交付}。
                \end{enumerate}
          \item 遗赠抚养协议。
                \begin{enumerate}
                  \item 解决老人养老。另一个类似制度。成年人意定监护。
                  \item 内容。生养死葬。
                  \item 自然人、组织均可（养老院）\yjt 扶养人\zd{不存在法定抚养义务}即可。
                  \item \zd{遗赠抚养协议 > 遗嘱、遗赠 > 法定继承}。满足死者遗愿。
                  \item 不用帮还债。
                \end{enumerate}
        \end{enumerate}
\end{enumerate}
