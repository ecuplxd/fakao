\section{租赁合同}
\begin{enumerate}
  \item \zx{负担行为}。\bs{移转标的物使用权}的合同。\bs{没有所有权、处分权，不影响租赁合同效力}。\al{出租盗赃物。租赁合同合法有效}！！
  \item 超过 20 年（首租、续租）无效、备案登记不影响效力、\zd{租期不满 6 月可口头}。
  \item \zx{维修义务}。\zd{有约从约 \yjt 出租人负有法定的维修义务} \yjt 承租人过错导致的承租人自己修。
        \begin{enumerate}
          \item \zd{承租人可自行维修，维修费用由出租人负担}。
          \item \zd{属于无因管理行为}。
          \item \zd{影响了承租人使用，应当相应减少租金 \hz 延长租期}。
        \end{enumerate}
  \item 承租人权利。
        \begin{enumerate}
          \item 法定解除权。根本违约。
          \item \zx{买卖不破租（登记）赁}。前提已合法占有租赁物。打时间差。
                \al{甲把画租给乙，约定 10 日后交付，第二天甲占有改定方式把画卖给了丙 \yjt 乙未占有 \yjt 不适用买卖不破租赁}。
                \bs{租赁期间内，发生所有权变动的，不影响租赁合同的效力。本案，xx 租赁权成立在先，买卖不破租赁}。
          \item \zx{优先购买权}。
          \item 续租权。同等条件优先承租。
        \end{enumerate}
  \item \zx{承租人义务}。
        \begin{enumerate}
          \item \zd{禁止擅自转租}。\zx{构成根本违约 \yjt 出租人解除权}。主观起算 6 个月除斥期间。
          \item 未经同意不得添附。
          \item 转租 \yjt 大房东要和二房东解约 \yjt \zx{转租合法有效；第三人代为履行抗辩权}。见合同的履行部分。
        \end{enumerate}
  \item 租赁合同无效（含续租）。\yi 违建。\er 不能收取租金，但有占有使用费（不当得利）。
  \item \zx{装饰装修扩建}。\al{装空调}。
        \begin{enumerate}
          \item \zd{征得同意} \yjt 都有过错按份。
          \item \zd{恢复原状、赔偿损失}。
          \item \zx{费用有约从约，无约承租人}。
        \end{enumerate}
  \item \zx{不定期租赁合同}。
        \begin{enumerate}
          \item \dydy 6 月以上。
          \item 没有约定 \hz \zd{约定不明}视为不定期。承租人任意解除权。
                \begin{enumerate}
                  \item \al{甲租乙房，未约定租期，仅约定甲父失踪回来后合同终止}。
                  \item \al{口头约定 5 年租期 \yjt 虽然口头，但\xhx{约定明确}}。
                \end{enumerate}
        \end{enumerate}
  \item 临时建筑出租。
        \begin{enumerate}
          \item 一审辩论终结前取得批准的，租赁合同有效。
          \item 租赁期超使用年限部分无效，一审辩论终结前取得延长资格的，有效。\al{无权将该房屋继续出租 \dui}。
        \end{enumerate}
  \item 一房数租。
        \begin{enumerate}
          \item 出租人瑕疵担保责任。
          \item 优先权。合法占有 \dy 登记 \dy 合同成立顺序。\al{甲租给乙并已交付 \yjt 甲租给丙 \yjt 丙转租给丁}。
        \end{enumerate}
\end{enumerate}
