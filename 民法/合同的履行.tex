\section{合同的履行}
\begin{enumerate}
  \item 电子合同（\al{网购}）。规则无区别。
        \begin{enumerate}
          \item 要约。发布的商品信息。
          \item 承诺。选择商品，提交订单成功。
          \item 成立。提交订单成功时。
          \item 时间。签收时间 \dy 凭证载明的时间 \dy 实际提供服务的时间 \dy 数据传输到达的时间。
        \end{enumerate}
  \item 涉他合同 vs 束己合同（\al{租赁}）。是否只能约束合同当事人。利他合同。
        \begin{enumerate}
          \item 为第三人创设权利/向第三人履行。\zd{第三人是否享有独立的请求权}。
                \begin{enumerate}
                  \item 不真正利他。\zd{不是合同当事人，纯粹的受益}。\al{甲和乙花店协议，乙需每天给丙送花；给你点外卖}。
                  \item 真正利他。\al{传统型的融资租赁合同}。
                        \begin{enumerate}
                          \item 双方约定将第三人纳入合同中。\zd{内部通知 + 外部不反对}。
                          \item 第三人是合同当事人，\zd{享}有独立的请求\zd{权}，\zd{却不负义务}。\al{约定丙也可以直接向乙请求履行 \dui}。
                          \item 抗辩权移转。\al{丙要求乙送花，乙说甲没付钱 \yjt 乙对丙行使先/顺序履行抗辩权}。
                        \end{enumerate}
                \end{enumerate}
          \item 为第三人创设义务/由第三人履行。\zd{连环买卖}。\al{甲卖乙，乙转售给丙，甲应乙的要求直接交付给丙}。
        \end{enumerate}

        \begin{kk}{第三人代为履行/涤除权}
          明知的非债清偿。\zd{第三人履行该债务具有合法利益}。新增。
          \begin{enumerate}
            \item 基于合同目的。\al{甲 1W 卖表 \yjt 乙 2W 卖 \yjt 丙，丙为拿到表付 1W 给甲，付 1W 给乙}。
            \item 基于特殊感情帮还钱。\zd{货币之债没有人身专属性，谁来履行都行}。无协议。
            \item 实现对债务人的赠与。\al{清空购物车、本公司代替甲公司承担 4000 万元的返还义务}。
            \item 第三人未通知债务人。
                  \begin{enumerate}
                    \item 对债务人构成无因管理。在原债务范围内，基于无因管理进行追偿。
                    \item \zd{履行瑕疵。债务人不承担违约责任}。\al{甲可就该违约赔偿金向乙公司追偿 \cuo}。
                  \end{enumerate}
          \end{enumerate}

        \end{kk}

  \item \zd{双务合同}履行中的抗辩权。谁先履行谁吃亏。\zd{买卖合同比较典型，但不限于买卖合同}。
        \begin{enumerate}
          \item 同时履行。同时履行抗辩权。\al{一手交钱一手交货}。
          \item 顺序履行。\al{甲乙约定甲先付钱，1 月后乙交付图书，甲只付了一半价款。1 月后甲要求乙交付图书}。
                \begin{enumerate}
                  \item 先履行一方。\zd{不安抗辩权 \yjt 可以中止履行}。考点。
                        \begin{enumerate}
                          \item 有不安理由的\zd{确切证据}。\al{可能性 \cuo。乙卖画给甲，约定甲先付款，后甲发现乙只是代丙保管 \dui}。
                          \item 情形法定。\zd{丧失商誉、经营状况严重恶化、逃债、其他}。\al{乙的电梯因事故质监局介入调查 \dui}。
                          \item 通知对方。
                        \end{enumerate}
                  \item 后履行一方。先履行一方\zd{未履行、瑕疵履行} \yjt 先/顺序履行抗辩权。不是说履行完了就不能主张了。
                \end{enumerate}
          \item 相应的抗辩。\zd{主对主，从对从，全部对全部，部分对部分}。
        \end{enumerate}
  \item \zd{履行治愈}。采用合同书形式订立合同时，在签章之前，\zd{一方已履行主要义务，对方接受时}，合同成立。\al{借钱}。
\end{enumerate}
