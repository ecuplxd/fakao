\section{货物销售合同公约}
\begin{enumerate}
  \item 适用范围。
        \begin{enumerate}
          \item 原则买卖双方的\zd{营业地}要在不同的国家。
                \begin{kk}{客体不适用}
                  \begin{enumerate}
                    \item 消费者合同。购买目的供\zd{自用、家人使用、私人使用}。
                    \item 特殊的交易方式（\al{拍卖}）、依法律令状进行的销售。
                    \item \zd{电力、飞机、船舶}。标的物特殊。
                    \item 有价证券 \hz 货币。\al{企业债券、货币的买卖}。
                  \end{enumerate}
                \end{kk}

                \begin{kk}{内容不涉及}
                  \begin{enumerate}
                    \item 不调整\zd{合同的效力}，但调整了合同的\zd{成立}。
                    \item 不调整\zd{所有权转移}（物权），但规定了\zd{所有权担保}（债权）。
                    \item 不调整\zd{产品责任}问题（侵权），但要求卖方承担对买方的\zd{质量担保责任}（合同）。
                  \end{enumerate}
                \end{kk}

                \begin{kk}{效力任意性}
                  当事人意思自治优先。合同 \dy 公约规定，但\zd{约定内容不一样，对公约的排除效力不一样}。
                  \begin{enumerate}
                    \item \zd{准据法的选择 \dy 公约适用}。在我国，合同中若明确选择\zd{准据法}，将\zd{完全排除}公约适用。
                    \item 可以通过合同条款约定\zd{部分排除 \hz 改变}公约规定。\al{选择贸易术语 \dui}。
                  \end{enumerate}
                \end{kk}
        \end{enumerate}
  \item 卖方义务的特殊要求。
        \begin{enumerate}
          \item 如果有义务归还价款，必须同时从支付价款之日起支付价款\zd{利息}。
          \item \zd{交货}义务。
                \begin{enumerate}
                  \item 持续交付。只要\zd{某一次}交付构成\zd{根本违约，则整体无效/违约}。
                  \item 一切\zd{以合同记载的规格为准}。样品不构成产品的质量标准。
                \end{enumerate}
          \item \zd{交单}义务。
                \begin{enumerate}
                  \item 若\zd{提前交单}享有在交单时间届满前\zd{纠正单据错误}的权利，但应\zd{赔偿买方损失}。
                  \item 其他权利无中生有。\al{可以改变移交单据的地点和方式 \cuo}。
                \end{enumerate}
          \item \zd{知识产权担保}义务。第三方不能根据知识产权主张任何权利或要求（\zd{不侵犯他人的知识产权）}。
                \begin{enumerate}
                  \item 地点。\zd{买方营业地、合同预期的货物转售 \hz 使用地}。
                  \item 免责。
                        \begin{enumerate}
                          \item 买卖双方知假买假，知假卖假。
                          \item \zd{按买方}指示供货。\xjt \zd{卖方无过错}。
                          \item 买方知情，但未在合理期限内通知卖方。
                        \end{enumerate}
                \end{enumerate}
        \end{enumerate}
  \item 买方义务的特殊要求。
        \begin{enumerate}
          \item \zd{接收}义务。
                \begin{enumerate}
                  \item 采取一切理应采取的辅助性工作，以期卖方能交付货物。
                  \item \zd{好不好都得收}，不收导致的\zd{损失买方承担}。
                \end{enumerate}
          \item \zd{拒收}权。卖方\zd{提前交货、多交}（只能拒收多交部分）。
          \item \zd{检验}货物。承担 \zd{2 个时间 2 个地点}的限制。
                \begin{enumerate}
                  \item 实际可行的\zd{最短时间}内验货。有问题，\zd{声明不符}的时间为\zd{收货后 2 年}内。
                  \item 运输目的地就是验货点，\zd{卖方已知 + 同意转运}可以延伸到转运目的地。
                \end{enumerate}
        \end{enumerate}
  \item 买卖中间的风险转移。\yi 有约从约 \yjt \zd{交货时}。\er 若买卖的是在途货物，买卖\zd{合同成立时}风险转移。
\end{enumerate}
