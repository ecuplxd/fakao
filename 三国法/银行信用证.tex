  \item 开证行。\zd{单证一致、单单一致。表面一致}。
        \begin{enumerate}
          \item 有义务收卖方单给卖方钱。要求开证申请人（买方）付款赎单。
          \item 告诉买方、银行、凭提单去取货、履行押金项下的权利。
          \item 高效率付款。审单期仅限 \zd{5 个工作日}。
          \item 2 个可以。单证不符，\zd{可以自行联系}开证申请人，开证申请人放弃\zd{不符点通知}，银行\zd{可以释放单据}。

          \item 免责。\zd{信用证独立原则}。效力独立于买卖合同。
                \begin{enumerate}
                  \item \zd{对单据的实质免责}。最重要的免责条款。
                  \item 对传递延误 \hz 遗失免责。
                  \item 不可抗力免责。
                  \item \zd{对买卖合同的实际履行状况免责}。
                \end{enumerate}

  \item 保兑信用证。一家银行开出的信用证，被另一家银行加以保证兑付。
        \begin{enumerate}
          \item \zd{连带责任}。
          \item 本质就是一个\zd{加持}的信用证。
        \end{enumerate}
  \item 信用证欺诈。
        \begin{enumerate}
          \item \zd{伪造}单据 \hz 以假货充真货。卖方诈买方钱。国际经济法调整。
          \item \zd{以保函换取与信用证相符的提单}。\al{倒签提单、预借提单、以保函换取清洁}。
          \item 开立\zd{假信用证}。银行业务员。\zd{没有真实交易基础}。信用证诈骗罪。
          \item 开立\zd{“软条款”}信用证。买方诈卖方货。卖方可要求买方申请修改信用证，否则可以拒绝接受此信用证。
                \begin{enumerate}
                  \item 暂不生效条款。
                  \item 限制性付款条款。\zd{其实就是附条件}？\al{清关后支付、货物检验合格后支付}。
                  \item 加列难以实现的限制性条件。\al{禁止转船、申请人派员检验、啥都要经开征人同意、另行通知}。
                \end{enumerate}
        \end{enumerate}

        \begin{kk}{信用证欺诈例外原则。止付令。}
          非常苛刻的条件限制。
          \begin{enumerate}
            \item 只能由\zd{有管辖权的法院}发出。司法干预。
            \item 有欺诈的\zd{确凿证据}，有可靠充分的\zd{担保}。
            \item 时间限制。要早于\zd{任何}一家关联银行已经\zd{善意}付款 \hz 承兑信用证项下的票据。
            \item \zd{48 小时}做出裁定。
          \end{enumerate}
        \end{kk}
  \item \zd{单证 \hz 单单表面不符和止付令不用付钱，其他都要！可以解决信用证一多半的题目}。
\end{enumerate}
