\section{有限责任公司}
\begin{enumerate}
  \item 人合性。排斥外人，维护原有股权结构。\zx{50 个以下}股东出资设立。
  \item \bs{股权转让}。章程 \dy 有约从约 \dy 法律。\bs{要求善意取得、善意取得、善意取得，禁止恶意}。
        \begin{enumerate}
          \item 对内。完全自由。\al{甲可将其股权分割为两部分，分别转让给乙、丙}。
          \item 对外。\zx{过半数同意}。\al{被分手}。
                \begin{enumerate}
                  \item 书面通知。30 日决定 \yjt 不同意又不买 \yjt 视为同意。
                  \item \zx{同等条件下}（数量、价格、期限、支付方式）优先购买权。
                  \item \zx{多个 \yjt 协商 \yjt 认缴比例}。
                  \item 期限。章程 \yjt 通知 \yjt 30 日。
                  \item 转让方\zd{可以反悔}，承担\zd{缔约过失责任}。虚假谈判，恶意磋商。
                  \item \zx{转让自由}。\bs{仅仅起诉确认无效的，不予支持}。不得既不行使优先购买权，又认定转让无效。
                  \item \zd{恶意转让}诉讼时效。主观知道 30 日 \hz 登记 1 年内。
                \end{enumerate}
          \item 常考情形。
                \begin{enumerate}
                  \item 离婚股权分割 = 对外转让规则。
                  \item 继承。当然继承 \yjt 章程\zx{可以规定不允许继承}。
                  \item 强制执行。\zx{法院[通知 + 告知] \yjt 20 日内行使}优先购买权。
                  \item \zx{瑕疵股权转让}。\zx{出资期限届满后}，不因转让而无责 \yjt \zx{瑕疵担保责任，受让人故意则连带}。
                  \item \zx{一股二卖}。\bs{登记对抗，适用善意取得}。民商结合。
                  \item \zx{股权让与担保}。真担保假让与，\bs{不发生股权转让效果}，\zx{股权价值}优先受偿。\bs{注意区分股权、股权质权}。
                \end{enumerate}
          \item 异议股东/小股东的\bs{回购请求权}。\zx{异议 = 投反对票}。股权退给公司，公司把钱退给股东。\zd{协商前置}。
                \begin{enumerate}
                  \item \zx{5 年营利，5 年不分}。\bs{题目打时间差}。\al{协议\xhx{约定}头 2 年不分，3 年后甲起诉}。
                  \item \zx{合并、分立；转让主要财产}。\al{决议将公司的\xhx{分店转让} \dui}。
                  \item 公司死翘翘，要\zx{改章程使公司继续存续}。不要和公司僵局混。
                  \item \bs{就这 3 种情况}。
                \end{enumerate}
        \end{enumerate}
  \item 组织机构。\zx{人事任免找上级，不能跨部门}。\yi 决议一经做出，\zx{立即生效}。\er 各自只对自己\zx{直接上级}负责。
        \begin{enumerate}
          \item 股东会。\bs{宏观}机构。经营方针、投资计划。选董事、监事；关键词\zd{审议、批准}。2 - 50 人。
                \begin{enumerate}
                  \item 定期会议、\zx{临时会议（1/10 表决权、1/3 董事、监事、持股合计达 40\%）}。
                        \al{甲 25\%，乙 8\%，丙7\%。甲单独，甲乙丙共同召集和主持}。
                  \item \zd{召集、主持。由内而外，依次补位}。
                  \item 15 日通知。\zd{全体书面同意}可以不开。\zd{不强行要求全体股东参加}。
                  \item 表决规则。\zx{资本多数决}。
                        \begin{enumerate}
                          \item 特别事项。\zx{2/3，34\% 一票否决权}。\bs{改形式、改章程；增资、减资；合并、分立}。
                          \item 一般事项。公司法没规定，考试默认\zx{过半数}。
                        \end{enumerate}
                  \item \bs{决议效力}。实践中中小公司不重视，解决股东矛盾的切入点。合规、合同管理、内控、会议管理。
                        \begin{enumerate}
                          \item \zx{不成立}。程序存在根本缺陷。\zx{没开会}（\al{伪造决议}）、\zx{没表决、人不够、票不够}。
                          \item \zx{无效}。内容违法、违规。\bs{做题不要轻易选无效}。
                          \item \zx{可撤销}。违反章程；程序（召集、表决方式）\zx{瑕疵}（要有\bs{实质影响}） \yjt \zx{60 日}向法院申请撤销。
                        \end{enumerate}
                  \item \zx{决议效力之诉}。\bs{公司为被告，决议涉及的其他利害关系人为第三人}。内外有别。
                        \begin{enumerate}
                          \item 无效、不成立之诉。股东、董事、监事。
                          \item 撤销之诉。\zx{仅限股东}提起；\zx{可以应公司申请，要求原告提供担保}。
                        \end{enumerate}
                \end{enumerate}
          \item 董事会。日常经营\bs{决策}。3 - 13 人。\xydy 3 年一届。1 人 1 票。表决\zx{有回避}制度。
                \begin{enumerate}
                  \item 微观。经营计划、投资方案。关键词\zd{制定、方案、执行}。
                  \item 董事会选出\zd{董事长（可兼任高管）}。
                        \begin{enumerate}
                          \item 代表董事会领导公司的方向与策略。产生方式法律留白。
                          \item \zd{决定}内部管理机构的设置。
                        \end{enumerate}
                  \item \zx{任内辞职导致低于法定人数 \yjt 选出新董事前，应继续履行董事职责}。
                  \item \zx{简化}制度。规模小 \hz 股东人数少，可设\zx{执行董事（可兼任经理）}代替董事会。互斥。
                \end{enumerate}
          \item \zx{经理}。董事会负责聘任、薪资、解聘。主持生产经营\bs{管理}工作、\zd{实施董事会决议}。
                \begin{enumerate}
                  \item \zx{提名权}。\al{决定\xhx{聘请}甲担任公司的财务总监 \cuo}。
                  \item 制度设计。\yi \zx{具体规章，自决}。\al{考勤}。\er \zx{基本管理制度，拟定}。\al{财务制度、内部管理机构的设置}。
                \end{enumerate}
          \item \zx{监事会}。3 人以上。监事主席 \yjt 监事过半数、3 年一届。考的少，不做记忆要求。
                \begin{enumerate}
                  \item 代表公司\zd{参加诉讼}。
                  \item \zx{聘请协助调查。费用公司承担}。
                  \item 必须\zx{要有职工代表}，董高不得兼任。
                  \item 决议\zd{半数}通过。
                  \item \zx{简化}制度。规模小 \hz 股东人数少，可设 \zx{1 - 2 名监事}代替监事会。
                \end{enumerate}
          \item \zx{法定代表人}。\zx{董事长、执行董事、经理 3 选 1}。\zx{以公司名义起诉}。
        \end{enumerate}
  \item 一人有限责任公司。一个股东（\zx{自然人 \hz 法人}）。\bs{全资子公司字眼}。\al{\xhx{甲公司}欲单独出资设立一家子公司 \dui}。
        \begin{enumerate}
          \item \zx{自然人只能设立一个}。不设立股东会，或者说这个股东 = 股东会。
          \item 实践中要注重财务规范，不然\zx{容易被打穿承担连带责任}。
        \end{enumerate}
  \item 国有独资。没考过。
\end{enumerate}
