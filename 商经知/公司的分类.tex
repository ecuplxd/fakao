\section{公司的分类}
\begin{enumerate}
  \item 不重要，1 道题或 1 个选项。
  \item 按责任大小。\scx{无限责任、两合、股份两合}、有限责任、股份有限。
        \begin{enumerate}
          \item 股份是否拆分 \yjt 流动性不同 \yjt 规模大小不同 \yjt 监管要求不同。
        \end{enumerate}
  \item 按信用基础。人合、资合（\al{上市公司}）、人合兼资合。\bs{有限公司人合为主，兼有资合}。
  \item 股份能否流通。开放、封闭。
  \item \zd{按组织关系}。
        \begin{enumerate}
          \item 总分公司/机构 \yjt 1 个主体。\al{A 总，B、C 分}。
                \begin{enumerate}
                  \item 非法人组织 \yjt 过错其余的分公司/分机构\zx{共同承担}。
                  \item 题目出现\zd{营业执照}，分公司有程序权利能力（诉讼能力） \yjt 民诉。
                \end{enumerate}
          \item 母\zd{子公司} \yjt 2 个主体。子公司具有企业\zx{法人资格，独立承担}民事责任。
        \end{enumerate}
\end{enumerate}
