\section{普通合伙企业}
\begin{enumerate}
  \item 4 分左右。不考主观题。一张表格足以应对所有合伙企业法的考点。
  \item \bs{人合性！人合性！人合性}！
  \item 4 个共同。\bs{共同出资、共同经营、共享收益、共担风险（无限责任）}。
  \item 适合规模小、风险小的领域，追求成本可控。
  \item 设立。
        \begin{enumerate}
          \item 2 人以上。排除\zd{国有、上市、公益性、社会团体} \yjt 因为无限责任。
          \item 合伙协议。书面 + 签名生效。
          \item 出资。劳务 \yjt 协商。非货币 \yjt 协商 \hz 评估。
          \item 名称。\zd{标明“普通合伙”字样}。
        \end{enumerate}
  \item 财产的管理与便用。原则共同共有。
        \begin{enumerate}
          \item 生产经营所得。因为不纳企业所得税 \yjt 合伙人\bs{分别缴纳所得税}。
          \item \zd{财产份额} \ydy 股权。是权益概念，有身份属性。
                \begin{enumerate}
                  \item 转让。
                        \begin{enumerate}
                          \item 对内。通知即可、没有优先购买权。
                          \item 对外。\zd{一致同意}、有优先权。
                        \end{enumerate}
                  \item 出质、质押 \ydy 对外转让。\yi 登记生效。\er \zd{未经一致同意，一律无效}。该合伙人承担赔偿责任。
                  \item 请求分割。清算之前。
                \end{enumerate}
        \end{enumerate}
  \item 决议。
        \begin{enumerate}
          \item 一般事项。原则 1 人 1 票，不看出资，全体\zd{过半数}。
          \item 特别事项（破坏人合性 + 风险大）。一致同意。基本都涉及人和钱。\bs{我范保协、地、人名、菜}。
                \begin{enumerate}
                  \item 改名、变更经营范围、营业地点、合伙协议。\al{提议\xhx{暂停营业} 1 月，甲乙同意，丙反对。有效}。
                  \item 处分财产。不动产、知识产权、财产份额。\al{出售、抵押}。
                  \item 聘任管理人、吸收新合伙人。
                  \item 提供担保、自我交易。\zd{\sjt 搞些容易迷惑人的选项}。
                \end{enumerate}
        \end{enumerate}
  \item 执行。随意约定，非常灵活；\bs{同等权利，内外有别}。\al{甲乙丙丁均有对外签约权；约定甲乙执行}。
        \begin{enumerate}
          \item 随意、灵活。\al{甲能力不足，不再享有对外签约权，乙对外签约限制为每笔 3W}。
          \item 内外有别。\al{丙代表合伙企业与 xx 签订的合同无效 \cuo}。
          \item \zd{异议}权。事前监督，会导致\zd{执行暂停}。\al{甲单独执行，乙异议权 \dui}。
          \item \zd{监督}权。事后监督，\bs{不执行的主体才有（有执行无监督，互斥）}。\al{甲乙一起执行，丙丁监督 \dui}。
          \item 委托权的撤销。\al{甲违约，其他合伙人（乙丙丁）行使撤销权 \dui}。
          \item 外聘经营管理人。一般事项内随便，\zd{特别事项经授权}。
        \end{enumerate}
  \item 合伙人义务。\yi 绝对禁止 \yjt 竞业禁止。\er \bs{相对禁止} \yjt 自我交易。\bs{效力待定 \dui。无效 \cuo}。
  \item 利润分配、亏损分担。约定（漏人无效） \yjt 合伙协议 \yjt 协商 \yjt \bs{实缴出资比例} \yjt 平均。
        \begin{enumerate}
          \item 合伙债务。先合伙，再个人无限。
          \item 个人债务。\zd{禁止抵销、禁止代位}（你公司的钱 \bdy 你自己的钱）。\al{甲欠乙 10W，乙欠甲合伙 10W}。
                \begin{enumerate}
                  \item 清偿顺序。个人财产 \yjt 所获收益清偿 \yjt 财产份额。
                  \item 法院强制执行。通知全体，没人买又不同意转让份额 \yjt \zd{削减份额/退伙结算}。
                \end{enumerate}
        \end{enumerate}
  \item 新人入伙。
        \begin{enumerate}
          \item 如实告知义务 + 一致同意。告知 = \bs{加入后对之前的债务承担无限责任}。
          \item 继承 \ydy 新入伙。无民继承者经一致同意，可转为有限合伙人。\bs{财产权利和身份权利分别处理}。
          \item 不同意，又不买，视为同意。
        \end{enumerate}
  \item 退伙。退伙事由实际发生之日生效。
        \begin{enumerate}
          \item 自愿。约定期限到了说走就走，没约定提前 30 日。
          \item 法定。\zd{主体资格出问题}。死亡、破产、丧失偿债能力、丧失许可资格、全部份额被强制执行。\zd{丧失民事行为能力可以不退伙，转为有限合伙人}。
          \item 除名。犯错、书面通知、收到生效。\sjt \bs{退伙前发生的合伙企业债务，退伙后不消灭}。
        \end{enumerate}
  \item 特殊普通合伙企业。
        \begin{enumerate}
          \item 特殊的债务承担。故意 \hz 重大过失 \yjt \bs{捣乱的无限连带，无辜的不担责} \yjt 内部追偿。
          \item 特殊的行业限制。专业知识、专业技能。\al{律师、会计、资产评估}。
        \end{enumerate}
\end{enumerate}
