\section{著作权法}
\begin{enumerate}
  \item 取得。本国 \yjt 创作完成之日，\zd{自动}产生。外国、无国籍 \yjt 双重国籍（作者、作品）。同国经。
  \item 客体。法律关系所指向的对象。作品。\zd{体育运动不是作品（竞技性）}。
        \begin{enumerate}
          \item 实质要件。\bs{独创性}，独立创作完成，不抄袭。
          \item \bs{为一定形式表现且感知}。\al{创意、宏大理念 \cuo}。感知，不看能不能出版、发行。\al{淫秽书刊}。
          \item 分类。
                \begin{enumerate}
                  \item 文字作品。\al{诗词、小说、散文、歌赋}。
                  \item \zd{口述}。要求即兴。\al{教授讲座录音、电视讲座}。
                  \item 音乐、戏剧、曲艺、舞蹈、杂技艺术。区别表演活动。
                  \item 美术。审美意义。\al{乐高积木、建筑}。
                  \item 摄影。
                  \item 视听。声音、连续的画面。\al{短视频}。
                  \item \zd{图形、模型}。工程设计图、产品设计图、地图、示意图。\zd{展示、实验、观测}用途的立体模型。
                  \item 软件。
                  \item 符合作品特征的其他智力成果。
                \end{enumerate}
          \item 不予保护。\zd{官方文件及其官方译文}、单纯事实消息、历法、公式。
        \end{enumerate}
  \item 权利内容。
        \begin{enumerate}
          \item 人身权。
                \begin{enumerate}
                  \item 发表（一次性）。作者 \yjt 继承人 \hz 遗赠人 \yjt “书稿”原件所有人。
                  \item 署名。\zx{不得协商、不得随意署名}。\zd{剽窃}。
                  \item 修改、完整（篡改，\al{切片电影}）。
                \end{enumerate}
          \item 财产权。
                \begin{enumerate}
                  \item 复制。可理解为“原样”抄袭。关于\zd{数字化}（载体形式变更）。\al{将小说转换为音频}。
                  \item 发行。出售 \hz 赠予方式\zd{向公众提供}作品的有形载体。
                  \item \bs{出租。仅限软件、视听、录音录像制品}作品。
                  \item 展览。仅限美术、摄影。
                  \item \zd{表演}。\zd{公开}/现场表演、机械/间接表演。许可 + 付费。\al{录播、KTV、MV、商店放歌}。
                  \item 放映。几乎不考。
                  \item 广播。非现场/现场受众 \yjt 广播电台/大喇叭。
                  \item \zd{信息网络传播}。主要针对的是上传到网络的行为。\al{提供链接 \cuo，直接提供内容 \dui}。
                  \item 摄制。\al{根据同名小说拍摄}。
                  \item 改编。\zd{再创作}过程。\al{剧本}。
                  \item 翻译、汇编。
                  \item 其他。许可（专有、非专有）；转让、出质、出版。
                \end{enumerate}
          \item 保护期限。署名、修改、完整（人身权）无限制，其他最后作者\zd{死后 50 年，12/31}。
        \end{enumerate}
  \item 著作权归属。有创作工作，\bs{辅助工作不算}。
        \begin{enumerate}
          \item \zd{署名者}推定为作者。
          \item 单位作者。单位主持 + 单位意志 + 单位承担责任。\al{计算机软件}。
          \item 他人执笔的报告和讲话。执笔人、讲话人、发表人分离。著作权归领导，执笔人获得报酬。
          \item 合作完成自传。有约 \yjt 自传人。
          \item 委托。有约 \yjt 受托人。委托人只能使用。
          \item \zd{演绎}人。改编、注释、翻译、整理。\zd{双重著作权，双许可，双付费}。\al{漫改}。
          \item \zd{汇编}人。双许可，双付费。\al{作文选、诗歌集、论文集}。
          \item 合作。
                \begin{enumerate}
                  \item 不得阻止除\bs{转让、出质、专有许可}以外的其他权利行使。
                  \item 可\zd{分割}的，单独行使。\al{甲作曲，乙作词合作《春风来》，乙不同意，甲\xhx{重新填词}改为《秋风来》}。
                  \item 不可分割的，\bs{著作权、物权共同共有}。
                \end{enumerate}
          \item \zd{视听制作者}。不归制片人，导演署名权，
          \item 职务作品。
                \begin{enumerate}
                  \item 一般职务作品。\zd{2 年内}单位优先使用、相同方式给别人要经单位同意。
                  \item 特殊职务作品。作者署名权。四图一件、三社两台（媒体）。单位承担责任。
                        工程设计图、产品设计图、地图、示意图、计算机软件。报社、期刊社、通讯社、广播电台、电视台。
                \end{enumerate}
          \item 美术作品。原件转移（获得展览权、发表权）不视为著作权转移。\al{甲卖画给乙，乙拍照发微博}。
          \item 著作权集体管理组织。\zd{经授权}行使著作权。解决实践中维权难，授权难问题。
                \begin{enumerate}
                  \item 以自己名义主张权利。调解、仲裁、诉讼。
                  \item 收取使用费。协商 \yjt 主管部门裁决 \hz 起诉。
                \end{enumerate}
        \end{enumerate}
  \item \bs{邻接权/传播者权}。不属于著作权，叫\zd{与著作权有关}的权利，是一类传播者的权利。\bs{许可 + 付费}。
        \begin{enumerate}
          \item 图书、报刊的出版。版式设计。
          \item \zd{表演者}。\bs{复发租网录直播，要收钱}。
                \begin{enumerate}
                  \item 非职务。\bs{录音录像}；他人\bs{现场直播}和公开传送；复制、发行、出租；信息网络传播。\zd{获得报酬}。
                  \item 职务表演。表明身份，保护表演形象。
                \end{enumerate}
          \item 录音录像制作者。\bs{复发租网、要收钱}。复制、发行、出租；信息网络传播。\zd{获得报酬}。
          \item 广播电台、电视台播放。\bs{复网转录}。转播、录制、复制、信息网络传播。\zd{播放后才享权}。\al{偷未播节目带}。
        \end{enumerate}
  \item 不属于侵权。
        \begin{enumerate}
          \item 合理使用。想用就用，用了白用，不能滥用。\zd{是否形成销售替代}。
                \begin{enumerate}
                  \item \zd{自学}、引用、评论、报道、\zd{教学}、公务、图书馆。
                  \item \zd{两端免费}表演。不以营利为目的。\al{年会 \dui、赈灾义演 \cuo}。
                  \item \zd{公共场所陈放}。临摹、绘画、摄影、录像。\al{拓印 \cuo}。
                  \item 通用语言 \yjt \zd{少数民族语言}。单向、国内作品、国内出版。不要求翻译者国籍。
                  \item \zd{阅读障碍}。
                \end{enumerate}
          \item 法定许可。仅限传播者、\bs{只需付费}。
                \begin{enumerate}
                  \item \zd{报刊}转载。\zd{网站、个人转载许可 + 付费}。
                  \item 录音录像制作者\zd{音乐再录}。\al{\xhx{请人}重新翻唱然后录制}。\zd{不包括翻录视听、录像制品 \yjt 侵犯复制权}。
                  \item 广播电台、电视台\zd{播放已发表}的作品。电影例外。
                  \item 义务教育和国家教育规划编写的\zd{教科书}。\zd{题目只说教材默认就是义务教育教科书}。
                \end{enumerate}
        \end{enumerate}
  \item 网络服务提供者。采取的任何行动都要求立即、仅限\zd{扩大范围连带}、权利人\zd{只能请求一次}。同民诉。
\end{enumerate}
