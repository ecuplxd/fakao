\section{著作权法}
\begin{enumerate}
  \item 取得。本国 \yjt 创作完成之日，自动产生；外国、无国籍 \yjt 双重国籍（作者、作品）。伯尔尼公约。同国经。
  \item 客体。法律关系所指向的对象。作品。\zd{体育运动不是作品（竞技性）}。
        \begin{enumerate}
          \item 实质要件。独创性，独立创作完成，不抄袭。
          \item 为一定形式表现且感知。\al{创意、宏大理念}。
          \item 不看能不能出版、发行。\al{淫秽书刊}。
          \item 分类。
                \begin{enumerate}
                  \item 文学。诗词、小说、散文、歌赋。
                  \item 口述。要求即兴。
                  \item 音乐、戏剧、曲艺、舞蹈、杂技。区别表演活动。
                  \item 美术 \yjt 乐高积木、建筑 \yjt 审美意义。
                  \item 摄影。
                  \item 视听。声音、连续的画面。\al{短视频}。
                  \item 图形、模型。地图、工程 \hz 产品示意图；\zd{展示、实验、观测用途}的立体模型。
                  \item 计算机软件。
                \end{enumerate}
          \item 不予保护。\zd{官方文件及其官方译文}；单纯事实消息；历法、公式。
        \end{enumerate}
  \item 内容。
        \begin{enumerate}
          \item 人身权。发表（一次性）、署名、修改、完整（篡改，\al{切片电影}）。
          \item 财产权。
                \begin{enumerate}
                  \item 使用。复制、\zd{数字化}。
                  \item 发行。有形载体。
                  \item \zd{出租}。仅限视听、软件作品。
                  \item 展览。仅限美术、摄影。
                  \item \zd{表演}。\zd{公开}。现场、机械表演（\al{录播、KTV、MV）}。
                  \item 放映。几乎不考。
                  \item 广播。非现场/现场受众 \yjt 广播电台/大喇叭。
                  \item \zd{信息网络传播}。受众可选择特定时间、地点。\al{点播}。表演者、录音录像制作者、广播电台、电视台。
                  \item 设置。\al{根据同名小说拍摄}。
                  \item 改编。剧本。
                  \item 翻译；汇编。
                  \item 许可。专有、非专有。
                  \item 转让；出质；出版。
                \end{enumerate}
        \end{enumerate}
  \item 主体。创作工作、辅助工作。
        \begin{enumerate}
          \item 单位作者。单位主持 + 单位意志 + 单位承担责任。\al{计算机软件}。
          \item 署名者推定为作者。
          \item 他人执笔的报告和讲话。讲话人、发表人、执笔人分离。著作权归领导，执笔人获得报酬。
          \item 合作完成自传。有约从约，无约归自传人。
          \item 著作权集体管理组织。
                \begin{enumerate}
                  \item 维权难，授权难问题。可以\zd{授权}集体管理组织行使著作权。
                  \item 以自己名义主张权利（诉讼、仲裁、调解）。
                  \item 收取使用费。协商 \yjt 主管部门裁决 \hz 起诉。
                \end{enumerate}
          \item 演绎。改编、注释、翻译、整理。\zd{双重著作权，双重许可}。
          \item 汇编。\al{作文选}。
          \item 合作。协商一致；转让、专有许可使用、出质双重同意。分割行使。
          \item 委托。有约 \yjt 受托人。
          \item 视听。\zd{制作者}。不归制片人，导演署名权，
          \item 职务作品。
                \begin{enumerate}
                  \item 一般\blh。2 年内，相同方式（要同意），单位优先使用。
                  \item 特殊\blh。作者署名权。四图一件、三社两台（媒体），单位承担责任。
                  \item 美术作品。原件转移（展览权）不视为著作权转移。
                \end{enumerate}
        \end{enumerate}
  \item 限制。
        \begin{enumerate}
          \item 保护期限。署名、修改、完整除外。最后作者死后 50 年，12.31。
          \item 合理使用。想用就用，用了白用，不能滥用。\zd{是否形成销售替代}。
                \begin{enumerate}
                  \item \zd{自学}、引用、评论、报道、\zd{教学}、公务、馆。
                  \item \zd{免费表演。两端免费，不以营利为目的}。\al{年会}。
                  \item 公共场所陈放。临摹、绘画、摄影、录像。\scx{拓印}。
                  \item 通用语言翻译为少数民族语言。单向；国内作品；国内出版。不要求翻译者国籍。
                  \item \zd{阅读障碍}。
                \end{enumerate}
          \item 法定许可使用。不能白嫖；只能是传播者；已发表的。不包括视听、录像制品。
                \begin{enumerate}
                  \item 报刊转载。主体不包括网站。
                  \item 音乐再录。不是第一次被录制。
                  \item 广播电台、电视台特权。电影例外。
                  \item \zd{义务教育}教科书。
                \end{enumerate}
          \item 邻接权/传播者权。不属于著作权，是一类人（传播者）的权利。
                \begin{enumerate}
                  \item 出版者。
                  \item \zd{表演者}。他人现场直播公开传送、录音录像、复制发行出租、信息网络。
                  \item 职务表演。表明身份，保护表演形象。
                  \item 录像制品。操作设备。网络、复发、出租。获酬权。
                \end{enumerate}
        \end{enumerate}
  \item 侵权。实际损失 \hz 违法所得 \yjt 使用费 \yjt 自由裁量（500 - 500W）。
\end{enumerate}
