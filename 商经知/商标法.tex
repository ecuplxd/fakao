\section{商标法}
\begin{enumerate}
  \item 3 分左右。最简单。
  \item 向\bs{商标评审委员会}\zd{申请宣告无效}（\al{撤销 \cuo}），宣告无效由\bs{商标局}做出。
  \item 注册商标。能区别。文字、图像、声音（\al{QQ 来消息声音}）、数字、符号。
  \item 分类。
        \begin{enumerate}
          \item 商品商标、服务商标。
          \item 集体商标、证明商标。都是组织控制。
                \begin{enumerate}
                  \item 开放/封闭、证明身份/品质。\al{沙县小吃；绿色食品}。
                  \item 证明商标表明使用该商标的经营者 \hz 服务的提供者\zd{不属于同一组织}。
                \end{enumerate}
        \end{enumerate}
  \item 根本不得使用。
        \begin{enumerate}
          \item 国家、官方、红十字、民族歧视相关。
          \item 欺骗 + 误认。\al{百消灵}。
          \item 品格不高，不良影响。\al{老榴芒}。
          \item 县级以上地名。
                \begin{enumerate}
                  \item 地名还有其他含义，且印象高于地名。\al{长安汽车}。
                  \item 作为集体、证明商标的一部分。\al{烟台苹果、山西陈醋}。
                  \item 已使用的继续有效。\al{青岛啤酒}。
                \end{enumerate}
        \end{enumerate}
  \item 不能注册，可以默默用。
        \begin{enumerate}
          \item 通用名称。\al{水牌矿泉水}。
          \item 仅仅表示商品质量、主要原料、功能、用途、重量、数量等特点。\al{发光牌灯泡 \cuo，光明牌灯泡 \dui}。
          \item 缺乏显著特征。例外。经过使用后获得显著特征。\al{两面针牙膏}。
          \item 含有虚假地理标志。例外。作为集体商标、证明商标存在。
        \end{enumerate}
  \item 申请。
        \begin{enumerate}
          \item 自愿原则。例外。烟草强制注册。
          \item \zd{在先申请为原则}，在先使用为补充。申请日上午、下午不分前后，均算同一日。
          \item 不以使用为目的的\zd{恶意}注册。蹭热点，抢热度。防御商标。\zd{商标体系架构}。\al{老干爹}。
          \item \zd{抢注}。5 年。
          \item 外国人必须委托依法设立的商标\zd{代理}机构。告知、不能助纣为虐、只能帮别人。
          \item 分类填报。\zd{全品类注册}。为提高效率，可一标多类（一份申请表搞定）。
          \item 不予注册。绝对理由，谁都可以随时提异议/申请无效。相对理由，仅限被损害的主体。
        \end{enumerate}
  \item 转让。\zd{要一并转让}，公告之日生效。\al{xx 碳酸饮料，xx 果汁}。
  \item 许可。\bs{商标局备案对抗}，公告之日生效。
  \item 无效宣告。\zd{注册后}发现不符合规定。绝对，相对。
        \begin{enumerate}
          \item \bs{注册日 5 年内}。\al{乔丹案}。
          \item 自始无效，溯及既往。\zd{已执行、已履行除外}。
        \end{enumerate}
  \item 使用管理。
        \begin{enumerate}
          \item 自行改变相关信息 \yjt 工商责令限期改正 \yjt 商标局撤销。\bs{就这一个撤销事由，其他都是错的}。
          \item 丧失显著性 \hz 连续 3 年不用。冷静期 1 年。\al{阿司匹林}。
        \end{enumerate}
  \item \bs{侵权}。\zd{不包括购买和使用行为}。考点集中在这里。
        \begin{enumerate}
          \item \zd{专用权以核准注册的商标和核定使用的商品为限}。
          \item \zd{假冒}。\al{钻孔茅台酒}。
          \item \zd{仿冒}。很像，但不完全一样。\zd{善意侵权不赔钱}。\al{周住洗衣粉、雷碧}。
          \item \zd{制造、销售}。销售不管善意恶意，有价无价。\al{甲商场将假冒“茉莉花”牌纸手帕作为赠品进行促销活动}。
          \item 反向假冒。换掉商标 + 再次销售。\al{倒入自制啤酒桶，自制“侠客”牌散装啤酒}。
          \item 帮助侵权。要求明知。\al{甲公司明知某企业生产假冒啤酒，仍向其出租仓库}。
          \item 其他。当字号用、类似商品、突出使用。
        \end{enumerate}
  \item \bs{驰名商标}。为相关公众所熟知（动态过程）。\zd{题干明确说“xx 被认定为驰名商标”的才算驰名商标}。
        \begin{enumerate}
          \item 不能写入判决主文。
          \item 不得将“驰名商标”字样用于包装、装潢、广告宣传 \yjt 驰名商标 \bdy 商品质量。
          \item 没注册 \ydy 一般的注册商标。\bs{不得请求赔偿。抢注不受 5 年限制}。
          \item 注册了更强的保护。
                \begin{enumerate}
                  \item 跨类别保护。他人再注册 \yjt \bs{申请宣告无效不受 5 年限制}。
                  \item 把别人商标当字号用，相同类别商品，突出。用字多的侵犯商标权，字少的适用反不正当竞争。
                \end{enumerate}
        \end{enumerate}
  \item 不构成侵权。
        \begin{enumerate}
          \item 描述性使用。\al{下列公司都在用飞书}。
          \item 说明商品 \hz 服务用途。\al{本墨水适用惠普打印机}。
          \item 购买之后，商标用尽。
          \item 通用元素的正常使用。\al{农夫山泉、恒大冰泉}。
          \item 先使用/先用权。权利人可以要求附加适当区别标识。
        \end{enumerate}
\end{enumerate}
